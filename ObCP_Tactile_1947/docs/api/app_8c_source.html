385/**
386 * @brief Centrer une chaîne dans la largeur du TFT.
387 *
388 * La largeur est calculée avec une avance de six pixels par caractère et le facteur
389 * d'agrandissement. La primitive texte limite ensuite les écritures à la bande active.
390 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
391 * @param[in] y Ordonnée dans le repère écran, en pixels.
392 * @param[in] text Chaîne valide terminée par zéro, conservée pendant l’appel.
393 * @param[in] scale Facteur entier positif d’agrandissement des glyphes.
394 * @param[in] color Couleur encodée au format RGB565.
395 */
396static void centered_text(Gfx *g, int y, const char *text, unsigned scale, uint16_t color)
397{
398 int width = ((int)strlen(text) * 6 - 1) * (int)scale;
399 gfx_text(g, (LCD_WIDTH - width) / 2, y, text, scale, color);
400}

402/**
403 * @brief Dessiner un bouton rectangulaire et son libellé.
404 *
405 * Le rectangle, le fond, le cadre et le texte sont recomposés à chaque rendu. Cette
406 * fonction dessine seulement. Les tests tactiles utilisent le même rectangle ailleurs.
407 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
408 * @param[in] r Rectangle écran, avec bord droit et bas exclus.
409 * @param[in] label Libellé valide terminé par zéro.
410 * @param[in] fill Couleur RGB565 du fond du bouton.
411 * @param[in] ink Couleur RGB565 du libellé.
412 */
413static void button(Gfx *g, Rect r, const char *label, uint16_t fill, uint16_t ink)
414{
415 gfx_fill(g, r, fill);
416 gfx_outline(g, r, COLOR_PANEL_HI);
417 int width = (int)strlen(label) * 6 - 1;
418 gfx_text(g, r.x + (r.w - width) / 2, r.y + (r.h - 7) / 2, label, 1, ink);
419}

501/*
502 * Effectuer une transition vers un autre écran.
503 *
504 * Arrête les voix, ferme les dialogues et annule les gestes en cours. Prépare le jeu ou
505 * l'assistant demandé, puis invalide l'écran. Les objets de dessin et les réglages restent
506 * dans App.
507 */
508void app_open(App *a, Screen screen)
509{
510 audio_stop(&a->audio);
511 a->screen = screen;
512 a->modal = MODAL_NONE;
513 a->notice = NULL;
514 a->test_color = 0;
515 a->paint.stroke = false;
516 a->keys = false;
517 game_load_level(a, 0);
518 if (screen == SCREEN_TOUCH_CAL) {
519 touch_cal_start(&a->touch_cal_session);
520 a->cal_can_cancel = touch_cal_get(&a->touch_cal, a->diag.touch_controller) != NULL;
521 a->diag.hits = 0;
522 } else if (screen == SCREEN_TILT) {
523 a->tilt_guide = (TiltGuide){0};
524 }
525 damage_full(&a->dirty);
526}

578/**
579 * @brief Traiter Jouer, Pause, Rejouer et Menu principal.
580 *
581 * Les sorties sont examinées avant la disponibilité de l'accéléromètre. Une défaite
582 * recharge le même niveau, une victoire avance au suivant. L'éclat de course laisse
583 * d'abord sa place à l'animation.
584 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
585 * @param[in] x Abscisse dans le repère écran, en pixels.
586 * @param[in] y Ordonnée dans le repère écran, en pixels.
587 */
588static void game_touch(App *a, int x, int y)
589{
590 GamePhase *phase = game_phase(a);
591 if (phase == NULL) {
592 return;
593 }
594 if (a->screen == SCREEN_RACE && a->race.crash_ms > 0U) {
595 return; /* Le bouton Menu du bandeau reste traité par app_touch(). */
596 }
597 if (*phase != GAME_RUN && rect_contains(game_menu, x, y)) {
598 app_open(a, SCREEN_MENU);
599 return;
600 }
601 if ((*phase == GAME_RUN || *phase == GAME_PAUSE) && rect_contains(pause_btn, x, y)) {
602 if (*phase == GAME_RUN) {
603 *phase = GAME_PAUSE;
604 } else if (*phase == GAME_PAUSE) {
605 *phase = GAME_RUN;
606 }
607 damage_full(&a->dirty);
608 } else if (rect_contains(play_btn, x, y) && *phase != GAME_RUN) {
609 unsigned level = game_level(a);
610 if (*phase == GAME_WON) {
611 game_load_level(a, level + 1U < GAME_LEVELS ? level + 1U : 0U);
612 } else if (*phase == GAME_LOST) {
613 game_load_level(a, level);
614 }
615 if (a->tilt.calibrated && a->tilt.valid) {
616 *phase = GAME_RUN;
617 }
618 damage_full(&a->dirty);
619 }

641/**
642 * @brief Traiter un appui dans une boîte de dialogue.
643 *
644 * Distingue palette, composantes RVB, diamètre, confirmations et emplacements.
645 * L'application du RVB modifie une case de palette et invalide tous les pixels qui peuvent

647 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
648 * @param[in] x Abscisse dans le repère écran, en pixels.

651static void modal_touch(App *a, int x, int y)
652{

656 show_modal(a, MODAL_NONE);

680 }
681 return;
682 }
683 if (a->modal == MODAL_COLORS && rect_contains(edit_apply, x, y)) {
684 uint16_t color = a->paint.palette[a->paint.ink];
685 a->rgb[0] = (uint8_t)(((color >> 11) * 255U + 15U) / 31U);
686 a->rgb[1] = (uint8_t)((((color >> 5) & 63U) * 255U + 31U) / 63U);

688 show_modal(a, MODAL_RGB);
689 return;
690 }
691 if ((a->modal == MODAL_COLORS || a->modal == MODAL_BRUSH) && rect_contains(edit_cancel, x, y)) {
692 show_modal(a, MODAL_NONE);
693 return;
694 }
695 if (a->modal != MODAL_COLORS && rect_contains(cancel, x, y)) {

707 }

712 show_modal(a, MODAL_NONE);
713 damage_add(&a->dirty, paint_canvas);
714 } else {
715 app_open(a, SCREEN_MENU);
716 }
717 }
718 } else {

728 request_store(a, job, i);
729 break;
730 }
731 }
732 }

1012/*
1013 * Distribuer un événement tactile à l'écran approprié.
1014 *
1015 * Les travaux flash et le calibrage brut ont leur propre entrée. Les dialogues reçoivent
1016 * ensuite les gestes, puis le canevas, le retour et les écrans. Cet ordre évite qu'un
1017 * appui serve deux commandes.
1018 */
1019void app_touch(App *a, TouchEvent e)
1020{
1021 if (a->job != JOB_NONE || e.kind == TOUCH_NONE || a->screen == SCREEN_TOUCH_CAL ||
1022 a->screen == SCREEN_SPLASH) {
1023 return;
1024 }
1025 if (e.kind == TOUCH_DOWN && a->notice != NULL) {
1026 set_notice(a, NULL);
1027 }
1028 if (a->modal != MODAL_NONE) {
1029 if (a->modal == MODAL_RGB && (e.kind == TOUCH_DOWN || e.kind == TOUCH_MOVE)) {
1030 rgb_drag(a, e.x, e.y);
1031 }
1032 if (e.kind == TOUCH_UP) {
1033 audio_key_release(&a->audio);
1034 } else if (e.kind == TOUCH_DOWN) {
1035 modal_touch(a, e.x, e.y);
1036 }
1037 return;
1038 }
1039 if (a->screen == SCREEN_PAINT) {
1040 paint_touch(a, e);
1041 return;
1042 }
1043 if (e.kind == TOUCH_DOWN && a->screen != SCREEN_MENU && rect_contains(home, e.x, e.y)) {
1044 bool child = a->screen == SCREEN_DIAG || a->screen == SCREEN_TILT || a->screen == SCREEN_RESPONSE;
1045 app_open(a, child ? SCREEN_SETTINGS : SCREEN_MENU);
1046 return;
1047 }
1048 if (a->screen == SCREEN_RESPONSE && (e.kind == TOUCH_DOWN || e.kind == TOUCH_MOVE)) {
1049 response_touch(a, e);
1050 return;
1051 }
1052 if (a->screen == SCREEN_GROOVE) {
1053 groove_touch(a, e);
1054 } else if (e.kind == TOUCH_DOWN) {
1055 if (a->screen == SCREEN_MENU) {
1056 for (unsigned i = 0; i < 6; ++i) {
1057 if (rect_contains(menu_card(i), e.x, e.y)) {
1058 app_open(a, (Screen)(SCREEN_PAINT + i));
1059 break;
1060 }
1061 }
1062 } else if (a->screen == SCREEN_SETTINGS) {
1063 settings_touch(a, e.x, e.y);
1064 } else if (a->screen == SCREEN_DIAG) {
1065 diag_touch(a, e.x, e.y);

1242/*
1243 * Exécuter la requête de stockage en attente.
1244 *
1245 * Les formats propres au dessin, au motif, au tactile et aux axes sont traités séparément.
1246 * Le résultat prépare le message de fin et sera utilisé par board.c pour l'impulsion de
1247 * LED.
1248 */
1249StoreResult app_store_task(App *a, Storage *storage)
1250{
1251 if (a->job == JOB_NONE) {
1252 return STORE_EMPTY;
1253 }
1254 if (a->job == JOB_TOUCH_CAL_SAVE) {
1255 StoreResult saved = touch_cal_save(&a->touch_cal, storage, OBCP_LCD_MADCTL);
1256 a->job = JOB_NONE;
1257 app_open(a, SCREEN_SETTINGS);
1258 set_notice(a, saved == STORE_OK ? "CALIBRAGE ENREGISTRE" : "CAL EN RAM / ERREUR FLASH");
1259 return saved;
1260 }
1261 if (a->job == JOB_TILT_SAVE) {
1262 a->tilt.axes = a->tilt_guide.axes;
1263 tilt_project(&a->tilt, a->tilt.axes, &a->tilt.x, &a->tilt.y);
1264 const uint8_t data[AXES_BYTES] = {AXES_VERSION, OBCP_LCD_MADCTL, a->tilt.axes};

1266 a->job = JOB_NONE;

1268 set_notice(a, saved == STORE_OK ? "AXES ENREGISTRES" : "AXES EN RAM / ERREUR FLASH");

1270 }
1271 if (a->job == JOB_PREFERENCES_SAVE) {
1272 const uint8_t data[3] = {1, a->tilt.response_x, a->tilt.response_y};
1273 StoreResult saved = storage_save(storage, SAVE_PREFERENCES, data, sizeof(data));
1274 a->job = JOB_NONE;
1275 app_open(a, SCREEN_SETTINGS);
1276 set_notice(a, saved == STORE_OK ? "REACTIVITE ENREGISTREE" : "REACTIVITE EN RAM / ERREUR FLASH");

1278 }
1279 StoreResult result = STORE_EMPTY;
1280 if (a->job == JOB_DRAW_SAVE || a->job == JOB_DRAW_OPEN) {
1281 SaveObject object = a->slot == 0U ? SAVE_DRAW_1 : SAVE_DRAW_2;
1282 if (a->job == JOB_DRAW_SAVE) {
1283 result = paint_save(&a->paint, storage, object);
1284 } else {
1285 result = paint_load(&a->paint, storage, object);

1289 a->paint.modified = false;
1290 }
1291 } else if (a->job == JOB_MUSIC_SAVE) {
1292 PatternBook candidate = a->book;
1293 candidate.pattern[a->slot] = a->audio.pattern;
1294 candidate.present |= (uint8_t)(1U << a->slot);
1295 uint8_t data[SEQ_BOOK_BYTES];
1296 book_encode(&candidate, data);
1297 result = storage_save(storage, SAVE_MUSIC, data, sizeof(data));
1298 if (result == STORE_OK) {
1299 a->book = candidate;
1300 }
1301 } else if ((a->book.present & (1U << a->slot)) != 0U) {
1302 a->audio.pattern = a->book.pattern[a->slot];
1303 result = STORE_OK;
1304 damage_full(&a->dirty);
1305 }
1306 const char *message = "ERREUR FLASH";
1307 if (result == STORE_OK) {
1308 message = a->job == JOB_DRAW_SAVE || a->job == JOB_MUSIC_SAVE ? "ENREGISTRE" : "CHARGE";
1309 } else if (result == STORE_EMPTY) {
1310 message = "EMPLACEMENT VIDE OU INVALIDE";
1311 }
1312 a->job = JOB_NONE;
1313 set_notice(a, message);
1314 return result;
1315}
