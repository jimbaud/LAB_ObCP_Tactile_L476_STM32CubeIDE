1/**
2 * @file obcp_storage.h
3 * @brief Format d’objet A/B, CRC32 et validation transactionnelle.
4 * @ingroup obcp_storage
5 *
6 * Chaque objet possède deux emplacements sur des pages distinctes. La génération et les
7 * CRC identifient la copie exploitable. Une nouvelle copie reçoit son marqueur final dans
8 * un double mot encore vierge après contrôle de toutes ses données.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */

51/* Toutes les adresses sont des offsets en octets dans la réserve du linker.
52 * program doit écrire un double mot aligné. erase reçoit des pages entières.
53 * Les fonctions sont synchrones et appelées uniquement dans la boucle principale. */
54/**
55 * @brief Interface physique de stockage injectable dans les tests.
56 *
57 * Toutes les adresses sont des offsets dans la réserve du linker. Les callbacks synchrones
58 * sont appelés dans la boucle principale. Le port STM32 et la flash simulée partagent ce
59 * contrat.
60 */
61typedef struct {
62 void *context; /**< Contexte opaque du port. Peut être NULL pour la flash physique unique. */
63 bool (*read)(void *context, uint32_t offset, void *data, size_t count); /**< Copier count octets depuis offset vers data. true signifie que toute la lecture a réussi, y compris le contrôle ECC du port STM32. */
64 bool (*erase)(void *context, uint32_t offset, uint32_t count); /**< Effacer count octets à partir de offset. Offset et count doivent représenter des pages complètes de 2 Kio. true confirme l'effacement. */
65 bool (*program)(void *context, uint32_t offset, const uint8_t bytes[8]); /**< Programmer les huit octets bytes sur un offset aligné de huit octets encore vierges. true confirme le succès physique du double mot. */
66} Storage;

68/**
69 * @brief Lire un objet dont la longueur exacte est connue.
70 *
71 * Sélectionne d'abord la copie récente valide, puis vérifie que sa longueur égale celle
72 * demandée avant de copier les octets vers l'appelant.
73 * @param[in,out] s Port de stockage et ses callbacks synchrones. Pointeur valide.
74 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
75 * @param[out] data Tampon de destination valide. Sa capacité est donnée par size, capacity
76 * ou count.
77 * @param[in] size Longueur du bloc en octets.
78 * @return STORE_OK, STORE_EMPTY si absent ou de longueur différente, STORE_IO ou STORE_INVALID.
79 */
80StoreResult storage_read(Storage *s, SaveObject object, void *data, size_t size);
81/* Lire la dernière copie, dont la longueur peut appartenir à un ancien format.
82 * actual reçoit la longueur enregistrée. Une capacité insuffisante laisse data intact. */
83/**
84 * @brief Lire un objet en découvrant sa longueur de format.
85 *
86 * Le tampon est donné avec sa capacité. La taille réellement enregistrée est fournie
87 * seulement au succès. Une copie trop grande est rejetée avant la copie des données.
88 * @param[in,out] s Port de stockage et ses callbacks synchrones. Pointeur valide.
89 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
90 * @param[out] data Tampon de destination valide. Sa capacité est donnée par size, capacity
91 * ou count.
92 * @param[in] capacity Capacité du tampon de destination en octets.
93 * @param[out] actual Longueur réelle enregistrée, écrite seulement au succès.
94 * @return STORE_OK, STORE_EMPTY, STORE_IO ou STORE_INVALID.
95 */
96StoreResult storage_read_sized(Storage *s, SaveObject object, void *data, size_t capacity, size_t *actual);
97/**
98 * @brief Écrire et valider la copie inactive d'un objet.
99 *
100 * Garde la copie active, efface l'autre, écrit l'en-tête incomplet et les données, relit
101 * les CRC puis programme le marqueur final. La génération active est reconnue même si la
102 * nouvelle longueur de format diffère.
103 * @param[in,out] s Port de stockage et ses callbacks synchrones. Pointeur valide.
104 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
105 * @param[in] data Bloc source valide de size octets.
106 * @param[in] size Longueur du bloc en octets.
107 * @return STORE_OK au succès, STORE_IO en cas d'échec physique ou STORE_INVALID pour une requête hors contrat.
108 */
109StoreResult storage_save(Storage *s, SaveObject object, const void *data, size_t size);
110/**
111 * @brief Calculer le CRC32 d'un bloc de données.

114 * tests contrôlent notamment la chaîne 123456789.

116 * @param[in] size Longueur du bloc en octets.
117 * @return CRC32 du bloc fourni.
118 */
119uint32_t storage_crc32(const void *data, size_t size);
