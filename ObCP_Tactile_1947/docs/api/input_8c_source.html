17/*
18 * Passer des coordonnées natives capacitives au paysage.
19 *
20 * Vérifie les plages portrait du FT62xx, puis applique la rotation et éventuellement le
21 * demi-tour choisi dans la configuration. Les sorties sont des pixels du TFT.
22 */
23bool touch_landscape(uint16_t raw_x, uint16_t raw_y, bool flipped, int16_t *x, int16_t *y)
24{
25 if (raw_x >= 240U || raw_y >= 320U) {
26 return false;
27 }
28 /* Le FT62xx exprime un portrait 240 × 320. La rotation échange X et Y puis inverse la nouvelle abscisse. */
29 *x = (int16_t)(319U - raw_y);
30 *y = (int16_t)raw_x;
31 if (flipped) {
32 *x = (int16_t)(319 - *x);
33 *y = (int16_t)(239 - *y);
34 }
35 return true;
36}

38/*
39 * Produire un événement à partir du contact et de sa position.
40 *
41 * Compare l'état reçu au précédent. Le premier appui donne DOWN, un changement de position
42 * donne MOVE et la disparition du contact donne UP. L'état mémorisé est mis à jour.
43 */
44TouchEvent touch_update(TouchState *s, bool down, int16_t x, int16_t y)
45{
46 TouchKind kind = TOUCH_NONE;
47 if (down) {
48 if (!s->down) {
49 kind = TOUCH_DOWN;
50 } else if (s->x != x || s->y != y) {
51 kind = TOUCH_MOVE;
52 }
53 s->x = x;
54 s->y = y;
55 } else if (s->down) {
56 kind = TOUCH_UP;
57 }
58 s->down = down;
59 return (TouchEvent){kind, s->x, s->y};
60}

89/**
90 * @brief Choisir la valeur médiane de trois coordonnées.
91 *
92 * Les comparaisons écartent une valeur aberrante isolée sans faire la moyenne des
93 * extrêmes.
94 * @param[in] a Valeur numérique à comparer.
95 * @param[in] b Valeur numérique à comparer.
96 * @param[in] c Troisième valeur à classer pour la médiane.
97 * @return Valeur du milieu après classement.
98 */
99static uint16_t median(uint16_t a, uint16_t b, uint16_t c)
100{
101 uint16_t low = a < b ? a : b;
102 uint16_t high = a > b ? a : b;
103 return c < low ? low : (c > high ? high : c);
104}

106/*
107 * Stabiliser un contact résistif et ses coordonnées.
108 *
109 * Deux mesures confirment les fronts. Une histoire de trois points permet ensuite une
110 * médiane par axe. Le filtre conserve son état d'un appel au suivant.
111 */
112RawTouch touch_filter(ResistiveFilter *filter, RawTouch raw)
113{
114 /* Confirmer chaque front avec deux mesures. Un échantillon parasite
115 * au lever du stylet ne termine pas le trait à lui seul. */
116 if (raw.down != filter->current.down) {
117 if (!filter->changing) {
118 filter->changing = true;
119 return filter->current;
120 }
121 filter->used = 0;
122 filter->next = 0;
123 }
124 filter->changing = false;
125 if (raw.down) {
126 filter->history[filter->next] = raw.point;
127 /* Le modulo fait de ces trois cases un historique circulaire sans déplacement de tableau. */
128 filter->next = (uint8_t)((filter->next + 1U) % 3U);
129 if (filter->used < 3U) {
130 ++filter->used;
131 }
132 if (filter->used == 3U) {
133 const RawPoint *p = filter->history;
134 raw.point.x = median(p[0].x, p[1].x, p[2].x);
135 raw.point.y = median(p[0].y, p[1].y, p[2].y);
136 }
137 }
138 filter->current = raw;
139 return raw;
140}

142/*
143 * Reconnaître le maintien tactile demandé au reset.
144 *
145 * Le premier contact arme une mesure de durée. Un relâchement précoce consomme le geste.
146 * Au seuil de 1,3 s, l'événement de calibrage est émis une seule fois.
147 */
148BootTouchEvent touch_boot_sample(TouchBoot *boot, bool down, uint32_t now)
149{
150 if (!boot->armed) {
151 return BOOT_TOUCH_NONE;
152 }
153 if (!down) {
154 /* Une dalle libre à la première mesure donne un démarrage normal.
155 * Un appui de secours relâché trop tôt est consommé en entier. */
156 boot->armed = false;
157 return boot->held ? BOOT_TOUCH_USED : BOOT_TOUCH_NONE;
158 }
159 if (!boot->held) {
160 boot->held = true;
161 boot->since = now;
162 }
163 if (now - boot->since >= TOUCH_BOOT_HOLD_MS) {
164 boot->armed = false;
165 return BOOT_TOUCH_CALIBRATE;
166 }
167 return BOOT_TOUCH_USED;
168}

181/*
182 * Transformer le niveau de SW1 en appui court ou long.
183 *
184 * Valide un changement après 28 ms. Le court est émis au relâchement, le long après 900 ms
185 * et une seule fois. Les différences de ticks non signés tolèrent leur débordement.
186 */
187ButtonEvent button_update(ButtonState *b, bool pressed, uint32_t now)
188{
189 if (pressed != b->raw) {
190 b->raw = pressed;
191 b->changed_at = now;
192 }
193
194 /* Valider le front seulement si le contact est resté stable assez longtemps. */
195 if (b->stable != b->raw && now - b->changed_at >= BUTTON_DEBOUNCE_MS) {
196 b->stable = b->raw;
197 if (b->stable) {
198 b->pressed_at = b->changed_at;
199 b->consumed = false;
200 } else if (!b->consumed) {
201 b->consumed = true;
202 return b->changed_at - b->pressed_at >= BUTTON_HOLD_MS ? BUTTON_LONG : BUTTON_SHORT;
203 }
204 }
205 if (b->stable && b->raw && !b->consumed && now - b->pressed_at >= BUTTON_HOLD_MS) {
206 b->consumed = true;
207 return BUTTON_LONG;
208 }
209 return BUTTON_NONE;
210}

237/**
238 * @brief Borner une commande de jeu.
239 *
240 * La saturation évite qu'une forte inclinaison dépasse les amplitudes prévues par les
241 * modèles physiques.
242 * @param[in] x Commande normalisée à limiter à l’intervalle −1 à +1.
243 * @return Valeur limitée à l'intervalle −1 à +1.
244 */
245static float clamp_command(float x)
246{
247 if (x > 1.0f) {
248 return 1.0f;
249 }
250 if (x < -1.0f) {
251 return -1.0f;
252 }
253 return x;
254}

256/**
257 * @brief Appliquer la zone morte et mettre une accélération à l'échelle.
258 *
259 * La petite variation autour du neutre donne zéro. Au-delà, la commande est continue et
260 * bornée. Le gain de réactivité sera appliqué ensuite.
261 * @param[in] g Accélération en g avant normalisation.
262 * @return Commande normalisée avant gain de réactivité.
263 */
264static float normalize(float g)
265{
266 const float dead_g = 0.045f;
267 if (g > dead_g) {
268 return clamp_command((g - dead_g) / 0.40f);
269 }
270 if (g < -dead_g) {
271 return clamp_command((g + dead_g) / 0.40f);
272 }
273 return 0.0f;
274}

276/*
277 * Projeter le filtre brut dans un repère écran choisi.
278 *
279 * Permute les composantes, applique leurs signes, la zone morte puis les gains X/Y. Un
280 * capteur non prêt donne deux commandes nulles. L'aperçu peut ainsi essayer le repère
281 * proposé avant sa validation.
282 */
283void tilt_project(const Tilt *t, uint8_t axes, float *x, float *y)
284{
285 if (!t->calibrated || !t->valid) {
286 *x = *y = 0.0f;
287 return;
288 }
289 /* Chaque option est un bit indépendant. Le ET binaire teste la permutation sans modifier les inversions. */
290 bool swap = (axes & TILT_SWAP_XY) != 0U;
291 float dx = swap ? t->filtered_y : t->filtered_x;
292 float dy = swap ? t->filtered_x : t->filtered_y;
293 /* Le gain agit après la zone morte pour garder un neutre stable.
294 * À forte réactivité, une petite inclinaison donne une grande commande. */
295 *x = clamp_command(normalize((axes & TILT_INVERT_X) != 0U ? -dx : dx) *
296 (0.6f + 0.2f * t->response_x));
297 *y = clamp_command(normalize((axes & TILT_INVERT_Y) != 0U ? -dy : dy) *
298 (0.6f + 0.2f * t->response_y));
299}

301/*
302 * Valider, recentrer et filtrer une mesure XYZ.
303 *
304 * Contrôle l'ordre de grandeur de la norme. Les 32 premières mesures valides définissent
305 * le neutre. Le filtre IIR suit ensuite les écarts, avec le coefficient propre au bon axe
306 * écran.
307 */
308void tilt_sample(Tilt *t, bool valid, float g_x, float g_y, float g_z)
309{
310 float norm_sq = g_x * g_x + g_y * g_y + g_z * g_z;
311 t->valid = valid && norm_sq >= 0.35f && norm_sq <= 3.0f;
312 if (!t->valid) {
313 t->x = 0.0f;
314 t->y = 0.0f;
315 return;
316 }
317 if (!t->calibrated) {
318 /* La référence et le filtre restent dans le repère du capteur. */
319 t->sum_x += g_x;
320 t->sum_y += g_y;
321 ++t->neutral_count;
322 if (t->neutral_count == 32U) {
323 t->zero_x = t->sum_x / 32.0f;
324 t->zero_y = t->sum_y / 32.0f;
325 t->calibrated = true;
326 }
327 return;
328 }
329
330 /* La réactivité X/Y se règle dans le repère de l'écran. Retrouver
331 * l'alpha de chaque axe brut avant d'appliquer le filtre IIR à 100 Hz. */
332 bool swap = (t->axes & TILT_SWAP_XY) != 0U;
333 float alpha_x = 0.10f + 0.08f * ((swap ? t->response_y : t->response_x) - 1U);
334 float alpha_y = 0.10f + 0.08f * ((swap ? t->response_x : t->response_y) - 1U);
335 t->filtered_x += alpha_x * (g_x - t->zero_x - t->filtered_x);
336 t->filtered_y += alpha_y * (g_y - t->zero_y - t->filtered_y);
337 tilt_project(t, t->axes, &t->x, &t->y);
338}

365/**
366 * @brief Classer une inclinaison dominante dans le repère brut.
367 *
368 * Une composante doit dépasser 0,13 g et l'autre rester assez faible. Les diagonales sont
369 * donc rejetées pour ne pas confondre l'axe demandé.
370 * @param[in] t Référence, filtre et commandes de l’accéléromètre, consulté sans
371 * modification. Pointeur valide.
372 * @return ±1 pour X brut, ±2 pour Y brut, zéro sans direction nette.
373 */
374static int8_t direction(const Tilt *t)
375{
376 float x = amplitude(t->filtered_x), y = amplitude(t->filtered_y);
377 /* Les valeurs ±1 désignent le X brut. ±2 désignent le Y brut.
378 * Exiger une direction dominante pour écarter une inclinaison diagonale. */
379 if (x >= 0.13f && y <= 0.55f * x) {
380 return t->filtered_x > 0.0f ? 1 : -1;
381 }
382 if (y >= 0.13f && x <= 0.55f * y) {
383 return t->filtered_y > 0.0f ? 2 : -2;
384 }
385 return 0;
386}

388/**
389 * @brief Exiger seize observations successives d'une même pose.
390 *
391 * Un changement ou une pose indéfinie remet le compteur à zéro. La validation a lieu une
392 * seule fois par série stable.
393 * @param[in,out] guide Progression et candidat de l’assistant d’orientation. Pointeur valide.
394 * @param[in] pose Code de pose candidate, zéro si indéfinie.
395 * @return true quand le seuil de stabilité est atteint.
396 */
397static bool stable_pose(TiltGuide *guide, int8_t pose)
398{
399 if (guide->direction != pose || pose == 0) {
400 guide->direction = pose;
401 guide->stable = 0;
402 }
403 if (pose == 0 || ++guide->stable < 16U) {
404 return false;
405 }
406 guide->stable = 0;
407 return true;
408}

410/*
411 * Faire avancer la reconnaissance droite puis bas.
412 *
413 * Après le neutre, le guide valide une inclinaison à droite, exige le retour à plat, puis
414 * exige l'autre axe pour le bas. Les signes et la permutation sont déduits des poses
415 * stabilisées.
416 */
417bool tilt_guide_sample(TiltGuide *guide, const Tilt *tilt)
418{
419 if (!tilt->valid || !tilt->calibrated) {
420 guide->stable = 0;
421 return false;
422 }
423 switch (guide->phase) {
424 case TILT_GUIDE_NEUTRAL:
425 guide->phase = TILT_GUIDE_RIGHT;
426 return true;
427 case TILT_GUIDE_RIGHT: {
428 int8_t pose = direction(tilt);
429 if (stable_pose(guide, pose)) {
430 guide->right = pose;
431 guide->phase = TILT_GUIDE_FLAT;
432 return true;
433 }
434 break;
435 }
436 case TILT_GUIDE_FLAT:
437 if (stable_pose(guide, amplitude(tilt->filtered_x) < 0.065f &&
438 amplitude(tilt->filtered_y) < 0.065f ? 3 : 0)) {
439 guide->phase = TILT_GUIDE_DOWN;
440 return true;
441 }
442 break;
443 case TILT_GUIDE_DOWN: {
444 int8_t pose = direction(tilt);
445 if (pose == guide->right || pose == -guide->right) {
446 pose = 0;
447 }
448 if (stable_pose(guide, pose)) {
449 guide->axes = (guide->right == 2 || guide->right == -2 ? TILT_SWAP_XY : 0U) |
450 (guide->right < 0 ? TILT_INVERT_X : 0U) |
451 (pose < 0 ? TILT_INVERT_Y : 0U);
452 guide->phase = TILT_GUIDE_PREVIEW;
453 return true;
454 }
455 break;
456 }
457 default: break;
458 }
459 return false;
460}
