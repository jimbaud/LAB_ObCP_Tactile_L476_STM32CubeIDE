2 * @file ili9341.c
3 * @brief Commandes du TFT ILI9341 et transport SPI1 DMA.
4 * @ingroup obcp_drivers
5 *
6 * Le TFT utilise CS sur D10 et D/C sur D9. L’initialisation et les petites commandes sont
7 * synchrones. Les pixels sont transmis par DMA1 canal 3. Le STMPE610 peut emprunter SPI1
8 * uniquement entre ces bandes.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */

15 * Voir docs/PROVENANCE.md. Le transport SPI et le DMA sont propres à l'ObCP. */

17 * @brief État volatile du transfert, partagé entre la boucle et les callbacks SPI.

30enum {
31 CMD_SWRESET = 0x01, /**< Commande ILI9341 de réinitialisation logicielle. */
32 CMD_SLEEP_OUT = 0x11, /**< Commande de sortie du mode sommeil. */
33 CMD_DISPLAY_ON = 0x29, /**< Commande d’activation de l’affichage. */
34 CMD_COLUMN = 0x2A, /**< Commande de définition des colonnes de la fenêtre. */
35 CMD_PAGE = 0x2B, /**< Commande de définition des lignes de la fenêtre. */
36 CMD_RAM_WRITE = 0x2C, /**< Commande précédant les octets RGB565 de la fenêtre. */
37 CMD_READ_ID = 0xD3, /**< Commande de lecture de l’identifiant du contrôleur TFT. */
38};

43/**
44 * @brief Appliquer le niveau logique de sélection du TFT.
45 *
46 * Le périphérique est sélectionné quand CS vaut zéro. Une sélection basse doit couvrir la
47 * commande et tous les octets de sa bande.
48 * @param[in] level Indice de niveau à base zéro. Une valeur hors plage choisit le premier
49 * niveau.
50 */
51static void select_lcd(GPIO_PinState level)
52{
53 HAL_GPIO_WritePin(TFT_CS_GPIO_Port, TFT_CS_Pin, level);
54}

56/**
57 * @brief Émettre une commande en plaçant D/C à zéro.
58 *
59 * CS passe bas, l'octet de commande est envoyé puis D/C repasse haut pour les paramètres.
60 * La désélection reste à la charge du code appelant.
61 * @param[in] command Octet de commande du protocole.
62 * @return true si l'octet de commande a été envoyé.
63 */
64static bool begin_command(uint8_t command)
65{
66 select_lcd(GPIO_PIN_RESET);
67 HAL_GPIO_WritePin(TFT_DC_GPIO_Port, TFT_DC_Pin, GPIO_PIN_RESET);
68 bool ok = HAL_SPI_Transmit(&hspi1, &command, 1, 3) == HAL_OK;
69 HAL_GPIO_WritePin(TFT_DC_GPIO_Port, TFT_DC_Pin, GPIO_PIN_SET);
70 return ok;
71}

73/**
74 * @brief Envoyer une commande avec ses paramètres puis libérer CS.
75 *
76 * Le message complet reste sélectionné et utilise un délai HAL limité. count peut être nul
77 * lorsqu'une commande ne demande aucun paramètre.
78 * @param[in] code Code de commande ILI9341.
79 * @param[in] params Paramètres de la commande, count octets. NULL est permis si count vaut
80 * zéro.
81 * @param[in] count Nombre d’octets de l’opération ou de la transaction.
82 * @return true si commande et paramètres ont été transmis.
83 */
84static bool command(uint8_t code, const uint8_t *params, uint16_t count)
85{
86 bool ok = begin_command(code);
87 if (ok && count > 0U) {
88 ok = HAL_SPI_Transmit(&hspi1, params, count, 3) == HAL_OK;
89 }
90 select_lcd(GPIO_PIN_SET);
91 return ok;
92}

94/**
95 * @brief Lire le code du contrôleur TFT à une fréquence réduite.
96 *
97 * Sauvegarde le prescaler SPI1, le réduit pour la lecture de D3 puis rétablit sa valeur.
98 * Les octets d'identité sont recomposés dans l'ordre du protocole.
99 * @return Identifiant sur 16 bits, ou zéro si l'échange échoue.
100 */
101static uint16_t read_id(void)
102{
103 uint32_t speed = READ_BIT(hspi1.Instance->CR1, SPI_CR1_BR);
104 /* La lecture série du contrôleur est plus lente que l'écriture. */
105 __HAL_SPI_DISABLE(&hspi1);
106 MODIFY_REG(hspi1.Instance->CR1, SPI_CR1_BR, SPI_BAUDRATEPRESCALER_16);
107 uint8_t tx[4] = {0};
108 uint8_t rx[4] = {0};
109 bool ok = begin_command(CMD_READ_ID) &&
110 HAL_SPI_TransmitReceive(&hspi1, tx, rx, sizeof(rx), 4) == HAL_OK;
111 select_lcd(GPIO_PIN_SET);
112 __HAL_SPI_DISABLE(&hspi1);
113 MODIFY_REG(hspi1.Instance->CR1, SPI_CR1_BR, speed);
114 return ok ? (uint16_t)((uint16_t)rx[2] << 8 | rx[3]) : 0U;
115}

117/*
118 * Initialiser le TFT en paysage RGB565.
119 *
120 * Lit l'identifiant, applique la séquence d'alimentation et de gamma, choisit MADCTL et
121 * sort le contrôleur du sommeil. Cette étape unique peut utiliser les délais requis au
122 * démarrage.
123 */
124bool lcd_init(uint16_t *id)
125{
126 static const struct { uint8_t cmd, n, data[15]; } init[] = {
127 {0xEF,3,{0x03,0x80,0x02}}, {0xCF,3,{0x00,0xC1,0x30}},
128 {0xED,4,{0x64,0x03,0x12,0x81}}, {0xE8,3,{0x85,0x00,0x78}},
129 {0xCB,5,{0x39,0x2C,0x00,0x34,0x02}}, {0xF7,1,{0x20}}, {0xEA,2,{0x00,0x00}},
130 {0xC0,1,{0x23}}, {0xC1,1,{0x10}}, {0xC5,2,{0x3E,0x28}}, {0xC7,1,{0x86}},
131 {0x36,1,{OBCP_LCD_MADCTL}}, {0x3A,1,{0x55}}, /* paysage et RGB565 */
132 {0xB1,2,{0x00,0x18}}, {0xB6,3,{0x08,0x82,0x27}}, {0xF2,1,{0x00}},
133 {0x26,1,{0x01}},
134 {0xE0,15,{0x0F,0x31,0x2B,0x0C,0x0E,0x08,0x4E,0xF1,0x37,0x07,0x10,0x03,0x0E,0x09,0x00}},
135 {0xE1,15,{0x00,0x0E,0x14,0x03,0x11,0x07,0x31,0xC1,0x48,0x08,0x0F,0x0C,0x31,0x36,0x0F}},
136 };
137 *id = 0;
138 HAL_Delay(12);
139 if (!command(CMD_SWRESET, NULL, 0)) {
140 return false;
141 }
142 HAL_Delay(150);
143 *id = read_id();
144 for (size_t i = 0; i < sizeof(init) / sizeof(init[0]); ++i) {
145 if (!command(init[i].cmd, init[i].data, init[i].n)) {
146 return false;
147 }
148 }
149 if (!command(CMD_SLEEP_OUT, NULL, 0)) {
150 return false;
151 }
152 HAL_Delay(120);
153 return command(CMD_DISPLAY_ON, NULL, 0);
154}

156/**
157 * @brief Définir un intervalle de colonnes ou de lignes du TFT.
158 *
159 * Le protocole du contrôleur reçoit début et fin inclus sur deux octets chacun. L'appelant
160 * transforme donc une largeur en dernière coordonnée moins un.
161 * @param[in] cmd Commande de fenêtre de colonnes ou de pages.
162 * @param[in] start Première coordonnée incluse de la fenêtre.
163 * @param[in] end Dernière coordonnée incluse du protocole ILI9341.
164 * @return true si la commande de fenêtre est transmise.
165 */
166static bool window(uint8_t cmd, int start, int end)
167{
168 uint8_t data[4] = {(uint8_t)(start >> 8), (uint8_t)start,
169 (uint8_t)(end >> 8), (uint8_t)end};
170 return command(cmd, data, sizeof(data));
171}

173/**
174 * @brief Commencer le transfert DMA des pixels d'un rectangle.
175 *
176 * Définit la fenêtre et RAM_WRITE, puis garde CS actif jusqu'au callback de fin. Le tampon
177 * fourni doit rester intact tant que le transport est BUSY.
178 * @param[in] context Contexte opaque transmis sans allocation. Le port physique peut ne pas
179 * l’utiliser.
180 * @param[in] r Rectangle écran, avec bord droit et bas exclus.
181 * @param[in] data Tampon valide pour la longueur demandée par size, len ou count.
182 * @param[in] count Nombre d’octets de l’opération ou de la transaction.
183 * @return true si le DMA prend effectivement la propriété du tampon.
184 */
185static bool send_band(void *context, Rect r, const uint8_t *data, uint16_t count)
186{
187 (void)context;
188 if (state == TRANSFER_BUSY) {
189 return false;
190 }
191 bool ok = window(CMD_COLUMN, r.x, r.x + r.w - 1) &&
192 window(CMD_PAGE, r.y, r.y + r.h - 1) &&
193 begin_command(CMD_RAM_WRITE);
194 if (ok) {
195 /* CS reste à zéro pendant l'envoi de toute la bande. */
196 started_ms = HAL_GetTick();
197 state = TRANSFER_BUSY;
198 if (HAL_SPI_Transmit_DMA(&hspi1, data, count) == HAL_OK) {
199 return true;
200 }
201 }
202 select_lcd(GPIO_PIN_SET);
203 state = TRANSFER_ERROR;
204 return false;
205}

207/**
208 * @brief Consulter le transfert et surveiller son délai maximal.
209 *
210 * Un transfert qui dépasse 50 ms est interrompu et libère CS. Renderer pourra demander une
211 * image complète pour reprendre un écran cohérent.
212 * @param[in] context Contexte opaque transmis sans allocation. Le port physique peut ne pas
213 * l’utiliser.
214 * @return État BUSY, DONE ou ERROR du transport.
215 */
216static TransferState poll_band(void *context)
217{
218 (void)context;
219 if (state == TRANSFER_BUSY && HAL_GetTick() - started_ms > 50U) {
220 (void)HAL_SPI_Abort(&hspi1);
221 select_lcd(GPIO_PIN_SET);
222 state = TRANSFER_ERROR;
223 }
224 return state;
225}

227/*
228 * Exposer les callbacks de transport à Renderer.
229 *
230 * Le moteur graphique n'a ainsi pas besoin de connaître SPI1 ni la HAL. Le contexte opaque
231 * est inutilisé par ce port unique.
232 */
233DisplayPort lcd_port(void)
234{
235 return (DisplayPort){NULL, send_band, poll_band};
236}

238/*
239 * Vérifier si SPI1 peut servir le tactile résistif.
240 *
241 * Le transport ne doit plus être BUSY, la HAL doit être prête et le bit matériel BSY doit
242 * être effacé.
243 */
244bool lcd_spi_available(void)
245{
246 return state != TRANSFER_BUSY && HAL_SPI_GetState(&hspi1) == HAL_SPI_STATE_READY &&
247 !__HAL_SPI_GET_FLAG(&hspi1, SPI_FLAG_BSY);
248}

250/**
251 * @brief Terminer une bande après l'interruption DMA SPI.
252 *
253 * Pour SPI1, désélectionne le TFT et publie DONE. Aucun dessin n'est effectué dans
254 * l'interruption. Les autres handles sont ignorés.
255 * @param[in,out] spi Handle HAL du bus SPI déjà initialisé. Pointeur valide.
256 * @note Contexte interruption. Garder ce traitement court et sans opération flash.
257 */
258void HAL_SPI_TxCpltCallback(SPI_HandleTypeDef *spi)
259{
260 if (spi == &hspi1) {
261 select_lcd(GPIO_PIN_SET);
262 state = TRANSFER_DONE;
263 }
264}

276 if (spi == &hspi1) {
