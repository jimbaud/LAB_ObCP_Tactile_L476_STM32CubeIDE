169/**
170 * @brief Convertir une cellule de carte en rectangle écran.
171 *
172 * Ajoute les marges de la scène au produit de l'indice par la taille de tuile.
173 * @param[in] col Indice de colonne dans la carte.
174 * @param[in] row Indice de ligne dans la carte.
175 * @return Rectangle de la cellule, en pixels.
176 */
177static Rect cell(int col, int row)
178{
179 return (Rect){(int16_t)(LEFT + col * TILE), (int16_t)(TOP + row * TILE), TILE, TILE};
180}

185 * Le départ est placé dans la case intérieure de coordonnées colonne un et ligne un.

196/**
197 * @brief Limiter une valeur flottante à un intervalle.
198 *
199 * Le même helper borne une vitesse ou trouve le point le plus proche sur une tuile murale.
200 * @param[in] v Valeur flottante à limiter.
201 * @param[in] low Borne inférieure de saturation.
202 * @param[in] high Borne supérieure de saturation.
203 * @return Valeur saturée entre low et high.
204 */
205static float clamp(float v, float low, float high)
206{
207 return v < low ? low : (v > high ? high : v);
208}

210/*
211 * Vérifier si le disque de la bille touche une zone interdite.
212 *
213 * Limite les cases examinées à l'emprise du disque. Pour chaque mur, le point de rectangle
214 * le plus proche du centre permet de tester la distance au carré.
215 */
216bool maze_solid(const Maze *m, float x, float y)
217{
218 if (x < LEFT + BALL_RADIUS || y < TOP + BALL_RADIUS ||
219 x >= LEFT + COLS * TILE - BALL_RADIUS || y >= TOP + ROWS * TILE - BALL_RADIUS) {
220 return true;
221 }
222 int c0 = ((int)x - BALL_RADIUS - LEFT) / TILE;
223 int c1 = ((int)x + BALL_RADIUS - LEFT) / TILE;
224 int r0 = ((int)y - BALL_RADIUS - TOP) / TILE;
225 int r1 = ((int)y + BALL_RADIUS - TOP) / TILE;
226 for (int row = r0; row <= r1; ++row) {
227 for (int col = c0; col <= c1; ++col) {
228 if (maps[m->level][row][col] == '#') {
229 Rect tile = cell(col, row);
230 float dx = x - clamp(x, tile.x, tile.x + tile.w);
231 float dy = y - clamp(y, tile.y, tile.y + tile.h);
232 if (dx * dx + dy * dy < BALL_RADIUS * BALL_RADIUS) {
233 return true;
234 }
235 }
236 }
237 }
238 return false;
239}

241/**
242 * @brief Calculer l'emprise de la bille pour le rendu.
243 *
244 * Le rayon et un pixel de bord sont inclus afin d'effacer complètement l'ancienne image.
245 * @param[in] m État physique et temporel du labyrinthe, consulté sans modification.
246 * Pointeur valide.
247 * @return Rectangle englobant la bille.
248 */
249static Rect ball_bounds(const Maze *m)
250{
251 return (Rect){(int16_t)((int)m->x - BALL_RADIUS), (int16_t)((int)m->y - BALL_RADIUS),
252 2 * BALL_RADIUS + 1, 2 * BALL_RADIUS + 1};
253}

268/*
269 * Convertir le temps restant en secondes affichables.
270 *
271 * Arrondit vers le haut pour que la dernière seconde soit visible. La valeur est nulle dès
272 * que la limite du niveau est atteinte.
273 */
274uint32_t maze_seconds_left(const Maze *m)
275{
276 uint32_t limit = limits_ms[m->level];
277 return m->elapsed_ms >= limit ? 0 : (limit - m->elapsed_ms + 999U) / 1000U;
278}

280/*
281 * Intégrer la physique et le temps sur vingt millisecondes.
282 *
283 * Calcule les vitesses avec inertie, avance séparément X et Y pour permettre le glissement
284 * contre un mur, puis teste arrivée et expiration. Les emprises modifiées sont invalidées.
285 */
286void maze_step(Maze *m, float tilt_x, float tilt_y, Damage *damage)
287{
288 if (m->phase != GAME_RUN) {
289 return;
290 }
291 Rect old = ball_bounds(m);
292 /* v est en px/s. À 100 px/s et 20 ms, on parcourt au plus 2 px
293 * par axe. Le test séparé laisse la bille glisser le long d'un mur. */
294 const float dt = GAME_STEP_MS / 1000.0f;
295 m->vx = clamp(0.96f * m->vx + 550.0f * tilt_x * dt, -100.0f, 100.0f);
296 m->vy = clamp(0.96f * m->vy + 550.0f * tilt_y * dt, -100.0f, 100.0f);
297 float x = m->x + dt * m->vx;
298 if (maze_solid(m, x, m->y)) {
299 m->vx = 0.0f;
300 } else {
301 m->x = x;
302 }
303 float y = m->y + dt * m->vy;
304 if (maze_solid(m, m->x, y)) {
305 m->vy = 0.0f;
306 } else {
307 m->y = y;
308 }
309 Rect current = ball_bounds(m);
310 if (old.x != current.x || old.y != current.y) {
311 damage_add(damage, old);
312 damage_add(damage, current);
313 }
314 m->elapsed_ms += GAME_STEP_MS;
315 if (rect_contains(maze_finish(), (int)m->x, (int)m->y)) {
316 m->phase = GAME_WON;
317 } else if (m->elapsed_ms >= limits_ms[m->level]) {
318 m->phase = GAME_LOST;
319 }
320 if (m->phase != GAME_RUN) {
321 damage_full(damage);
322 }
323}

325/*
326 * Dessiner les murs, les cases spéciales et la bille.
327 *
328 * Les rangées hors de la bande sont sautées. Le niveau sélectionne la carte, tandis que
329 * les positions physiques ne sont jamais modifiées par ce rendu.
330 */
331void maze_draw(const Maze *m, Gfx *g)
332{
333 for (int row = 0; row < ROWS; ++row) {
334 if (TOP + (row + 1) * TILE <= g->clip.y || TOP + row * TILE >= g->clip.y + g->clip.h) {
335 continue;
336 }
337 for (int col = 0; col < COLS; ++col) {
338 Rect tile = cell(col, row);
339 char type = maps[m->level][row][col];
340 uint16_t color = type == '#' ? RGB565(40,68,103) : RGB565(16,27,43);
341 gfx_fill(g, tile, color);
342 if (type == '#') {
343 gfx_outline(g, tile, RGB565(51,88,123));
344 } else if (type == 'G' || type == 'S') {
345 gfx_disc(g, tile.x + 8, tile.y + 8, 6,
346 type == 'G' ? RGB565(62,233,170) : RGB565(57,112,170));
347 }
348 }
349 }
350 gfx_disc(g, (int)m->x, (int)m->y, BALL_RADIUS, RGB565(255,201,73));
351 gfx_disc(g, (int)m->x - 1, (int)m->y - 2, 2, RGB565(255,241,186));
352}
