1/**
2 * @file touch_drivers.h
3 * @brief Reconnaissance des shields et façade tactile commune.
4 * @ingroup obcp_drivers
5 *
6 * L’application lit toujours un RawTouch et un état TouchRead. La façade choisit un
7 * protocole capacitif I²C, résistif SPI ou résistif I²C. Un état WAIT conserve la
8 * possibilité de servir d’autres tâches entre deux étapes.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */

19/**
20 * @brief Vérifier l'identifiant de famille du tactile capacitif.
21 *
22 * Le registre 0xA3 est lu. Les identifiants 0x06, 0x36 et 0x64 sont admis. Le résultat de
23 * sortie commence à zéro pour éviter une identité résiduelle.
24 * @param[out] chip Pointeur valide recevant l’identifiant lu du composant.
25 * @return true si une signature compatible est reçue.
26 */
27bool ft6206_probe(uint8_t *chip);
28/**
29 * @brief Lire l'état et les coordonnées du premier doigt.
30 *
31 * Une rafale contient TD_STATUS et les deux coordonnées sur 12 bits. L'événement natif de
32 * relâchement ne doit pas devenir un nouvel appui. Les coordonnées sont écrites seulement
33 * si le doigt est présent.
34 * @param[out] pressed Pointeur valide recevant true lorsqu’un doigt est présent.
35 * @param[out] raw_x Coordonnée native de sortie, écrite seulement lorsque le contact est
36 * présent.
37 * @param[out] raw_y Coordonnée native de sortie, écrite seulement lorsque le contact est
38 * présent.
39 * @return true si la transaction réussit, y compris lorsque pressed vaut false.
40 */
41bool ft6206_sample(bool *pressed, uint16_t *raw_x, uint16_t *raw_y);
42/**
43 * @brief Reconnaître 0x0811 et configurer le tactile.
44 *
45 * Les modes SPI zéro puis un sont essayés à 1 MHz. La phase reconnue est conservée pour
46 * les acquisitions suivantes. Les réglages du TFT sont restaurés à la sortie.
47 * @param[out] chip Pointeur valide recevant l’identifiant lu du composant.
48 * @return true si l'identifiant attendu et l'initialisation sont valides.
49 */
50bool stmpe610_probe(uint16_t *chip);
51/**
52 * @brief Acquérir une observation entre deux bandes TFT.
53 *
54 * BUSY donne WAIT pour rendre la main. Sinon le pilote emprunte le SPI, lit ses points,
55 * puis remet les paramètres initiaux quelle que soit l'issue.
56 * @param[out] sample Structure RawTouch valide recevant l’observation physique.
57 * @return État tactile OK, WAIT, EMPTY ou ERROR.
58 */
59TouchRead stmpe610_sample(RawTouch *sample);
60/**
61 * @brief Tester le protocole de conversion du TSC2007.
62 *
63 * Remet l'état d'acquisition à zéro, demande TEMP0 avec mise en sommeil et attend une
64 * milliseconde au seul démarrage. Le contrôle suppose un périphérique dédié à 0x48.
65 * @return true si une réponse de conversion compatible est reçue.
66 */
67bool tsc2007_probe(void);
68/**
69 * @brief Avancer d'une étape dans l'observation résistive.
70 *
71 * Alterner commande, attente puis lecture permet de servir l'écran et l'audio. Z1 et Z2
72 * décident du contact avec hystérésis. X et Y sont lus seulement quand le contact est
73 * retenu.
74 * @param[out] sample Structure RawTouch valide recevant l’observation physique.
75 * @return TOUCH_READ_WAIT pendant une conversion, OK au résultat complet ou ERROR sur incident de bus.
76 */
77TouchRead tsc2007_sample(RawTouch *sample);
