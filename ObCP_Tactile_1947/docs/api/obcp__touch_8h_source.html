1/**
2 * @file obcp_touch.h
3 * @brief Reconnaissance des shields et façade tactile commune.
4 * @ingroup obcp_drivers
5 *
6 * L’application lit toujours un RawTouch et un état TouchRead. La façade choisit un
7 * protocole capacitif I²C, résistif SPI ou résistif I²C. Un état WAIT conserve la
8 * possibilité de servir d’autres tâches entre deux étapes.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */
14#ifndef OBCP_TOUCH_H
15#define OBCP_TOUCH_H

17#include <stdbool.h>

20/**
21 * @brief Contrôleurs tactiles reconnus sur les shields compatibles.
22 */
23typedef enum {
24 TOUCH_CTRL_NONE, /**< Aucun contrôleur compatible reconnu. */
25 TOUCH_CTRL_FT62XX, /**< Capacitif FT62xx sur I²C à 0x38. */
26 TOUCH_CTRL_STMPE610, /**< Résistif sur SPI1, identifié par 0x0811. */
27 TOUCH_CTRL_TSC2007, /**< Résistif sur I²C à 0x48, reconnu par conversion. */
28} TouchController;
29
30/**
31 * @brief Résultat de la reconnaissance tactile.
32 *
33 * Le type pilote la façade. L’identifiant complète l’affichage quand le composant possède
34 * un registre d’identité.
35 */
36typedef struct {
37 TouchController controller; /**< Famille de contrôleur ou TOUCH_CTRL_NONE. */
38 uint16_t id; /**< Identifiant natif. Zéro pour TSC2007, reconnu par protocole. */
39} TouchDevice;
40
41/**
42 * @brief Position native avant calibrage et rotation.
43 *
44 * Les familles résistives livrent des conversions 12 bits. Le capacitif livre les
45 * coordonnées portrait propres à sa dalle.
46 */
47typedef struct {
48 uint16_t x; /**< Coordonnée horizontale dans les unités natives du contrôleur. */
49 uint16_t y; /**< Coordonnée verticale dans les unités natives du contrôleur. */
50} RawPoint;
51/**
52 * @brief Observation physique d’un contrôleur tactile.
53 *
54 * Le type de contrôleur définit les unités. La transformation vers les pixels intervient
55 * seulement après filtrage et choix du profil.
56 */
57typedef struct {
58 RawPoint point; /**< Coordonnées natives du contact quand down est vrai. */
59 uint16_t z1; /**< Pression Z1 du TSC2007 ou mesure Z du STMPE610. Zéro pour FT62xx. */
60 uint16_t z2; /**< Pression Z2 du TSC2007, sinon zéro. */
61 bool down; /**< true si le pilote retient un contact présent. */
62} RawTouch;
63
64/* WAIT cède la main au DMA ou à l'ADC. EMPTY reporte la lecture au
65 * prochain échantillonnage. Les deux conservent le dernier contact. */
66/**
67 * @brief États de progression d’une acquisition physique.
68 */
69typedef enum {
70 TOUCH_READ_OK, /**< Observation complète, avec ou sans contact. */
71 TOUCH_READ_WAIT, /**< Le pilote attend une conversion ou la disponibilité de SPI1. */
72 TOUCH_READ_EMPTY, /**< FIFO vide sous un contact, conserver le dernier état logique. */
73 TOUCH_READ_ERROR, /**< Erreur de bus ou réponse incompatible. */
74} TouchRead;
75
76/* Retourne false si le bus SPI est encore occupé par une bande du TFT. */
77/**
78 * @brief Reconnaître le contrôleur tactile présent.
79 *
80 * Vérifie d'abord que SPI1 peut être utilisé, essaie les signatures FT62xx puis STMPE610,
81 * enfin le protocole TSC2007. Une recherche terminée peut produire TOUCH_CTRL_NONE.
82 * @param[in,out] device Type et identifiant du contrôleur tactile. Pointeur valide.
83 * @return true si la recherche a pu être effectuée. false quand le SPI est occupé.
84 */
85bool touch_probe(TouchDevice *device);
86/**
87 * @brief Acquérir un RawTouch avec le pilote reconnu.
88 *
89 * Remet d'abord la structure de sortie à zéro. Le contrôleur détermine le pilote, et les
90 * états asynchrones sont propagés à board.c.
91 * @param[in] device Type et identifiant du contrôleur tactile, consulté sans modification.
92 * Pointeur valide.
93 * @param[out] sample Structure RawTouch valide recevant l’observation physique.
94 * @return TOUCH_READ_OK, WAIT, EMPTY ou ERROR.
95 */
96TouchRead touch_read(const TouchDevice *device, RawTouch *sample);
97/**
98 * @brief Classer le type de tactile.
99 *
100 * La reconnaissance TSC2007 ou STMPE610 indique qu'un profil affine est requis avant la
101 * conversion en pixels.
102 * @param[in] controller Famille de contrôleur tactile.
103 * @return true pour les deux contrôleurs résistifs pris en charge.
104 */
105bool touch_resistive(TouchController controller);
106/**
107 * @brief Fournir un nom de contrôleur à l'interface.
108 *
109 * Le texte appartient aux constantes du programme et suit la valeur reconnue par
110 * touch_probe().
111 * @param[in] controller Famille de contrôleur tactile.
112 * @return Chaîne constante descriptive du contrôleur.
113 */
114const char *touch_name(TouchController controller);
