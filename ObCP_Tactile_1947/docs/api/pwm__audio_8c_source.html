140/**
141 * @brief Préparer et lancer le timer avec son anneau circulaire.
142 *
143 * Vérifie ARR et RCR, remplit l'anneau au centre de la PWM, branche les callbacks, démarre
144 * le DMA puis la sortie complémentaire et sa requête de mise à jour.
145 * @return true si les démarrages DMA et timer ont réussi.
146 */
147static bool start(void)
148{
149 DMA_HandleTypeDef *handle = dma();
150 if (handle == NULL || htim1.Init.Period != 1023U || htim1.Init.RepetitionCounter != 3U) {
151 return false;
152 }
153 epoch = 0;
154 served = 0;
155 dma_error = false;
156 for (unsigned i = 0; i < 2U * HALF_SAMPLES; ++i) {
157 ring[i] = AUDIO_CENTER;
158 }
159 /* Le DMA alterne deux moitiés. Chaque interruption incrémente epoch ; la synthèse reste dans la boucle principale. */
160 handle->XferHalfCpltCallback = half_done;
161 handle->XferCpltCallback = half_done;
162 handle->XferErrorCallback = dma_fault;
163 __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_3, AUDIO_CENTER);
164 __HAL_TIM_SET_COUNTER(&htim1, 0U);
165 htim1.Instance->EGR = TIM_EGR_UG;
166 __HAL_TIM_CLEAR_FLAG(&htim1, TIM_FLAG_UPDATE);
167 if (HAL_DMA_Start_IT(handle, (uint32_t)(uintptr_t)ring,
168 (uint32_t)(uintptr_t)&htim1.Instance->CCR3,
169 2U * HALF_SAMPLES) != HAL_OK) {
170 return false;
171 }
172 if (HAL_TIMEx_PWMN_Start(&htim1, TIM_CHANNEL_3) != HAL_OK) {
173 (void)HAL_DMA_Abort(handle);
174 return false;
175 }
176 connect_timer();
177 __HAL_TIM_ENABLE_DMA(&htim1, TIM_DMA_UPDATE);
178 return true;
179}

181/**
182 * @brief Arrêter le transport et revenir au niveau de repos.
183 *
184 * Déconnecte d'abord PB1, désactive la requête DMA, arrête la sortie PWM puis interrompt
185 * le DMA. L'état running est remis à false.
186 */
187static void stop(void)
188{
189 drive_low();
190 __HAL_TIM_DISABLE_DMA(&htim1, TIM_DMA_UPDATE);
191 (void)HAL_TIMEx_PWMN_Stop(&htim1, TIM_CHANNEL_3);
192 (void)HAL_DMA_Abort(dma());
193 running = false;
194}

196/*
197 * Produire au bon moment la moitié libre de l'anneau.
198 *
199 * Compare epoch à served, synthétise dans staging puis relit le compteur DMA. La copie se
200 * fait seulement si le DMA utilise toujours l'autre moitié avec une marge de huit
201 * échantillons. Les échéances perdues sont comptées.
202 */
203void pwm_audio_task(Audio *audio, bool needed)
204{
205 if (!needed) {
206 if (running) {
207 stop();
208 }
209 failed = false;
210 return;
211 }
212 if (!running && !failed) {
213 running = start();
214 failed = !running;
215 if (failed) {
216 ++missed;
217 }
218 }
219 if (!running) {
220 return;
221 }
222 if (dma_error) {
223 ++missed;
224 stop();
225 failed = true;
226 return;
227 }
228
229 uint32_t observed = epoch;
230 if (observed == served) {
231 return;
232 }
233 missed += observed - served - 1U;
234 served = observed;
235 /* Après une fin de première moitié, le DMA lit la seconde : la première est donc la moitié à refaire. La parité suit cette alternance. */
236 unsigned half = (observed & 1U) != 0U ? 0U : 1U;
237 audio_fill(audio, staging, HALF_SAMPLES);
238
239 /* On synthétise dans staging. Avant la copie, vérifier que le DMA
240 * lit toujours l'autre moitié et qu'il reste assez de temps. */
241 uint32_t left = __HAL_DMA_GET_COUNTER(dma());
242 bool writable = half == 0U ? left <= HALF_SAMPLES && left > COPY_GUARD
243 : left > HALF_SAMPLES + COPY_GUARD;
244 /* Une interruption survenue pendant le calcul invalide notre autorisation. Le contrôle du compteur ajoute une marge avant la copie. */
245 if (epoch == observed && writable) {
246 memcpy(&ring[half * HALF_SAMPLES], staging, sizeof(staging));
247 } else {
248 ++missed;
249 }
250}
