39/*
40 * Vérifier la géométrie réelle avant de modifier la flash.
41 *
42 * Contrôle capacité 1 Mio, absence de permutation des banques, pages de 2 Kio et bornes de
43 * la réserve. Ces contrôles relient les hypothèses du linker au composant lu.
44 */
45bool obcp_flash_ready(void)
46{
47 uintptr_t start = (uintptr_t)&__obcp_storage_start;
48 uintptr_t end = (uintptr_t)&__obcp_storage_end;
49 /* Sur le L476 à 1 Mio, les deux banques font 512 Kio. DUALBANK
50 * concerne les variantes 256/512 Kio. FB_MODE doit rester à zéro. */
51 return *(const volatile uint16_t *)FLASHSIZE_BASE == 1024U &&
52 (SYSCFG->MEMRMP & SYSCFG_MEMRMP_FB_MODE) == 0U &&
53 FLASH_PAGE_SIZE == OBCP_FLASH_PAGE &&
54 start >= BANK_2_ADDRESS && end <= UINT32_C(0x08100000) &&
55 end - start == OBCP_STORAGE_BYTES && start % OBCP_FLASH_PAGE == 0U;
56}

58/*
59 * Reconnaître une NMI provoquée par une erreur ECC double.
60 *
61 * Le bit ECCD est acquitté et un indicateur volatile mémorise l'erreur pour la lecture en
62 * cours. Les autres causes de NMI restent à traiter par le gestionnaire généré.
63 */
64bool obcp_flash_ecc_nmi(void)
65{
66 if ((FLASH->ECCR & FLASH_ECCR_ECCD) == 0U) {
67 return false;
68 }
69 __HAL_FLASH_CLEAR_FLAG(FLASH_FLAG_ECCD);
70 ecc_failed = true;
71 return true;
72}

74/**
75 * @brief Vérifier une plage d'offsets dans la réserve.
76 *
77 * Le test utilise count inférieur à la taille restante, après le contrôle d'offset. Cette
78 * forme évite un débordement de offset + count.
79 * @param[in] offset Offset en octets depuis le début de la réserve flash.
80 * @param[in] count Nombre d’octets de l’opération ou de la transaction.
81 * @return true si toute la plage reste dans la réserve.
82 */
83static bool in_range(uint32_t offset, size_t count)
84{
85 return offset <= OBCP_STORAGE_BYTES && count <= OBCP_STORAGE_BYTES - offset;
86}

88/**
89 * @brief Lire une plage mémoire et surveiller son ECC.
90 *
91 * Calcule l'adresse à partir du symbole du linker, copie vers le tampon puis consulte
92 * l'indicateur NMI. Le contexte opaque du port n'est pas utilisé.
93 * @param[in] context Contexte opaque transmis sans allocation. Le port physique peut ne pas
94 * l’utiliser.
95 * @param[in] offset Offset en octets depuis le début de la réserve flash.
96 * @param[out] data Tampon de destination valide. Sa capacité est donnée par size, capacity
97 * ou count.
98 * @param[in] count Nombre d’octets de l’opération ou de la transaction.
99 * @return true si les bornes et la lecture ECC sont valides.
100 */
101static bool flash_read(void *context, uint32_t offset, void *data, size_t count)
102{
103 (void)context;
104 if (!in_range(offset, count)) {
105 return false;
106 }
107 ecc_failed = false;
108 const void *source = (const void *)((uintptr_t)&__obcp_storage_start + offset);
109 memcpy(data, source, count);
110 return !ecc_failed;
111}

113/**
114 * @brief Effacer un nombre entier de pages en banque 2.
115 *
116 * Contrôle géométrie et alignement, déverrouille la flash, convertit l'adresse en numéro
117 * de page de banque 2, efface puis reverrouille.
118 * @param[in] context Contexte opaque transmis sans allocation. Le port physique peut ne pas
119 * l’utiliser.
120 * @param[in] offset Offset en octets depuis le début de la réserve flash.
121 * @param[in] count Nombre d’octets de l’opération ou de la transaction.
122 * @return true si tous les contrôles et l'effacement réussissent.
123 */
124static bool flash_erase(void *context, uint32_t offset, uint32_t count)
125{
126 (void)context;
127 if (!obcp_flash_ready() || !in_range(offset, count) ||
128 offset % OBCP_FLASH_PAGE != 0U || count == 0U || count % OBCP_FLASH_PAGE != 0U ||
129 HAL_FLASH_Unlock() != HAL_OK) {
130 return false;
131 }
132 uint32_t addr = (uint32_t)(uintptr_t)&__obcp_storage_start + offset;
133 FLASH_EraseInitTypeDef params = {
134 .TypeErase = FLASH_TYPEERASE_PAGES, .Banks = FLASH_BANK_2,
135 .Page = (addr - BANK_2_ADDRESS) / OBCP_FLASH_PAGE,
136 .NbPages = count / OBCP_FLASH_PAGE
137 };
138 uint32_t bad_page;
139 __HAL_FLASH_CLEAR_FLAG(FLASH_FLAG_ALL_ERRORS);
140 HAL_StatusTypeDef result = HAL_FLASHEx_Erase(&params, &bad_page);
141 (void)HAL_FLASH_Lock();
142 return result == HAL_OK;
143}

145/**
146 * @brief Programmer exactement un double mot aligné.
147 *
148 * Copie les huit octets vers un uint64_t local pour respecter l'alignement, effectue la
149 * programmation puis reverrouille le contrôleur.
150 * @param[in] context Contexte opaque transmis sans allocation. Le port physique peut ne pas
151 * l’utiliser.
152 * @param[in] offset Offset en octets depuis le début de la réserve flash.
153 * @param[in] bytes Huit octets source du double mot à programmer.
154 * @return true si la programmation HAL réussit.
155 */
156static bool flash_program(void *context, uint32_t offset, const uint8_t bytes[8])
157{
158 (void)context;
159 if (!obcp_flash_ready() || !in_range(offset, 8U) || (offset & 7U) != 0U ||
160 HAL_FLASH_Unlock() != HAL_OK) {
161 return false;
162 }
163 uint64_t word;
164 memcpy(&word, bytes, sizeof(word));
165 uint32_t addr = (uint32_t)(uintptr_t)&__obcp_storage_start + offset;
166 __HAL_FLASH_CLEAR_FLAG(FLASH_FLAG_ALL_ERRORS);
167 HAL_StatusTypeDef result = HAL_FLASH_Program(FLASH_TYPEPROGRAM_DOUBLEWORD, addr, word);
168 (void)HAL_FLASH_Lock();
169 return result == HAL_OK;
170}

172/*
173 * Construire le port utilisé par storage.c.
174 *
175 * Le port contient les trois callbacks synchrones. Un port nul est renvoyé si les
176 * hypothèses de placement ne correspondent pas au composant.
177 */
178Storage obcp_flash_storage(void)
179{
180 if (!obcp_flash_ready()) {
181 return (Storage){0};
182 }
183 return (Storage){NULL, flash_read, flash_erase, flash_program};
184}
