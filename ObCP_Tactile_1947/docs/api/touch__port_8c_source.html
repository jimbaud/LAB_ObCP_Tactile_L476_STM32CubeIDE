1/**
2 * @file touch_port.c
3 * @brief Reconnaissance des shields et façade tactile commune.
4 * @ingroup obcp_drivers
5 *
6 * L’application lit toujours un RawTouch et un état TouchRead. La façade choisit un
7 * protocole capacitif I²C, résistif SPI ou résistif I²C. Un état WAIT conserve la
8 * possibilité de servir d’autres tâches entre deux étapes.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */
14#include "obcp_panel.h"
15#include "touch_drivers.h"
16
17/*
18 * Reconnaître le contrôleur tactile présent.
19 *
20 * Vérifie d'abord que SPI1 peut être utilisé, essaie les signatures FT62xx puis STMPE610,
21 * enfin le protocole TSC2007. Une recherche terminée peut produire TOUCH_CTRL_NONE.
22 */
23bool touch_probe(TouchDevice *device)
24{
25 if (!lcd_spi_available()) {
26 return false;
27 }
28
29 /* Essayer les signatures explicites avant la reconnaissance par
30 * protocole du TSC2007. Le bus I²C du shield réserve 0x38 et 0x48. */
31 uint8_t ft_id;
32 uint16_t stmpe_id;
33 if (ft6206_probe(&ft_id)) {
34 *device = (TouchDevice){TOUCH_CTRL_FT62XX, ft_id};
35 } else if (stmpe610_probe(&stmpe_id)) {
36 *device = (TouchDevice){TOUCH_CTRL_STMPE610, stmpe_id};
37 } else if (tsc2007_probe()) {
38 *device = (TouchDevice){TOUCH_CTRL_TSC2007, 0};
39 } else {
40 *device = (TouchDevice){0};
41 }
42 return true;
43}
44
45/*
46 * Acquérir un RawTouch avec le pilote reconnu.
47 *
48 * Remet d'abord la structure de sortie à zéro. Le contrôleur détermine le pilote, et les
49 * états asynchrones sont propagés à board.c.
50 */
51TouchRead touch_read(const TouchDevice *device, RawTouch *sample)
52{
53 /* Une structure remise à zéro protège l’appelant d’anciennes coordonnées en cas d’échec. Le code de retour indique si elle doit être exploitée. */
54 *sample = (RawTouch){0};
55 switch (device->controller) {
56 case TOUCH_CTRL_FT62XX:
57 return ft6206_sample(&sample->down, &sample->point.x, &sample->point.y)
58 ? TOUCH_READ_OK : TOUCH_READ_ERROR;
59 case TOUCH_CTRL_STMPE610:
60 return stmpe610_sample(sample);
61 case TOUCH_CTRL_TSC2007:
62 return tsc2007_sample(sample);
63 default:
64 return TOUCH_READ_ERROR;
65 }
66}
