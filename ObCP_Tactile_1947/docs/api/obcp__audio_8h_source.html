1/**
2 * @file obcp_audio.h
3 * @brief Synthèse numérique et séquenceur musical sans accès matériel.
4 * @ingroup obcp_audio
5 *
6 * Les échantillons sont des valeurs de comparaison CCR3 sur 10 bits, centrées autour de
7 * 512. Le transport DMA consomme ces valeurs à 15 625 échantillons par seconde. Le format
8 * des motifs encode chaque champ explicitement.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */
14#ifndef OBCP_AUDIO_H
15#define OBCP_AUDIO_H
16
17#include <stdbool.h>
18#include <stddef.h>
19#include <stdint.h>

21/* TIM1 à 64 MHz, ARR=1023, RCR=3. Un nouveau rapport toutes les 4 périodes. */
22/**
23 * @brief Dimensions, bornes et constantes du module.
24 */
25enum {
26 AUDIO_HZ = 15625, /**< Cadence de 15 625 échantillons audio par seconde. */
27 AUDIO_CENTER = 512, /**< Comparaison 512, centre du signal sur dix bits. */
28 AUDIO_MAX_DUTY = 1023, /**< Comparaison maximale du timer, ARR vaut 1023. */
29 AUDIO_VOICES = 5, /**< Cinq voix, quatre pistes et une voix de clavier. */
30 SEQ_TRACKS = 4, /**< Quatre pistes musicales. */
31 SEQ_STEPS = 16, /**< Seize doubles croches dans une boucle. */
32 SEQ_SLOTS = 4, /**< Quatre emplacements musicaux persistants. */
33 SEQ_MIN_BPM = 60, /**< Tempo minimum admis, 60 battements par minute. */
34 SEQ_MAX_BPM = 180, /**< Tempo maximum admis, 180 battements par minute. */
35 SEQ_PATTERN_BYTES = 2 + SEQ_TRACKS + SEQ_TRACKS * SEQ_STEPS, /**< Longueur encodée d'un motif, 70 octets. */
36 SEQ_BOOK_BYTES = 2 + SEQ_SLOTS * SEQ_PATTERN_BYTES, /**< Longueur du livre versionné, 282 octets. */
37};

39/**
40 * @brief Motif musical de quatre pistes et seize pas.
41 *
42 * La valeur zéro d’une note signifie silence. Les autres valeurs choisissent une des huit
43 * hauteurs.
44 */
45typedef struct {
46 uint16_t bpm; /**< Tempo en battements par minute, entre 60 et 180. */
47 uint8_t timbre[SEQ_TRACKS]; /**< Type de son de chaque piste. 0 kick, 1 snare, 2 hat, 3 bass. */
48 uint8_t notes[SEQ_TRACKS][SEQ_STEPS]; /**< Grille piste puis pas. Chaque case est zéro ou une hauteur de 1 à 8. */
49} Pattern;
50/**
51 * @brief Quatre emplacements de motifs musicaux.
52 *
53 * Le masque de présence distingue un motif enregistré d’un emplacement jamais initialisé.
54 */
55typedef struct {
56 Pattern pattern[SEQ_SLOTS]; /**< Tableau des quatre motifs, indexés de 0 à 3. */
57 uint8_t present; /**< Bits de présence. Le bit i indique que le motif i est exploitable. */
58} PatternBook;
59/**
60 * @brief État d’une voix de synthèse DDS et de son enveloppe.
61 *
62 * Quand remaining est non nul, length est positif. Les durées sont exprimées en
63 * échantillons audio.
64 */
65typedef struct {
66 uint32_t phase; /**< Accumulateur de phase non signé sur 32 bits. Son débordement réalise une période. */
67 uint32_t increment; /**< Pas ajouté à la phase pour chaque échantillon. */
68 uint16_t remaining; /**< Nombre d'échantillons restant à produire pour la voix. */
69 uint16_t length; /**< Durée de référence de l'enveloppe, en échantillons. */
70 uint8_t type; /**< Timbre interne, percussions, basse ou clavier. */
71} Voice;
72/**
73 * @brief Séquenceur et synthèse partagés par la groovebox et le diagnostic.
74 *
75 * Les interruptions ne modifient pas cette structure. audio_fill() fait avancer les phases
76 * et le tempo dans la boucle principale.
77 */
78typedef struct {
79 Pattern pattern; /**< Motif actuellement édité et joué. */
80 Voice voices[AUDIO_VOICES]; /**< Quatre voix de piste et une voix manuelle de clavier. */
81 uint32_t beat_fraction; /**< Reste de l'accumulateur de tempo, à comparer à 60 fois AUDIO_HZ. */
82 uint32_t test_phase; /**< Phase indépendante de la note de diagnostic à 440 Hz. */
83 uint16_t noise; /**< Registre LFSR 16 bits utilisé pour le bruit des percussions. */
84 uint8_t step; /**< Pas courant du séquenceur, de 0 à 15. */
85 uint8_t volume; /**< Pourcentage d'amplitude de sortie, de 0 à 100. */
86 bool playing; /**< true lorsque le séquenceur fait avancer les pas. */
87 bool test_tone; /**< true lorsque la note de diagnostic remplace le mélange des voix. */
88} Audio;

124void audio_key(Audio *a, uint8_t note);

134/**
