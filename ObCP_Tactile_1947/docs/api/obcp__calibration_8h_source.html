1/**
2 * @file obcp_calibration.h
3 * @brief Transformation affine et assistant de calibrage résistif.
4 * @ingroup obcp_input
5 *
6 * Trois contacts bruts définissent une base. Une quatrième cible indépendante vérifie la
7 * transformation avant sauvegarde. Les deux profils, TSC2007 et STMPE610, sont stockés
8 * sous forme de points pour reconstruire les coefficients au chargement.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */

35/**
36 * @brief Transformation affine du brut résistif vers les pixels.
37 *
38 * Les trois observations suffisent à reconstruire les six coefficients. Seuls les points
39 * sont encodés dans la flash.
40 */
41typedef struct {
42 RawPoint points[CAL_FIT_COUNT]; /**< Trois observations brutes ayant défini le profil. */
43 float x_from_x; /**< Coefficient de X brut dans le calcul de l'abscisse écran. */
44 float x_from_y; /**< Coefficient de Y brut dans le calcul de l'abscisse écran. */
45 float x_offset; /**< Décalage de l'abscisse écran, en pixels. */
46 float y_from_x; /**< Coefficient de X brut dans le calcul de l'ordonnée écran. */
47 float y_from_y; /**< Coefficient de Y brut dans le calcul de l'ordonnée écran. */
48 float y_offset; /**< Décalage de l'ordonnée écran, en pixels. */
49 bool valid; /**< true après le calcul réussi d'une base non dégénérée. */
50} TouchCalibration;

61/**
62 * @brief Acquisition progressive des quatre cibles.
63 *
64 * Un contact stable est moyenné puis validé au relâchement. Le quatrième contact contrôle
65 * le candidat obtenu avec les trois premiers.
66 */
67typedef struct {
68 TouchCalibration candidate; /**< Transformation construite et contrôlée pendant la session. */
69 RawPoint observations[CAL_FIT_COUNT]; /**< Trois observations moyennées pour l'ajustement affine. */
70 RawPoint anchor; /**< Première position du contact actuellement stabilisé. */
71 uint32_t sum_x; /**< Somme des abscisses brutes du contact courant. */
72 uint32_t sum_y; /**< Somme des ordonnées brutes du contact courant. */
73 uint8_t target; /**< Indice de la cible attendue, de 0 à 3. */
74 uint8_t samples; /**< Nombre de mesures stables du contact, de 0 à 6. */
75 bool wait_release; /**< true quand le contact courant doit d'abord être relâché. */
76 bool retry; /**< true après un rejet, pour demander de recommencer les cibles. */
77} TouchCalSession;

79/**
80 * @brief Événements de l’assistant de calibrage résistif.
81 */
82typedef enum {
83 CAL_UNCHANGED, /**< La session attend encore un événement utile. */
84 CAL_READY, /**< Le doigt a été relâché, une nouvelle cible peut être acquise. */
85 CAL_HELD, /**< Les six mesures de la cible sont stabilisées, attendre le relâchement. */
86 CAL_ADVANCED, /**< La session a validé une cible et avance à la suivante. */
87 CAL_RETRY, /**< Les observations sont incohérentes, reprendre les cibles. */
88 CAL_COMPLETE, /**< La quatrième cible confirme la transformation candidate. */
89} TouchCalEvent;

93/**
94 * @brief Calculer une transformation affine de trois points bruts.
95 *
96 * Le déterminant de deux vecteurs permet d'inverser la base. Un déterminant trop petit
97 * rejette des observations alignées ou trop proches. Le candidat remplace cal seulement si
98 * le calcul est admis.
99 * @param[in,out] cal Transformation affine et observations de référence. Pointeur valide.
100 * @param[in] points Trois observations brutes, dans l’ordre des cibles de référence.
101 * @return true si une transformation exploitable est calculée.
102 */
103bool touch_cal_fit(TouchCalibration *cal, const RawPoint points[CAL_FIT_COUNT]);
104/**
105 * @brief Transformer un point brut en pixels paysage.
106 *
107 * Vérifie le profil et les bornes des conversions 12 bits, applique les deux expressions
108 * affines puis borne les résultats à l'écran.
109 * @param[in] cal Transformation affine et observations de référence, consulté sans
110 * modification. Pointeur valide.
111 * @param[in] raw Observation physique ou point natif à interpréter.
112 * @param[out] x Abscisse calculée. Pointeur de sortie valide.
113 * @param[out] y Ordonnée calculée. Pointeur de sortie valide.
114 * @return true si les coordonnées ont été calculées.

116bool touch_cal_map(const TouchCalibration *cal, RawPoint raw, int16_t *x, int16_t *y);
117/**
118 * @brief Commencer une session en attendant le relâchement.
119 *

124void touch_cal_start(TouchCalSession *session);

133/**
134 * @brief Faire avancer l'acquisition d'une cible résistive.

136 * Six mesures proches sont moyennées, puis validées au relâchement. Les trois premières
137 * cibles construisent la transformation. La quatrième vérifie l'erreur et peut relancer

143TouchCalEvent touch_cal_sample(TouchCalSession *session, RawTouch raw);
