24/**
25 * @brief Lire une plage de registres FT62xx.
26 *
27 * Le pilote évite de démarrer si le bus est marqué occupé. La HAL utilise une adresse de
28 * registre sur huit bits et un délai limité à quatre millisecondes.
29 * @param[in] first Adresse du premier registre FT62xx à lire.
30 * @param[out] data Tampon recevant au moins len octets consécutifs de registres.
31 * @param[in] size Longueur du bloc en octets.
32 * @return true si tous les registres sont reçus.
33 */
34static bool read_regs(uint8_t first, uint8_t *data, uint16_t size)
35{
36 if (__HAL_I2C_GET_FLAG(&hi2c1, I2C_FLAG_BUSY)) {
37 return false;
38 }
39 return HAL_I2C_Mem_Read(&hi2c1, FT_ADDRESS, first, I2C_MEMADD_SIZE_8BIT,
40 data, size, 4U) == HAL_OK;
41}

43/*
44 * Vérifier l'identifiant de famille du tactile capacitif.
45 *
46 * Le registre 0xA3 est lu. Les identifiants 0x06, 0x36 et 0x64 sont admis. Le résultat de
47 * sortie commence à zéro pour éviter une identité résiduelle.
48 */
49bool ft6206_probe(uint8_t *chip)
50{
51 *chip = 0;
52 return read_regs(0xA3, chip, 1U) && (*chip == 0x06U || *chip == 0x36U || *chip == 0x64U);
53}

55/*
56 * Lire l'état et les coordonnées du premier doigt.
57 *
58 * Une rafale contient TD_STATUS et les deux coordonnées sur 12 bits. L'événement natif de
59 * relâchement ne doit pas devenir un nouvel appui. Les coordonnées sont écrites seulement
60 * si le doigt est présent.
61 */
62bool ft6206_sample(bool *pressed, uint16_t *raw_x, uint16_t *raw_y)
63{
64 uint8_t regs[5];
65 if (!read_regs(0x02, regs, sizeof(regs))) {
66 return false;
67 }
68
69 /* TD_STATUS, P1_XH, P1_XL, P1_YH, P1_YL. Les bits 7:6 de XH
70 * donnent l'événement natif. Le firmware exploite le premier doigt. */
71 *pressed = (regs[0] & 0x0FU) != 0U && (regs[1] >> 6) != 1U;
72 if (*pressed) {
73 *raw_x = (uint16_t)((uint16_t)(regs[1] & 15U) << 8 | regs[2]);
74 *raw_y = (uint16_t)((uint16_t)(regs[3] & 15U) << 8 | regs[4]);
75 }
76 return true;
77}
