3 * @brief Géométrie, primitives RGB565 et rendu par bandes DMA.
4 * @ingroup obcp_graphics

6 * Le repère écran a X vers la droite et Y vers le bas. Les bornes droites et basses des

34/* Les composantes 8 bits sont tronquées à 5, 6 et 5 bits. Aucun flottant ni allocation. */
35/**
36 * @brief Encoder trois composantes 8 bits dans un mot RGB565.
37 * @param[in] r Composante rouge comprise entre 0 et 255.
38 * @param[in] g Composante verte comprise entre 0 et 255.
39 * @param[in] b Composante bleue comprise entre 0 et 255.
40 * @return Mot uint16_t contenant rouge sur 5 bits, vert sur 6 et bleu sur 5.
41 * @pre Fournir des expressions sans effet de bord et dans la plage indiquée.
42 */
43#define RGB565(r, g, b) ((uint16_t)((((r) & 248U) << 8) | (((g) & 252U) << 3) | ((b) >> 3)))

45/* Origine en haut à gauche. Un rectangle contient [x, x+w[ et [y, y+h[. */
46/**
47 * @brief Rectangle entier du repère écran.
48 *
49 * Le rectangle contient les abscisses x à x+w exclu et les ordonnées y à y+h exclu. Une
50 * dimension non positive décrit un rectangle vide.
51 */
52typedef struct {
53 int16_t x; /**< Abscisse du bord gauche, en pixels. */
54 int16_t y; /**< Ordonnée du bord supérieur, en pixels. */
55 int16_t w; /**< Largeur en pixels. */
56 int16_t h; /**< Hauteur en pixels. */
57} Rect;
58/**
59 * @brief Vue de dessin sur un tampon de bande RGB565.
60 *
61 * Le pas mémoire est clip.w. Changer seulement clip modifierait aussi ce pas. Les
62 * primitives réalisent donc leur propre intersection tout en conservant cette vue.
63 */
64typedef struct {
65 Rect clip; /**< Position et dimensions écran de la bande stockée dans pixels. */
66 uint8_t *pixels; /**< Tampon d'au moins deux fois clip.w fois clip.h octets, ordre RGB565 fort puis faible. */
67} Gfx;

210/**
211 * @brief Propriété du tampon après un départ de transfert.
212 */
213typedef enum {
214 TRANSFER_BUSY, /**< Le transport possède encore le tampon. */
215 TRANSFER_DONE, /**< Le transport a terminé et libéré le tampon. */
216 TRANSFER_ERROR, /**< Le transport a échoué et libéré le tampon. */
217} TransferState;
218/**
219 * @brief Transport abstrait d’une bande vers un écran.
220 *
221 * send() prend le tampon jusqu’à poll() DONE ou ERROR. Le contexte peut pointer vers un
222 * pilote ou une simulation PC. Le moteur ne connaît pas la HAL.
223 */
224typedef struct {
225 void *context; /**< Contexte opaque transmis aux deux callbacks. NULL est permis pour le port TFT unique. */
226 bool (*send)(void *context, Rect r, const uint8_t *data, uint16_t count); /**< Callback synchrone de départ. true signifie que le transport possède maintenant les données jusqu'à la fin signalée par poll. */
227 TransferState (*poll)(void *context); /**< Callback de consultation de l'état du dernier envoi. Il rend BUSY, DONE ou ERROR sans dessiner de pixels. */
228} DisplayPort;

248 uint8_t index; /**< Indice du rectangle courant dans batch. */

250 bool in_flight; /**< true tant que le transport possède buffer. */
251 uint32_t batches; /**< Nombre de lots de rectangles complètement achevés. */
252 uint32_t bands; /**< Nombre de bandes transférées avec succès. */
253 uint32_t pixels; /**< Nombre cumulé de pixels transmis avec succès. */
254 uint32_t errors; /**< Nombre de départs ou de transferts échoués. */
255 uint8_t buffer[LCD_WIDTH * BAND_ROWS * 2]; /**< Seul tampon RGB565 de 320 fois 16 pixels, soit 10 240 octets. */
