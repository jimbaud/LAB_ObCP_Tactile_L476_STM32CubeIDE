2 * @file stmpe610.c
3 * @brief Tactile résistif historique sur le bus SPI1 partagé.
4 * @ingroup obcp_drivers
5 *
6 * Le pilote sauvegarde prescaler, polarité et phase du SPI avant son accès à 1 MHz, puis
7 * les restaure pour le TFT à 8 MHz. Il lit au plus huit points FIFO par passage et utilise
8 * le dernier point reçu.

25enum {
26 STMPE_ID = 0x0811, /**< Signature 16 bits attendue du STMPE610. */
27 STMPE_SYS_CTRL2 = 0x04, /**< Adresse du registre d’activation des blocs ADC et tactile. */
28 STMPE_INT_CTRL = 0x09, /**< Adresse de configuration globale des interruptions du contrôleur. */
29 STMPE_INT_EN = 0x0A, /**< Adresse du masque des sources d’interruption. */
30 STMPE_INT_STA = 0x0B, /**< Adresse des événements à acquitter. */
31 STMPE_ADC1 = 0x20, /**< Adresse des paramètres de conversion ADC. */
32 STMPE_ADC2 = 0x21, /**< Adresse de la fréquence ADC. */
33 STMPE_CTRL = 0x40, /**< Adresse de contrôle du tactile et du bit de contact. */
34 STMPE_CFG = 0x41, /**< Adresse du délai de stabilisation et de la moyenne tactile. */
35 STMPE_FIFO_STA = 0x4B, /**< Adresse de commande et d’état de la FIFO tactile. */
36 STMPE_FIFO_SIZE = 0x4C, /**< Adresse du nombre de points présents dans la FIFO. */
37 STMPE_FRACTION_Z = 0x56, /**< Adresse de répartition des bits de pression. */
38 STMPE_DATA = 0x57, /**< Adresse d’accès non incrémenté aux coordonnées et à la pression. */
39 STMPE_DRIVE = 0x58, /**< Adresse du courant de commande de la dalle. */
40 STMPE_TOUCHED = 0x80, /**< Masque du bit de présence d’un contact. */
41 STMPE_FIFO_RESET = 0x01, /**< Masque du bit de remise à zéro de la FIFO. */
42 STMPE_MAX_POINTS = 8, /**< Au plus huit points retirés de la FIFO par appel. */
43};

47/**
48 * @brief Changer temporairement fréquence et phase du SPI1.
49 *
50 * Le périphérique est désactivé avant la modification des bits BR, CPHA et CPOL. Le masque
51 * préserve les autres réglages matériels.
52 * @param[in] settings Bits BR, CPHA et CPOL du SPI à appliquer.
53 * @return Anciennes valeurs des bits modifiés, à restaurer après la transaction.
54 */
55static uint32_t set_spi(uint32_t settings)
56{
57 /* Sauvegarder uniquement les bits modifiés permet de rendre au TFT le même SPI1, après la transaction tactile à 1 MHz. */
58 const uint32_t mask = SPI_CR1_BR | SPI_CR1_CPHA | SPI_CR1_CPOL;
59 uint32_t previous = READ_BIT(hspi1.Instance->CR1, mask);
60 __HAL_SPI_DISABLE(&hspi1);
61 MODIFY_REG(hspi1.Instance->CR1, mask, settings);
62 return previous;
63}

65/**
66 * @brief Échanger un message sous la sélection tactile D8.
67 *
68 * Le TFT doit déjà avoir libéré SPI1. La sélection tactile est abaissée pour l'échange,
69 * puis remontée même si la HAL signale une erreur.
70 * @param[in] tx Octets à émettre, au moins len ou count cases selon la signature.
71 * @param[out] rx Tampon de réception, de même capacité que le nombre d’octets échangés.
72 * @param[in] count Nombre d’octets de l’opération ou de la transaction.
73 * @return true si l'échange SPI se termine correctement.
74 */
75static bool transaction(uint8_t *tx, uint8_t *rx, uint16_t count)
76{
77 HAL_GPIO_WritePin(TOUCH_SPI_CS_GPIO_Port, TOUCH_SPI_CS_Pin, GPIO_PIN_RESET);
78 HAL_StatusTypeDef status = HAL_SPI_TransmitReceive(&hspi1, tx, rx, count, 2);
79 HAL_GPIO_WritePin(TOUCH_SPI_CS_GPIO_Port, TOUCH_SPI_CS_Pin, GPIO_PIN_SET);
80 return status == HAL_OK;
81}

83/**
84 * @brief Lire un registre STMPE610 avec son octet factice.
85 *
86 * Le bit de lecture et l'adresse sont émis en premier. La valeur utile arrive au troisième
87 * octet du protocole utilisé par ce composant.
88 * @param[in] reg Adresse du registre du composant.
89 * @param[out] value Pointeur valide recevant la valeur du registre ou de la conversion.
90 * @return true si l'octet de registre est reçu.
91 */
92static bool reg_read(uint8_t reg, uint8_t *value)
93{
94 uint8_t tx[] = {(uint8_t)(0x80U | reg), 0, 0};
95 uint8_t rx[sizeof(tx)];
96 if (!transaction(tx, rx, sizeof(tx))) {
97 return false;
98 }
99 *value = rx[2];
100 return true;
101}

103/**
104 * @brief Écrire un registre STMPE610.
105 *
106 * Émet une paire adresse-valeur, les données simultanément reçues étant ignorées.
107 * @param[in] reg Adresse du registre du composant.
108 * @param[in] value Valeur à utiliser dans ce traitement.
109 * @return true si l'écriture SPI réussit.
110 */
111static bool reg_write(uint8_t reg, uint8_t value)
112{
113 uint8_t tx[] = {reg, value};
114 uint8_t rx[sizeof(tx)];
115 return transaction(tx, rx, sizeof(tx));
116}

118/**
119 * @brief Appliquer les réglages ADC et FIFO du tactile.
120 *
121 * L'interrogation se fait par scrutation. La séquence règle résolution, moyenne, délais et
122 * activation, puis vérifie le registre de contrôle.
123 * @return true si la séquence et sa relecture de contrôle réussissent.
124 */
125static bool init_touch(void)
126{
127 /* Interrogation par scrutation. Laisser l'IRQ de la dalle désactivée. */
128 static const uint8_t init[][2] = {
129 {STMPE_SYS_CTRL2, 0x00}, {STMPE_CTRL, 0x00}, {STMPE_INT_CTRL, 0x00}, {STMPE_INT_EN, 0x00},
130 {STMPE_ADC1, 0x68}, {STMPE_ADC2, 0x02}, /* 12 bits, 96 cycles à 6,5 MHz. */
131 {STMPE_CFG, 0xA4}, /* Moyenne de 4, délai 1 ms, stabilisation 5 ms. */
132 {STMPE_FRACTION_Z, 0x06}, {STMPE_DRIVE, 0x01}, {STMPE_INT_STA, 0xFF},
133 {STMPE_FIFO_STA, STMPE_FIFO_RESET}, {STMPE_FIFO_STA, 0}, {STMPE_CTRL, 0x01}
134 };
135 for (unsigned i = 0; i < sizeof(init) / sizeof(init[0]); ++i) {
136 if (!reg_write(init[i][0], init[i][1])) {
137 return false;
138 }
139 }
140 uint8_t control;
141 return reg_read(STMPE_CTRL, &control) && (control & 0x7FU) == 1U;
142}

144/*
145 * Reconnaître 0x0811 et configurer le tactile.
146 *
147 * Les modes SPI zéro puis un sont essayés à 1 MHz. La phase reconnue est conservée pour
148 * les acquisitions suivantes. Les réglages du TFT sont restaurés à la sortie.
149 */
150bool stmpe610_probe(uint16_t *chip)
151{
152 *chip = 0;
153 if (!lcd_spi_available()) {
154 return false;
155 }
156 /* SPI1 reçoit 64 MHz de l'APB2. Diviser par 64 donne 1 MHz. Les
157 * révisions anciennes sont essayées en modes 0 puis 1. */
158 uint32_t previous = set_spi(SPI_BAUDRATEPRESCALER_64);
159 bool found = false;
160 for (unsigned mode = 0; mode < 2U && !found; ++mode) {
161 uint8_t hi, lo;
162 spi_phase = mode == 0U ? SPI_PHASE_1EDGE : SPI_PHASE_2EDGE;
163 set_spi(SPI_BAUDRATEPRESCALER_64 | spi_phase);
164 if (reg_read(0, &hi) && reg_read(1, &lo)) {
165 *chip = (uint16_t)((uint16_t)hi << 8 | lo);
166 found = *chip == STMPE_ID;
167 }
168 }
169 bool ready = found && init_touch();
170 set_spi(previous);
171 return ready;
172}

174/**
175 * @brief Vider la file de points du STMPE610.
176 *
177 * Le bit RESET est écrit à un puis à zéro. Cette remise à zéro traite aussi les points
178 * accumulés pendant une opération lente de stockage.
179 * @return true si les deux écritures réussissent.
180 */
181static bool clear_fifo(void)
182{
183 return reg_write(STMPE_FIFO_STA, STMPE_FIFO_RESET) && reg_write(STMPE_FIFO_STA, 0);
184}

186/**
187 * @brief Lire un lot FIFO et retenir son observation récente.
188 *
189 * Contrôle le registre de fonctionnement et le compteur. Une FIFO vide sous un doigt donne
190 * EMPTY, tandis qu'un relâchement donne OK sans contact. Les octets reçus sont assemblés
191 * en X/Y 12 bits et Z 8 bits.
192 * @param[out] sample Structure RawTouch valide recevant l’observation physique.
193 * @return État de lecture OK, EMPTY ou ERROR.
194 */
195static TouchRead read_points(RawTouch *sample)
196{
197 uint8_t control, count;
198 if (!reg_read(STMPE_CTRL, &control) || !reg_read(STMPE_FIFO_SIZE, &count) ||
199 (control & 0x7FU) != 1U || count > 128U) {
200 return TOUCH_READ_ERROR;
201 }
202 if ((control & STMPE_TOUCHED) == 0U) {
203 return count == 0U || clear_fifo() ? TOUCH_READ_OK : TOUCH_READ_ERROR;
204 }
205 if (count == 0U) {
206 return TOUCH_READ_EMPTY;
207 }
208 unsigned take = count > STMPE_MAX_POINTS ? STMPE_MAX_POINTS : count;
209 uint8_t data[4];
210 for (unsigned i = 0; i < take; ++i) {
211 for (unsigned byte = 0; byte < sizeof(data); ++byte) {
212 if (!reg_read(STMPE_DATA, &data[byte])) {
213 return TOUCH_READ_ERROR;
214 }
215 }
216 }
217 /* Limiter le temps SPI même après une longue écriture flash. */
218 if (count > take && !clear_fifo()) {
219 return TOUCH_READ_ERROR;
220 }
221 /* Le même registre livre X[11:0], Y[11:0], puis Z[7:0]. */
222 *sample = (RawTouch){
223 {(uint16_t)((uint16_t)data[0] << 4 | data[1] >> 4),
224 (uint16_t)((uint16_t)(data[1] & 0x0FU) << 8 | data[2])},
225 data[3], 0, true
226 };
227 return TOUCH_READ_OK;
228}

230/*
231 * Acquérir une observation entre deux bandes TFT.
232 *
233 * BUSY donne WAIT pour rendre la main. Sinon le pilote emprunte le SPI, lit ses points,
234 * puis remet les paramètres initiaux quelle que soit l'issue.
235 */
236TouchRead stmpe610_sample(RawTouch *sample)
237{
238 if (!lcd_spi_available()) {
239 return TOUCH_READ_WAIT;
240 }
241 uint32_t previous = set_spi(SPI_BAUDRATEPRESCALER_64 | spi_phase);
242 TouchRead result = read_points(sample);
243 set_spi(previous);
244 return result;
245}
