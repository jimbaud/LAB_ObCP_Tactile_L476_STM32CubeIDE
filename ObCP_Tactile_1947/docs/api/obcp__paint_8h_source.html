1/**
2 * @file obcp_paint.h
3 * @brief Canevas indexé, diamètres de trait et palette persistante.
4 * @ingroup obcp_apps
5 *
6 * Deux pixels tiennent dans un octet. Les seize indices désignent la palette du dessin. La
7 * case zéro sert au fond et à la gomme. Un dessin enrichi ajoute 36 octets aux pixels, et
8 * le lecteur accepte aussi le format historique de 32 000 octets.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */
14#ifndef OBCP_PAINT_H
15#define OBCP_PAINT_H
16
17/** @var PAINT_PALETTE
18 * @brief Palette RGB565 initiale copiée dans chaque nouveau dessin.
19 */
20/** @var PAINT_DIAMETERS
21 * @brief Six diamètres de trait et de gomme, 1, 3, 5, 9, 13 et 19 pixels.
22 */
23
24#include "obcp_gfx.h"
25#include "obcp_storage.h"

27/**
28 * @brief Dimensions, bornes et constantes du module.
29 */
30enum {
31 PAINT_HEIGHT = 200, /**< Hauteur utile du canevas, 200 pixels. */
32 PAINT_BYTES = LCD_WIDTH * PAINT_HEIGHT / 2, /**< 32 000 octets pour 320 × 200 indices sur quatre bits. */
33 PAINT_BRUSHES = 6, /**< Six diamètres proposés au crayon et à la gomme. */
34};
35extern const uint16_t PAINT_PALETTE[16];
36extern const uint8_t PAINT_DIAMETERS[PAINT_BRUSHES];
37
38/**
39 * @brief Format contigu des pixels et métadonnées du dessin.
40 *
41 * Les membres sont des octets et un _Static_assert vérifie la taille totale. La palette y
42 * est encodée explicitement en little-endian.
43 */
44typedef struct {
45 uint8_t pixels[PAINT_BYTES]; /**< 32 000 octets contenant deux indices de couleur par octet. */
46 uint8_t metadata[36]; /**< Signature PT, version, indice de diamètre et seize couleurs RGB565, soit 36 octets. */
47} PaintImage;
48/**
49 * @brief Canevas et état des outils de dessin.
50 *
51 * L’encre est un indice dans une palette de seize couleurs. Les pixels conservent cet
52 * indice, ce qui permet de recolorer une case de palette.
53 */
54typedef struct {
55 PaintImage image; /**< Pixels et espace de métadonnées utilisés par la sauvegarde. */
56 uint16_t palette[16]; /**< Seize couleurs RGB565 propres au dessin chargé. */
57 uint8_t ink; /**< Indice de la couleur du crayon, de 0 à 15. */
58 uint8_t brush; /**< Indice dans PAINT_DIAMETERS, de 0 à PAINT_BRUSHES moins un. */
59 bool erase; /**< true pour déposer l'indice zéro avec la gomme. */
60 bool stroke; /**< true si le contact courant a commencé sur le canevas. */
61 bool modified; /**< true si pixels ou palette ont changé depuis la dernière lecture ou sauvegarde réussie. */
62 int16_t last_x; /**< Abscisse précédente du trait, en pixels. */
63 int16_t last_y; /**< Ordonnée précédente du trait, en pixels. */
64} Paint;
65
66/**
67 * @brief Initialiser un canevas, une palette et le crayon.
68 *
69 * Met les pixels à l'indice zéro, copie la palette de départ, choisit l'encre huit et le
70 * diamètre cinq pixels. Le dessin n'est pas marqué modifié à cette étape.
71 * @param[out] p Canevas et palette du dessin, à initialiser par cet appel. Pointeur
72 * valide.
73 */
74void paint_init(Paint *p);
75/**
76 * @brief Effacer les pixels tout en gardant la palette.
77 *
78 * Le tableau est rempli de zéros, le geste en cours se termine et le dessin devient
79 * modifié. La couleur d'effacement est la première case de la palette.
80 * @param[in,out] p Canevas et palette du dessin. Pointeur valide.
81 */
82void paint_clear(Paint *p);
83/**
84 * @brief Lire l'indice de couleur d'un pixel.
85 *
86 * Calcule l'octet commun à deux abscisses, choisit le demi-octet bas ou haut puis applique
87 * le masque 15. Une coordonnée hors canevas donne l'indice de fond.
88 * @param[in] p Canevas et palette du dessin, consulté sans modification. Pointeur
89 * valide.
90 * @param[in] x Abscisse dans le repère écran, en pixels.
91 * @param[in] y Ordonnée dans le repère écran, en pixels.
92 * @return Indice compris entre zéro et quinze.
93 */
94uint8_t paint_get(const Paint *p, int x, int y);
95/**
96 * @brief Relier deux relevés tactiles par un trait continu.
97 *
98 * Bresenham choisit des centres entiers successifs et dab() y applique le diamètre
99 * courant. Le rectangle renvoyé couvre les deux extrémités et l'épaisseur du trait.
100 * @param[in,out] p Canevas et palette du dessin. Pointeur valide.
101 * @param[in] x0 Abscisse du début du segment, en pixels.
102 * @param[in] y0 Ordonnée du début du segment, en pixels.
103 * @param[in] x1 Abscisse de la fin du segment, en pixels.
104 * @param[in] y1 Ordonnée de la fin du segment, en pixels.
105 * @return Zone de canevas à recomposer, ou rectangle vide si une extrémité est hors canevas.
106 */
107Rect paint_segment(Paint *p, int x0, int y0, int x1, int y1);
108/**
109 * @brief Convertir les indices du canevas vers RGB565.
110 *
111 * Lit chaque indice dans la seule intersection entre canevas et bande, puis consulte la
112 * palette attachée au dessin. Modifier une case recolore donc tous les pixels de cet
113 * indice.
114 * @param[in] p Canevas et palette du dessin, consulté sans modification. Pointeur
115 * valide.
116 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
117 */
118void paint_draw(const Paint *p, Gfx *g);
119/**
120 * @brief Remplacer la case d'encre active de la palette.
121 *
122 * Une couleur réellement différente marque le dessin modifié. L'outil repasse au crayon,
123 * car l'utilisateur vient de choisir une encre.
124 * @param[in,out] p Canevas et palette du dessin. Pointeur valide.
125 * @param[in] color Couleur encodée au format RGB565.
126 */
127void paint_set_color(Paint *p, uint16_t color);
128/**
129 * @brief Encoder la palette et sauvegarder le dessin enrichi.
130 *
131 * Ajoute signature PT, version, choix de diamètre et couleurs little-endian, puis confie
132 * l'image complète au service A/B.
133 * @param[in,out] p Canevas et palette du dessin. Pointeur valide.
134 * @param[in,out] storage Port de stockage et ses callbacks synchrones. Pointeur valide.
135 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
136 * @return Résultat de storage_save().
137 */
138StoreResult paint_save(Paint *p, Storage *storage, SaveObject object);
139/**
140 * @brief Lire un dessin historique ou enrichi et sa palette.
141 *
142 * La longueur permet de distinguer les pixels historiques de PaintImage. Le format enrichi
143 * est validé avant de décoder les couleurs. Un succès termine le geste et retire
144 * l'indicateur modifié.
145 * @param[in,out] p Canevas et palette du dessin. Pointeur valide.
146 * @param[in,out] storage Port de stockage et ses callbacks synchrones. Pointeur valide.
147 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
148 * @return STORE_OK, résultat du port flash, ou STORE_INVALID si les métadonnées sont incompatibles.
149 */
150StoreResult paint_load(Paint *p, Storage *storage, SaveObject object);
