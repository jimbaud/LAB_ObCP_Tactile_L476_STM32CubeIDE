52/**
53 * @brief Convertir le centre horizontal du joueur en rectangle.
54 *
55 * La coordonnée verticale désigne le haut. Les dimensions du joueur sont fixes afin de
56 * partager la même emprise entre collision et dessin.
57 * @param[in] x Abscisse dans le repère écran, en pixels.
58 * @param[in] y Ordonnée dans le repère écran, en pixels.
59 * @return Rectangle de la voiture du joueur.

61static Rect car_box(float x, float y)
62{
63 return (Rect){(int16_t)((int)x - CAR_W / 2), (int16_t)y, CAR_W, CAR_H};
64}

66/**
67 * @brief Calculer l'emprise d'un adversaire selon son véhicule.
68 *
69 * Le numéro de voie donne le centre horizontal. Le type donne largeur et hauteur, ce qui
70 * distingue notamment moto étroite et camion long.
71 * @param[in] r Position, voie et type de l’adversaire, consulté sans modification.
72 * Pointeur valide.
73 * @return Rectangle de collision et de rendu de l'adversaire.
74 */
75static Rect rival_box(const Rival *r)
76{
77 int center = ROAD_X + ROAD_W / 6 + r->lane * ROAD_W / 3;
78 return (Rect){(int16_t)(center - vehicles[r->kind].width / 2), (int16_t)r->y,
79 vehicles[r->kind].width, vehicles[r->kind].height};
80}

82/**
83 * @brief Calculer la partie visible d'un marquage routier.
84 *
85 * Le décalage de défilement est ajouté à la ligne de base, puis l'emprise est limitée à la
86 * route.
87 * @param[in] lane Indice du séparateur de voie à dessiner.
88 * @param[in] y Ordonnée dans le repère écran, en pixels.
89 * @param[in] phase Décalage vertical des pointillés de route, en pixels.
90 * @return Rectangle visible du marquage.

92static Rect dash(unsigned lane, int y, int phase)
93{
94 return rect_intersection(road, (Rect){(int16_t)(ROAD_X + ROAD_W * (lane + 1U) / 3U - 2U),
95 (int16_t)(y + phase), 4, 15});

98/**
99 * @brief Invalider les marquages à un décalage donné.
100 *
101 * Cette routine est appelée avant et après l'avancement. Les anciennes marques sont donc
102 * effacées et les nouvelles recomposées.
103 * @param[in,out] d Liste de rectangles à recomposer. Pointeur valide.
104 * @param[in] phase Décalage vertical des pointillés de route, en pixels.
105 */
106static void dash_damage(Damage *d, int phase)
107{
108 for (unsigned lane = 0; lane < 2; ++lane) {
109 for (int y = ROAD_Y - 38; y < LCD_HEIGHT; y += 38) {
110 damage_add(d, dash(lane, y, phase));
111 }
112 }
113}

141/**
142 * @brief Créer un adversaire dans une place libre.
143 *
144 * Le générateur choisit une voie différente de la précédente et un type autorisé par le
145 * niveau. Un seul véhicule est créé par échéance pour éviter une rangée complète.
146 * @param[in,out] r Joueur, adversaires et progression de la course. Pointeur valide.
147 */
148static void spawn(Race *r)
149{
150 for (unsigned i = 0; i < RACE_CARS; ++i) {
151 if (!r->rival[i].active) {
152 r->seed = r->seed * UINT32_C(1664525) + UINT32_C(1013904223);
153 /* Un seul rival par apparition. Tous avancent à la même vitesse.
154 * Même au niveau 10, 360 ms * 260 px/s = 93,6 px séparent les
155 * véhicules, plus que camion + joueur (56 + 32 px). Une rangée
156 * complète ne ferme donc pas les trois voies au même instant. */
157 uint8_t lane = (uint8_t)((r->last_lane + 1U + (r->seed >> 31)) % 3U);
158 unsigned types = r->level >= 6U ? 4U : (r->level >= 4U ? 3U : (r->level >= 2U ? 2U : 1U));
159 Vehicle kind = (Vehicle)((r->seed >> 8) % types);
160 r->rival[i] = (Rival){.y = (float)(ROAD_Y - vehicles[kind].height),
161 .lane = lane, .active = true, .kind = kind};
162 r->last_lane = lane;
163 break;
164 }
165 }
166}

168/*
169 * Avancer direction, trafic et état de collision.
170 *
171 * En course, déplace le joueur, les marquages et les adversaires. Une collision précède la
172 * validation du score gagnant. Après une collision, seul le compteur de BOUM continue
173 * jusqu'au bilan.
174 */
175void race_step(Race *r, float tilt_x, Damage *damage)
176{
177 if (r->phase == GAME_LOST && r->crash_ms > 0U) {
178 unsigned previous = r->crash_ms / 100U;
179 r->crash_ms = r->crash_ms > GAME_STEP_MS ? (uint16_t)(r->crash_ms - GAME_STEP_MS) : 0;
180 if (previous != r->crash_ms / 100U || r->crash_ms == 0U) {
181 damage_full(damage);
182 }
183 return;
184 }
185 if (r->phase != GAME_RUN) {
186 return;
187 }
188 Rect previous = car_box(r->player_x, HERO_Y);
189 r->player_x += 300.0f * tilt_x * (GAME_STEP_MS / 1000.0f);
190 if (r->player_x < ROAD_X + CAR_W / 2 + 5) {
191 r->player_x = ROAD_X + CAR_W / 2 + 5;
192 } else if (r->player_x > ROAD_X + ROAD_W - CAR_W / 2 - 5) {
193 r->player_x = ROAD_X + ROAD_W - CAR_W / 2 - 5;
194 }
195 Rect current = car_box(r->player_x, HERO_Y);
196 if (current.x != previous.x) {
197 damage_add(damage, previous);
198 damage_add(damage, current);
199 }
200 dash_damage(damage, (int)r->scroll);
201 float speed = levels[r->level].traffic_step;
202 r->scroll += speed;
203 if (r->scroll >= 38.0f) {
204 r->scroll -= 38.0f;
205 }
206 dash_damage(damage, (int)r->scroll);
207
208 for (unsigned i = 0; i < RACE_CARS; ++i) {
209 Rival *v = &r->rival[i];
210 if (!v->active) {
211 continue;
212 }
213 damage_add(damage, rect_intersection(road, rival_box(v)));
214 v->y += speed;
215 Rect bounds = rival_box(v);
216 if (v->y >= LCD_HEIGHT) {
217 v->active = false;
218 ++r->score;
219 } else {
220 damage_add(damage, rect_intersection(road, bounds));
221 Rect impact = rect_intersection(bounds, current);
222 if (rect_valid(impact)) {
223 r->phase = GAME_LOST;
224 r->crash_ms = 900;
225 r->impact_x = impact.x + impact.w / 2;
226 r->impact_y = 183;
227 damage_full(damage);
228 return;
229 }
230 }
231 }
232 if (r->score >= race_goal(r)) {
233 r->phase = GAME_WON;
234 damage_full(damage);
235 return;
236 }
237 if (r->spawn_ms <= GAME_STEP_MS) {
238 spawn(r);
239 r->spawn_ms = levels[r->level].spawn_period;
240 } else {
241 r->spawn_ms -= GAME_STEP_MS;
242 }
243}

245/**
246 * @brief Dessiner les éléments visibles d'un véhicule.
247 *
248 * Les primitives sont intersectées avec la route en conservant le pas du tampon Gfx. Moto,
249 * fourgonnette et camion ont leurs détails et dimensions propres.
250 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
251 * @param[in] body Emprise graphique et de collision du véhicule.
252 * @param[in] kind Type de véhicule déterminant ses détails graphiques.
253 * @param[in] color Couleur encodée au format RGB565.
254 */
255static void draw_vehicle(Gfx *g, Rect body, Vehicle kind, uint16_t color)
256{
257 /* La bordure supérieure découpe les véhicules entrant par le haut. */
258 Gfx clipped = *g;
259 Rect visible = rect_intersection(g->clip, road);
260 if (!rect_valid(visible) || !rect_valid(rect_intersection(body, visible))) {
261 return;
262 }
263 /* Garder le pas du tampon Gfx. Couper chaque primitive au rectangle route. */
264 gfx_fill(&clipped, rect_intersection(road, body), color);
265 Rect windshield = {(int16_t)(body.x + 4), (int16_t)(body.y + 7),
266 (int16_t)(body.w - 8), 8};
267 gfx_fill(&clipped, rect_intersection(road, windshield), RGB565(29,47,73));
268 Rect plate = {(int16_t)(body.x + 5), (int16_t)(body.y + body.h - 7),
269 (int16_t)(body.w - 10), 3};
270 gfx_fill(&clipped, rect_intersection(road, plate), RGB565(236,238,213));
271 if (kind == VEHICLE_BIKE) {
272 /* Roues dans l'emprise de collision, guidon puis casque du pilote. */
273 gfx_fill(&clipped, rect_intersection(road, (Rect){body.x + 5, body.y, 5, body.h}), RGB565(14,23,35));
274 gfx_fill(&clipped, rect_intersection(road, (Rect){body.x, body.y + 8, body.w, 3}), color);
275 gfx_fill(&clipped, rect_intersection(road, (Rect){body.x + 4, body.y + 12, 7, 7}),
276 RGB565(252,209,82));
277 } else if (kind == VEHICLE_TRUCK || kind == VEHICLE_VAN) {
278 Rect load = {body.x + 3, body.y + 19, body.w - 6, body.h - 28};
279 gfx_fill(&clipped, rect_intersection(road, load), RGB565(180,196,210));
280 }
281}

283/**
284 * @brief Dessiner l'éclat jaune et le texte BOUM.
285 *
286 * Les triangles et le disque créent un éclat centré sur l'impact. La teinte alterne avec
287 * les tranches du compteur d'animation pour donner une brève pulsation.
288 * @param[in] r Joueur, adversaires et progression de la course, consulté sans
289 * modification. Pointeur valide.
290 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
291 */
292static void draw_crash(const Race *r, Gfx *g)
293{
294 int x = r->impact_x, y = r->impact_y;
295 uint16_t yellow = (r->crash_ms / 100U & 1U) ? RGB565(255,221,46) : RGB565(255,183,32);
296 /* Éclat dessiné en primitives, donc identique dans les bandes du TFT et
297 * dans les captures PC. Les pointes entourent le mot de bande dessinée. */
298 gfx_disc(g, x, y, 29, yellow);
299 gfx_triangle(g, x - 46, y - 11, x - 18, y - 24, x - 24, y + 13, yellow);
300 gfx_triangle(g, x + 46, y - 13, x + 18, y - 25, x + 23, y + 14, yellow);
301 gfx_triangle(g, x - 12, y - 39, x + 17, y - 21, x - 26, y - 7, yellow);
302 gfx_triangle(g, x + 12, y + 39, x - 18, y + 22, x + 27, y + 7, yellow);
303 gfx_triangle(g, x + 26, y - 39, x + 36, y - 39, x + 21, y - 13, RGB565(255,247,100));
304 gfx_triangle(g, x + 21, y - 13, x + 34, y - 19, x + 17, y + 2, RGB565(255,247,100));
305 gfx_text(g, x - 29, y - 7, "BOUM!", 2, RGB565(26,34,49));
306}

308/*
309 * Recomposer route, véhicules et collision.
310 *
311 * Le fond et les marquages sont reconstruits avant les adversaires et le joueur. Le BOUM
312 * est superposé si son compteur est encore positif.
313 */
314void race_draw(const Race *r, Gfx *g)
315{
316 gfx_fill(g, (Rect){0, ROAD_Y, LCD_WIDTH, LCD_HEIGHT - ROAD_Y}, RGB565(24,65,51));
317 gfx_fill(g, road, RGB565(42,53,66));
318 gfx_fill(g, (Rect){ROAD_X, ROAD_Y, 3, road.h}, RGB565(223,232,224));
319 gfx_fill(g, (Rect){ROAD_X + ROAD_W - 3, ROAD_Y, 3, road.h}, RGB565(223,232,224));
320 for (unsigned lane = 0; lane < 2; ++lane) {
321 for (int y = ROAD_Y - 38; y < LCD_HEIGHT; y += 38) {
322 gfx_fill(g, dash(lane, y, (int)r->scroll), RGB565(182,190,197));
323 }
324 }
325 for (unsigned i = 0; i < RACE_CARS; ++i) {
326 if (r->rival[i].active) {
327 static const uint16_t colors[] = {
328 RGB565(239,102,113), RGB565(251,198,65), RGB565(131,172,240), RGB565(203,151,231)
329 };
330 draw_vehicle(g, rival_box(&r->rival[i]), r->rival[i].kind, colors[r->rival[i].kind]);
331 }
332 }
333 draw_vehicle(g, car_box(r->player_x, HERO_Y), VEHICLE_CAR, RGB565(74,221,183));
334 if (r->crash_ms > 0U) {
335 draw_crash(r, g);
336 }
337}
