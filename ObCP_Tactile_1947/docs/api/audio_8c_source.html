3 * @brief Synthèse numérique et séquenceur musical sans accès matériel.
4 * @ingroup obcp_audio
5 *
6 * Les échantillons sont des valeurs de comparaison CCR3 sur 10 bits, centrées autour de

28/** @brief Constantes privées de ce module, avec leurs unités. */
29enum {
30 KICK, /**< Timbre de grosse caisse et indice de première piste. */
31 SNARE, /**< Timbre de caisse claire et indice de deuxième piste. */
32 HAT, /**< Timbre de charleston et indice de troisième piste. */
33 BASS, /**< Timbre de basse et indice de quatrième piste. */
34 KEYS, /**< Timbre tonal de la cinquième voix, réservée au clavier. */
35};
36#define STEP_THRESHOLD ((uint32_t)AUDIO_HZ * 60U)
37#define INC_440 ((uint32_t)((UINT64_C(440) << 32) / AUDIO_HZ))

45/**
46 * @brief Calculer une onde triangulaire à partir d'une phase DDS.
47 *
48 * Les neuf bits de poids fort désignent 512 positions dans la période. Deux segments
49 * linéaires produisent le signal sans table ni fonction trigonométrique.
50 * @param[in] phase Phase de l’oscillateur DDS sur 32 bits, une période correspondant à 2^32 incréments unitaires.
51 * @return Amplitude entière de l'onde triangulaire.
52 */
53static int triangle(uint32_t phase)
54{
55 int position = (int)(phase >> 23);
56 return position < 256 ? 2 * position - 255 : 767 - 2 * position;
57}

59/**
60 * @brief Réarmer une voix pour une note.
61 *
62 * La piste choisit le timbre, puis la note choisit l'incrément DDS pour la basse ou le
63 * clavier. La durée restante et la durée totale sont initialisées ensemble.
64 * @param[in,out] a État du motif, du séquenceur et des voix. Pointeur valide.
65 * @param[in] voice Indice de voix, entre zéro et AUDIO_VOICES moins un.
66 * @param[in] note Hauteur de 1 à 8. Zéro signifie silence.
67 */
68static void trigger(Audio *a, unsigned voice, uint8_t note)
69{
70 /* Les durées sont comptées en échantillons, pas en millisecondes. À 15 625 Hz, 2 300 échantillons durent 147,2 ms. */
71 static const uint16_t durations[AUDIO_VOICES] = {2300, 1350, 630, 2400, 5800};
72 static const uint32_t base[AUDIO_VOICES] = {33000000U, 44000000U, 88000000U, 0U, 0U};
73 if (note == 0U || note > 8U) {
74 return;
75 }
76 unsigned timbre = voice < SEQ_TRACKS ? a->pattern.timbre[voice] : KEYS;
77 uint32_t increment = base[timbre];
78 if (timbre == BASS || timbre == KEYS) {
79 increment = pitches[timbre == KEYS ? 1 : 0][note - 1U];
80 }
81 a->voices[voice] = (Voice){0U, increment, durations[timbre], durations[timbre], (uint8_t)timbre};
82}

84/**
85 * @brief Déclencher les voix actives du pas courant.
86 *
87 * Chaque piste possède sa voix. La valeur zéro est un silence et les valeurs 1 à 8 sont
88 * des hauteurs. Le moteur continue de faire décroître les voix déjà lancées.
89 * @param[in,out] a État du motif, du séquenceur et des voix. Pointeur valide.
90 */
91static void play_step(Audio *a)
92{
93 for (unsigned track = 0; track < SEQ_TRACKS; ++track) {
94 trigger(a, track, a->pattern.notes[track][a->step]);
95 }
96}

98/*
99 * Préparer un motif d'exemple et les oscillateurs.
100 *
101 * Réinitialise les états, choisit 112 BPM et un volume de 50 %, puis place quelques
102 * événements sur les quatre pistes. La sortie matérielle reste sous le contrôle de
103 * pwm_audio.c.
104 */
105void audio_init(Audio *a)
106{
107 memset(a, 0, sizeof(*a));
108 a->pattern.bpm = 112;
109 a->noise = UINT16_C(0xACE1);
110 a->volume = 50;
111 for (unsigned track = 0; track < SEQ_TRACKS; ++track) {
112 a->pattern.timbre[track] = (uint8_t)track;
113 }
114 a->pattern.notes[KICK][0] = 1;
115 a->pattern.notes[KICK][8] = 1;
116 a->pattern.notes[SNARE][4] = 1;
117 a->pattern.notes[SNARE][12] = 1;
118 for (unsigned step = 2; step < SEQ_STEPS; step += 4) {
119 a->pattern.notes[HAT][step] = 1;
120 }
121 a->pattern.notes[BASS][0] = 1;
122 a->pattern.notes[BASS][6] = 3;
123 a->pattern.notes[BASS][10] = 5;
124 a->pattern.notes[BASS][14] = 4;
125}

180/**
181 * @brief Calculer un échantillon d'une voix et avancer son état.
182 *
183 * Combine triangle ou bruit LFSR, glissement du kick, attaque puis décroissance. La phase
184 * et la durée restante appartiennent à la voix et persistent entre deux blocs.
185 * @param[in,out] a État du motif, du séquenceur et des voix. Pointeur valide.
186 * @param[in,out] v Phase et enveloppe de la voix. Pointeur valide.
187 * @return Échantillon signé. Zéro lorsque la voix est terminée.
188 */
189static int render_voice(Audio *a, Voice *v)
190{
191 if (v->remaining == 0) {
192 return 0;
193 }
194 /* LFSR 16 bits pour les percussions. Les voix tonales utilisent la phase. */
195 uint16_t noise_mask = (uint16_t)(0U - (a->noise & 1U));
196 a->noise = (uint16_t)((a->noise >> 1) ^ (noise_mask & UINT16_C(0xB400)));
197 int noise = (int)(a->noise & 511U) - 256;
198 int sample = triangle(v->phase);
199 if (v->type == SNARE || v->type == HAT) {
200 sample = v->type == SNARE ? (3 * noise + sample) / 4 : noise;
201 } else if (v->type == KICK && v->increment > 15000000U) {
202 v->increment -= 11500U; /* glissement vers le grave */
203 }
204 /* Le débordement non signé sur 32 bits reboucle la phase. Il représente une période de l’oscillateur DDS. */
205 v->phase += v->increment;
206 unsigned elapsed = v->length - v->remaining;
207 int attack = (int)(elapsed < 96U ? elapsed : 96U);
208 int level = 256 * (int)v->remaining / (int)v->length;
209 --v->remaining;
210 return sample * level * attack / (256 * 96);
211}

213/*
214 * Produire un bloc de rapports cycliques pour l'audio.
215 *
216 * Mélange les voix ou la note d'essai, applique le volume et borne l'amplitude autour de
217 * 512. L'accumulateur de tempo avance exactement à la cadence des échantillons. Appel dans
218 * la boucle principale.
219 */
220void audio_fill(Audio *a, uint16_t *dest, size_t samples)
221{
222 for (size_t i = 0; i < samples; ++i) {
223 int mix = 0;
224 if (a->test_tone) {
225 a->test_phase += INC_440;
226 mix = triangle(a->test_phase);
227 } else {
228 for (unsigned voice = 0; voice < AUDIO_VOICES; ++voice) {
229 mix += render_voice(a, &a->voices[voice]);
230 }
231 mix /= 2;
232 }
233 mix = mix * (int)a->volume / 100;
234 if (mix > 460) {
235 mix = 460;
236 } else if (mix < -460) {
237 mix = -460;
238 }
239 /* Ajouter le centre transforme le signal signé en comparaison PWM. Le bornage laisse une marge aux rails 0 et 1 023. */
240 dest[i] = (uint16_t)(AUDIO_CENTER + mix);
241
242 /* 4 doubles croches par temps. L'accumulateur évite l'erreur
243 * cumulative d'une durée entière arrondie à chaque pas. */
244 if (a->playing) {
245 a->beat_fraction += (uint32_t)a->pattern.bpm * 4U;
246 if (a->beat_fraction >= STEP_THRESHOLD) {
247 a->beat_fraction -= STEP_THRESHOLD;
248 a->step = (uint8_t)((a->step + 1U) % SEQ_STEPS);
249 play_step(a);
250 }
251 }
252 }
253}

255/*
256 * Encoder les quatre motifs dans un format stable.
257 *
258 * Écrit version, masque de présence, tempo little-endian, timbres et notes dans un tableau
259 * d'octets. Les octets ne dépendent pas du remplissage des structures C.
260 */
261void book_encode(const PatternBook *book, uint8_t data[SEQ_BOOK_BYTES])
262{
263 data[0] = 1; /* version du format musical */
264 data[1] = book->present;
265 for (unsigned slot = 0; slot < SEQ_SLOTS; ++slot) {
266 const Pattern *p = &book->pattern[slot];
267 size_t offset = 2U + slot * SEQ_PATTERN_BYTES;
268 data[offset] = (uint8_t)p->bpm;
269 data[offset + 1U] = (uint8_t)(p->bpm >> 8);
270 memcpy(&data[offset + 2U], p->timbre, sizeof(p->timbre));
271 memcpy(&data[offset + 2U + SEQ_TRACKS], p->notes, sizeof(p->notes));
272 }
273}

275/*
276 * Valider puis décoder le livre musical.
277 *
278 * Travaille d'abord sur un candidat local. Le livre de destination est remplacé seulement
279 * si version, masque, tempos, timbres et notes sont cohérents.
280 */
281bool book_decode(PatternBook *book, const uint8_t data[SEQ_BOOK_BYTES])
282{
283 /* Décoder dans un candidat local évite de remplacer un livre valide par un enregistrement partiellement incorrect. */
284 PatternBook decoded = {0};
285 if (data[0] != 1U || data[1] >= (1U << SEQ_SLOTS)) {
286 return false;
287 }
288 decoded.present = data[1];
289 for (unsigned slot = 0; slot < SEQ_SLOTS; ++slot) {
290 Pattern *p = &decoded.pattern[slot];
291 size_t offset = 2U + slot * SEQ_PATTERN_BYTES;
292 if ((decoded.present & (1U << slot)) == 0) {
293 continue;
294 }
295 p->bpm = (uint16_t)(data[offset] | (uint16_t)data[offset + 1U] << 8);
296 if (p->bpm < SEQ_MIN_BPM || p->bpm > SEQ_MAX_BPM) {
297 return false;
298 }
299 for (unsigned t = 0; t < SEQ_TRACKS; ++t) {
300 uint8_t timbre = data[offset + 2U + t];
301 if (timbre >= SEQ_TRACKS) {
302 return false;
303 }
304 p->timbre[t] = timbre;
305 for (unsigned step = 0; step < SEQ_STEPS; ++step) {
306 uint8_t note = data[offset + 2U + SEQ_TRACKS + t * SEQ_STEPS + step];
307 if (note > 8U) {
308 return false;
309 }
310 p->notes[t][step] = note;
311 }
312 }
313 }
314 *book = decoded;
315 return true;
316}
