3 * @brief Format d’objet A/B, CRC32 et validation transactionnelle.
4 * @ingroup obcp_storage
5 *
6 * Chaque objet possède deux emplacements sur des pages distinctes. La génération et les

43typedef struct {

69/**
70 * @brief Décoder un entier little-endian de quatre octets.
71 *
72 * Les décalages explicites évitent de dépendre de l'alignement de l'adresse ou de
73 * l'endianness du processeur.
74 * @param[in] p Adresse de quatre octets encodés en little-endian.
75 * @return Valeur entière recomposée sur 32 bits.
76 */
77static uint32_t get32(const uint8_t *p)
78{
79 return (uint32_t)p[0] | (uint32_t)p[1] << 8 | (uint32_t)p[2] << 16 | (uint32_t)p[3] << 24;
80}

82/**
83 * @brief Encoder un entier sur quatre octets little-endian.
84 *
85 * Le format de sauvegarde garde le même ordre d'octets sur la cible et dans les tests PC.
86 * @param[in,out] p Adresse de quatre octets encodés en little-endian.
87 * @param[in] value Valeur à utiliser dans ce traitement.
88 */
89static void put32(uint8_t *p, uint32_t value)
90{
91 for (unsigned i = 0; i < 4; ++i) {
92 p[i] = (uint8_t)(value >> (8U * i));
93 }
94}

96/**
97 * @brief Poursuivre le calcul bit à bit d'un CRC32.
98 *
99 * Le registre courant traite chaque octet et huit décalages avec le polynôme réfléchi
100 * 0xEDB88320. La complémentation finale est laissée à l'appelant.
101 * @param[in] crc Accumulateur CRC avant complément final.
102 * @param[in] data Bloc source valide de size octets.
103 * @param[in] size Longueur du bloc en octets.
104 * @return Nouvel accumulateur de CRC avant complément final.
105 */
106static uint32_t crc_continue(uint32_t crc, const uint8_t *data, size_t size)
107{
108 while (size-- > 0) {
109 crc ^= *data++;
110 for (unsigned bit = 0; bit < 8; ++bit) {
111 uint32_t mask = 0U - (crc & 1U);
112 crc = (crc >> 1) ^ (UINT32_C(0xEDB88320) & mask);
113 }
114 }
115 return crc;
116}

118/*
119 * Calculer le CRC32 d'un bloc de données.
120 *
121 * Initialise le registre à tous les bits à un et complémente le résultat. Les vecteurs de
122 * tests contrôlent notamment la chaîne 123456789.
123 */
124uint32_t storage_crc32(const void *data, size_t size)
125{
126 return ~crc_continue(UINT32_MAX, data, size);
127}

129/**
130 * @brief Calculer un CRC en relisant la flash par petits blocs.
131 *
132 * Un tampon de 128 octets borne la pile. Une erreur de lecture interrompt le calcul et
133 * reste distincte d'un CRC incorrect.
134 * @param[in,out] s Port de stockage et ses callbacks synchrones. Pointeur valide.
135 * @param[in] start Offset du premier octet à contrôler dans la réserve flash.
136 * @param[in] size Longueur du bloc en octets.
137 * @param[in] expected CRC de référence lu dans l’en-tête.
138 * @return STORE_OK, STORE_IO ou STORE_INVALID.
139 */
140static StoreResult crc_flash(Storage *s, uint32_t start, size_t size, uint32_t expected)
141{
142 uint8_t block[128];
143 uint32_t crc = UINT32_MAX;
144 while (size > 0) {
145 size_t n = size < sizeof(block) ? size : sizeof(block);
146 if (!s->read(s->context, start, block, n)) {
147 return STORE_IO;
148 }
149 crc = crc_continue(crc, block, n);
150 start += (uint32_t)n;
151 size -= n;
152 }
153 return ~crc == expected ? STORE_OK : STORE_INVALID;
154}

156/**
157 * @brief Valider l'en-tête, la longueur et le contenu d'une copie.
158 *
159 * Vérifie marqueur, signature, type, longueur admise et CRC d'en-tête, puis relit le CRC
160 * des données. La longueur de chaque copie est propre à sa version de format.
161 * @param[in,out] s Port de stockage et ses callbacks synchrones. Pointeur valide.
162 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
163 * @param[in] offset Offset en octets depuis le début de la réserve flash.
164 * @return Descripteur de copie avec validité, génération, longueur et éventuelle erreur de lecture.
165 */
166static Copy inspect(Storage *s, SaveObject object, uint32_t offset)
167{
168 Copy result = {.offset = offset};
169 uint8_t header[HEADER_BYTES];
170 if (!s->read(s->context, offset, header, sizeof(header))) {
171 result.read_error = true;
172 return result;
173 }
174 /* La marque occupe un double mot dédié, vierge jusqu'à la validation. */
175 if (get32(header + 24) != (uint32_t)COMMIT_MAGIC ||
176 get32(header + 28) != (uint32_t)(COMMIT_MAGIC >> 32) ||
177 get32(header) != STORAGE_MAGIC || get32(header + 4) != (UINT32_C(0x100) | (uint32_t)object) ||
178 get32(header + 8) == 0U || get32(header + 8) > slots[object].span - HEADER_BYTES ||
179 get32(header + 20) != storage_crc32(header, 20)) {
180 return result;
181 }
182 result.generation = get32(header + 12);
183 result.size = get32(header + 8);
184 StoreResult payload = crc_flash(s, offset + HEADER_BYTES, result.size, get32(header + 16));
185 result.valid = payload == STORE_OK;
186 result.read_error = payload == STORE_IO;
187 return result;
188}

190/**
191 * @brief Comparer deux générations avec débordement du compteur.
192 *
193 * Une différence non nulle inférieure à la moitié de l'espace uint32_t désigne la
194 * génération plus récente.
195 * @param[in] a Valeur numérique à comparer.
196 * @param[in] b Valeur numérique à comparer.
197 * @return true si a suit b selon la comparaison modulaire.
198 */
199static bool newer(uint32_t a, uint32_t b)
200{
201 /* Comparaison modulaire, y compris au passage de FFFFFFFF à 0. */
202 return a != b && a - b < UINT32_C(0x80000000);
203}

205/**
206 * @brief Choisir la dernière copie valide d'un objet.
207 *
208 * Inspecte A et B, puis compare leurs générations. Si aucune copie n'est valide, une
209 * erreur de lecture est conservée pour empêcher une écriture à l'aveugle.
210 * @param[in,out] s Port de stockage et ses callbacks synchrones. Pointeur valide.
211 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
212 * @return Copie choisie avec son état de validation.
213 */
214static Copy active_copy(Storage *s, SaveObject object)
215{
216 Slot slot = slots[object];
217 Copy a = inspect(s, object, slot.base);
218 Copy b = inspect(s, object, slot.base + slot.span);
219 if (b.valid && (!a.valid || newer(b.generation, a.generation))) {
220 return b;
221 }
222 a.read_error = a.read_error || b.read_error;
223 return a;
224}

226/**
227 * @brief Contrôler type d'objet, pointeur et longueur.
228 *
229 * Le numéro doit appartenir à la table et la charge utile tenir après les 32 octets d'en-
230 * tête dans une copie. Une taille nulle est rejetée.
231 * @param[in] object Identifiant stable d’objet, strictement inférieur à SAVE_OBJECTS.
232 * @param[in] data Tampon valide contenant size octets de données.
233 * @param[in] size Longueur du bloc en octets.
234 * @return true si la requête tient dans le stockage attribué à l'objet.
235 */
236static bool valid_request(SaveObject object, const void *data, size_t size)
237{
238 return (unsigned)object < SAVE_OBJECTS && data != NULL && size > 0 &&
239 size <= slots[object].span - HEADER_BYTES;
240}

242/*
243 * Lire un objet dont la longueur exacte est connue.
244 *
245 * Sélectionne d'abord la copie récente valide, puis vérifie que sa longueur égale celle
246 * demandée avant de copier les octets vers l'appelant.
247 */
248StoreResult storage_read(Storage *s, SaveObject object, void *data, size_t size)
249{
250 if (!valid_request(object, data, size)) {
251 return STORE_INVALID;
252 }
253 if (s == NULL || s->read == NULL) {
254 return STORE_IO;
255 }
256 Copy active = active_copy(s, object);
257 if (!active.valid) {
258 return active.read_error ? STORE_IO : STORE_EMPTY;
259 }
260 if (active.size != size) {
261 return STORE_EMPTY;
262 }
263 return s->read(s->context, active.offset + HEADER_BYTES, data, size) ? STORE_OK : STORE_IO;
264}

266/*
267 * Lire un objet en découvrant sa longueur de format.
268 *
269 * Le tampon est donné avec sa capacité. La taille réellement enregistrée est fournie
270 * seulement au succès. Une copie trop grande est rejetée avant la copie des données.
271 */
272StoreResult storage_read_sized(Storage *s, SaveObject object, void *data, size_t capacity, size_t *actual)
273{
274 if (!valid_request(object, data, capacity) || actual == NULL) {
275 return STORE_INVALID;
276 }
277 if (s == NULL || s->read == NULL) {
278 return STORE_IO;
279 }
280 Copy active = active_copy(s, object);
281 if (!active.valid) {
282 return active.read_error ? STORE_IO : STORE_EMPTY;
283 }
284 if (active.size > capacity) {
285 return STORE_INVALID;
286 }
287 if (!s->read(s->context, active.offset + HEADER_BYTES, data, active.size)) {
288 return STORE_IO;
289 }
290 *actual = active.size;
291 return STORE_OK;
292}

294/*
295 * Écrire et valider la copie inactive d'un objet.
296 *
297 * Garde la copie active, efface l'autre, écrit l'en-tête incomplet et les données, relit
298 * les CRC puis programme le marqueur final. La génération active est reconnue même si la
299 * nouvelle longueur de format diffère.
300 */
301StoreResult storage_save(Storage *s, SaveObject object, const void *data, size_t size)
302{
303 if (!valid_request(object, data, size)) {
304 return STORE_INVALID;
305 }
306 if (s == NULL || s->read == NULL || s->erase == NULL || s->program == NULL) {
307 return STORE_IO;
308 }
309 Slot slot = slots[object];
310 /* La longueur peut évoluer. Garder la dernière copie valide de l'ancien
311 * format jusqu'à la validation complète de la nouvelle copie. */
312 Copy active = active_copy(s, object);
313 if (!active.valid && active.read_error) {
314 return STORE_IO;
315 }
316 uint32_t dest = slot.base;
317 if (active.valid && active.offset == dest) {
318 dest += slot.span;
319 }
320
321 /* Offsets de l’en-tête en octets : 0 signature, 4 type/version, 8 longueur, 12 génération, 16 CRC données, 20 CRC en-tête, 24 marqueur final. */
322 uint8_t header[HEADER_BYTES];
323 put32(header, STORAGE_MAGIC);
324 put32(header + 4, UINT32_C(0x100) | (uint32_t)object);
325 put32(header + 8, (uint32_t)size);
326 put32(header + 12, active.valid ? active.generation + 1U : 0U);
327 uint32_t crc = storage_crc32(data, size);
328 put32(header + 16, crc);
329 put32(header + 20, storage_crc32(header, 20));
330 put32(header + 24, (uint32_t)COMMIT_MAGIC);
331 put32(header + 28, (uint32_t)(COMMIT_MAGIC >> 32));
332
333 if (!s->erase(s->context, dest, slot.span)) {
334 return STORE_IO;
335 }
336 /* Seuls les 24 premiers octets sont écrits ici. Le double mot de validation reste effacé pendant toute la copie. */
337 for (uint32_t i = 0; i < META_BYTES; i += 8) {
338 if (!s->program(s->context, dest + i, header + i)) {
339 return STORE_IO;
340 }
341 }
342 const uint8_t *source = data;
343 for (size_t i = 0; i < size; i += 8) {
344 uint8_t word[8];
345 /* La flash se programme par double mot de huit octets. Le dernier bloc est complété avec l’état effacé 0xFF. */
346 memset(word, 0xFF, sizeof(word));
347 size_t n = size - i < sizeof(word) ? size - i : sizeof(word);
348 memcpy(word, source + i, n);
349 if (!s->program(s->context, dest + HEADER_BYTES + (uint32_t)i, word)) {
350 return STORE_IO;
351 }
352 }
353
354 /* Relire les métadonnées et le CRC des données avant d’écrire la marque. Une coupure avant cette marque laisse l’ancienne copie active. */
355 uint8_t check[META_BYTES];
356 if (!s->read(s->context, dest, check, sizeof(check)) ||
357 memcmp(check, header, sizeof(check)) != 0 ||
358 crc_flash(s, dest + HEADER_BYTES, size, crc) != STORE_OK ||
359 !s->program(s->context, dest + META_BYTES, header + META_BYTES)) {
360 return STORE_IO;
361 }
362 return inspect(s, object, dest).valid ? STORE_OK : STORE_IO;
