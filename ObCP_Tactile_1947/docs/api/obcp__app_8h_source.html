1/**
2 * @file obcp_app.h
3 * @brief Écrans, commandes tactiles et requêtes de sauvegarde.
4 * @ingroup obcp_apps
5 *
6 * App réunit les modèles en RAM. Les événements modifient ces modèles et invalident des
7 * rectangles. Le dessin reçoit ensuite une bande RGB565. Les accès physiques à la flash
8 * sont différés vers board.c pour laisser afficher leur état.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */

72/**
73 * @brief Valeurs présentées dans les pages de réglage.
74 *
75 * Les pilotes et board.c mettent ces valeurs à jour. Le rendu les lit sans accès au
76 * matériel.
77 */
78typedef struct {
79 uint16_t raw_x; /**< Dernière abscisse native du contrôleur tactile, avant transformation. */
80 uint16_t raw_y; /**< Dernière ordonnée native du contrôleur tactile. */
81 uint16_t z1; /**< Première mesure de pression, en unités ADC propres au pilote. */
82 uint16_t z2; /**< Deuxième mesure de pression. Zéro lorsque le pilote ne la fournit pas. */
83 uint16_t tft_id; /**< Identifiant lu du TFT, normalement 0x9341. */
84 uint16_t touch_id; /**< Identifiant du contrôleur s'il en possède un. Zéro pour TSC2007. */
85 int16_t x; /**< Dernière abscisse écran valide, en pixels. */
86 int16_t y; /**< Dernière ordonnée écran valide, en pixels. */
87 uint32_t frames_per_second; /**< Lots de rectangles achevés par seconde, hors compteur d'images complet. */
88 uint32_t pixels_per_second; /**< Pixels acheminés vers le TFT par seconde. */
89 uint32_t lcd_errors; /**< Nombre d'erreurs signalées par le moteur de transport TFT. */
90 uint32_t audio_missed; /**< Nombre cumulé d'échéances ou erreurs audio. */
91 uint32_t last_save_ms; /**< Durée de la dernière opération de stockage, en millisecondes. */
92 TouchController touch_controller; /**< Type de tactile détecté et choix du protocole à utiliser. */
93 uint8_t hits; /**< Masque des cinq cibles touchées. Tous les bits attendus donnent 0x1F. */
94 bool touch_ok; /**< true quand le tactile reconnu est disponible. */
95 bool flash_ok; /**< true quand le port de stockage physique a été obtenu. */
96 bool accel_ok; /**< true quand un LIS3DH ou LIS3DSH a été configuré. */
97 const char *accel_name; /**< Nom constant du capteur, conservé par son pilote. */
98} Diagnostics;

100/**
101 * @brief État complet des applications et de leur navigation.
102 *
103 * Un seul App appartient à board.c. Les écrans partagent le tactile, le filtre
104 * d’inclinaison, le moteur audio et le service de stockage. Aucune allocation dynamique
105 * n’est requise.
106 */
107typedef struct {
108 Damage dirty; /**< Rectangles modifiés en attente d'un prochain lot de rendu. */
109 Paint paint; /**< Pixels indexés, palette, outils et état de modification du dessin. */
110 Maze maze; /**< Positions, vitesses, temps et phase du labyrinthe. */
111 Breakout breakout; /**< Balle, raquette, masques de briques et phase du casse-briques. */
112 Race race; /**< Joueur, trafic, objectif et animation de collision de la course. */
113 Audio audio; /**< Motif actif et états des oscillateurs audio. */
114 PatternBook book; /**< Quatre motifs musicaux chargés ou sauvegardés en RAM. */
115 TouchCalBook touch_cal; /**< Deux profils résistifs, un pour chaque famille prise en charge. */
116 TouchCalSession touch_cal_session; /**< Progression du calibrage brut à quatre cibles. */
117 Tilt tilt; /**< Neutre, filtre et commandes d'inclinaison communes aux jeux. */
118 TiltGuide tilt_guide; /**< Candidat de repère proposé par l'assistant droite puis bas. */
119 Diagnostics diag; /**< Mesures et compteurs montrés au diagnostic. */
120 Screen screen; /**< Écran actuellement sélectionné. */
121 Screen splash_target; /**< Écran à ouvrir après l'accueil, menu ou procédure de démarrage. */
122 Modal modal; /**< Dialogue affiché au-dessus de l'écran courant. */
123 StoreJob job; /**< Opération de stockage différée, ou JOB_NONE. */
124 const char *notice; /**< Message constant affiché dans le bandeau, ou NULL. */
125 uint8_t slot; /**< Emplacement choisi. Dessin de 0 à 1, motif de 0 à 3. */
126 uint8_t track; /**< Piste musicale sélectionnée de 0 à 3. */
127 uint8_t note; /**< Hauteur de clavier sélectionnée de 1 à 8. */
128 uint8_t shown_step; /**< Dernier pas de séquenceur dessiné, pour invalider son ancienne case. */
129 uint8_t test_color; /**< Mire de diagnostic de 0 à 4. Zéro désigne les cibles. */
130 uint8_t rgb[3]; /**< Composantes de travail rouge, vert et bleu, chacune de 0 à 255. */
131 uint16_t splash_ms; /**< Temps cumulé de l'accueil, en millisecondes de simulation. */
132 bool keys; /**< true lorsque la groovebox présente le clavier au lieu des pas. */
133 bool cal_can_cancel; /**< true si un ancien profil permet de viser le bouton Annuler. */
134} App;

136/**
137 * @brief Initialiser les modèles et les valeurs de départ.
138 *
139 * Met les états à zéro, prépare le crayon et la palette, initialise l'audio et les axes,
140 * puis demande une image complète. La lecture des objets persistants est une étape
141 * séparée.
142 * @param[out] a Contexte des applications en RAM, à initialiser par cet appel. Pointeur
143 * valide.
144 */
145void app_init(App *a);
146/**
147 * @brief Effectuer une transition vers un autre écran.
148 *
149 * Arrête les voix, ferme les dialogues et annule les gestes en cours. Prépare le jeu ou
150 * l'assistant demandé, puis invalide l'écran. Les objets de dessin et les réglages restent
151 * dans App.
152 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
153 * @param[in] screen Écran de destination.
154 */
155void app_open(App *a, Screen screen);
156/**
157 * @brief Afficher l'accueil en conservant sa destination.
158 *
159 * Mémorise l'écran qui devait s'ouvrir, remet le compteur à zéro et choisit l'accueil.
160 * app_step() termine l'accueil après 3,2 s. Les capteurs continuent d'être servis par
161 * board.c.
162 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
163 */
164void app_start_splash(App *a);
165/**
166 * @brief Ouvrir le calibrage résistif de secours.
167 *
168 * Le bouton d'annulation fondé sur l'ancien profil est désactivé car ce profil peut
169 * justement être erroné. Les cibles seront traitées dans le repère brut.
170 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
171 */
172void app_force_touch_calibration(App *a);
173/**
174 * @brief Distribuer un événement tactile à l'écran approprié.
175 *
176 * Les travaux flash et le calibrage brut ont leur propre entrée. Les dialogues reçoivent
177 * ensuite les gestes, puis le canevas, le retour et les écrans. Cet ordre évite qu'un
178 * appui serve deux commandes.
179 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
180 * @param[in] touch Événement de contact en coordonnées écran.
181 */
182void app_touch(App *a, TouchEvent touch);
183/**
184 * @brief Transmettre une observation brute au calibrage.
185 *
186 * Le contact ayant ouvert l'assistant doit d'abord être relâché. Une session complète met
187 * son candidat en RAM et demande sa sauvegarde. Un ancien profil permet l'annulation lors
188 * d'une entrée normale.
189 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
190 * @param[in] sample Observation brute tactile, à remplir ou à interpréter selon le pilote.
191 */
192void app_calibration_sample(App *a, RawTouch sample);
193/**
194 * @brief Traiter le raccourci facultatif de SW1.
195 *
196 * Cette entrée reçoit déjà un événement débouncé. Elle change la mire, ouvre un diagnostic
197 * ou revient au menu selon l'écran. Les commandes tactiles couvrent l'usage avec le shield
198 * monté.
199 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
200 * @param[in] event Événement validé du bouton utilisateur.
201 */
202void app_button(App *a, ButtonEvent event);
203/**
204 * @brief Avancer les états temporels de vingt millisecondes.
205 *
206 * Le moteur du jeu actif reçoit les commandes filtrées. Les variations de score ou de
207 * temps invalident le bandeau. L'accueil et le curseur de groovebox utilisent également ce
208 * passage coopératif.
209 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
210 */
211void app_step(App *a); /* pas de simulation de 20 ms */
212/**
213 * @brief Recomposer une bande RGB565 à partir de l'état App.
214 *
215 * Choisit l'écran, ajoute le dialogue puis le message. Cette fonction est appelée
216 * synchroniquement par renderer_task() avant le départ du DMA. Aucun tampon d'écran
217 * complet n'est requis.
218 * @param[in] context Pointeur App valide présenté sous une forme opaque au moteur graphique.
219 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
220 */
221void app_render(const void *context, Gfx *g);
222/**
223 * @brief Indiquer si le transport PWM doit fonctionner.
224 *
225 * Un travail flash suspend la production. L'audio est demandé lorsque la groovebox est
226 * ouverte ou lorsque la note de diagnostic est active.
227 * @param[in] a Contexte des applications en RAM, consulté sans modification. Pointeur
228 * valide.
229 * @return true si board.c doit servir l'audio à cet instant.
230 */
231bool app_wants_audio(const App *a);
232/**
233 * @brief Charger le livre des quatre motifs au démarrage.
234 *
235 * Le CRC est contrôlé par le service de stockage, puis le décodeur contrôle la version et
236 * les valeurs musicales avant d'accepter le livre.
237 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
238 * @param[in,out] storage Port de stockage et ses callbacks synchrones. Pointeur valide.
239 */
240void app_read_book(App *a, Storage *storage);
241/**
242 * @brief Charger les profils des deux tactiles résistifs.
243 *
244 * Le profil est associé à l'orientation MADCTL. Un objet absent ou invalide laisse un
245 * livre vide, ce qui déclenchera le calibrage lors de la reconnaissance du shield.
246 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
247 * @param[in,out] storage Port de stockage et ses callbacks synchrones. Pointeur valide.
248 */
249void app_read_calibration(App *a, Storage *storage);
250/**
251 * @brief Charger le repère des jeux depuis la flash.
252 *
253 * Le petit format contient version, orientation du TFT et bits d'axes. Le masque est
254 * vérifié pour garder les valeurs de départ si le contenu ne correspond pas.
255 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
256 * @param[in,out] storage Port de stockage et ses callbacks synchrones. Pointeur valide.
257 */
258void app_read_axes(App *a, Storage *storage);
259/**
260 * @brief Charger les deux valeurs de réactivité.
261 *
262 * La version et les bornes 1 à 10 sont contrôlées avant de remplacer les valeurs de
263 * départ. Cette opération ne modifie pas le neutre du capteur.
264 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
265 * @param[in,out] storage Port de stockage et ses callbacks synchrones. Pointeur valide.
266 */
267void app_read_preferences(App *a, Storage *storage);
268/**
269 * @brief Exécuter la requête de stockage en attente.
270 *
271 * Les formats propres au dessin, au motif, au tactile et aux axes sont traités séparément.
272 * Le résultat prépare le message de fin et sera utilisé par board.c pour l'impulsion de
273 * LED.
274 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
275 * @param[in,out] storage Port de stockage et ses callbacks synchrones. Pointeur valide.
276 * @return Résultat STORE_OK, STORE_EMPTY, STORE_IO ou STORE_INVALID. STORE_EMPTY si aucun travail n'est demandé.
277 */
278StoreResult app_store_task(App *a, Storage *storage);
279/**
280 * @brief Invalider les valeurs de diagnostic qui évoluent.
281 *
282 * Les zones des mesures, de l'aperçu d'axes ou de la réponse sont ajoutées aux rectangles
283 * en attente. Les éléments fixes restent hors de cette invalidation périodique.
284 * @param[in,out] a Contexte des applications en RAM. Pointeur valide.
285 */
286void app_diag_refresh(App *a);
