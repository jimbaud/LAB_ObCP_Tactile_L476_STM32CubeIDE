53/*
54 * Vérifier qu'un rectangle possède une surface.
55 *
56 * Un rectangle vide a une largeur ou une hauteur non positive et ne demande aucun dessin.
57 */
58bool rect_valid(Rect r) { return r.w > 0 && r.h > 0; }
59
60/*
61 * Tester l'appartenance d'un point à un rectangle.
62 *
63 * Les limites droite et basse sont exclues. Deux boutons adjacents ne peuvent donc pas
64 * consommer le même pixel de frontière.
65 */
66bool rect_contains(Rect r, int x, int y)
67{
68 return x >= r.x && x < r.x + r.w && y >= r.y && y < r.y + r.h;
69}

71/*
72 * Trouver la surface commune à deux rectangles.
73 *
74 * Le plus grand bord gauche et le plus petit bord droit donnent l'intervalle horizontal.
75 * Le même calcul est appliqué à la hauteur.
76 */
77Rect rect_intersection(Rect a, Rect b)
78{
79 int x = max_int(a.x, b.x);
80 int y = max_int(a.y, b.y);
81 int w = min_int(a.x + a.w, b.x + b.w) - x;
82 int h = min_int(a.y + a.h, b.y + b.h) - y;
83 return (Rect){(int16_t)x, (int16_t)y, (int16_t)w, (int16_t)h};
84}

86/*
87 * Construire une enveloppe rectangulaire commune.
88 *
89 * L'enveloppe couvre les deux rectangles. Elle sert notamment à fusionner une ancienne et
90 * une nouvelle emprise de sprite.
91 */
92Rect rect_union(Rect a, Rect b)
93{
94 int x = min_int(a.x, b.x);
95 int y = min_int(a.y, b.y);
96 return (Rect){(int16_t)x, (int16_t)y,
97 (int16_t)(max_int(a.x + a.w, b.x + b.w) - x),
98 (int16_t)(max_int(a.y + a.h, b.y + b.h) - y)};
99}

101/*
102 * Écrire un pixel dans le tampon de bande.
103 *
104 * Le clipping précède le calcul d'offset. Le mot RGB565 est rangé octet fort puis octet
105 * faible, ordre attendu par le SPI du TFT.
106 */
107void gfx_pixel(Gfx *g, int x, int y, uint16_t color)
108{
109 if (rect_contains(g->clip, x, y)) {
110 size_t i = 2U * (size_t)((y - g->clip.y) * g->clip.w + (x - g->clip.x));
111 g->pixels[i] = (uint8_t)(color >> 8);
112 g->pixels[i + 1U] = (uint8_t)color;
113 }
114}
115
116/*
117 * Remplir la partie visible d'un rectangle.
118 *
119 * Intersecte d'abord l'emprise avec la bande, puis parcourt ses lignes. Toute écriture
120 * conserve le pas de la bande d'origine.
121 */
122void gfx_fill(Gfx *g, Rect r, uint16_t color)
123{
124 r = rect_intersection(g->clip, r);
125 if (!rect_valid(r)) {
126 return;
127 }
128 uint8_t hi = (uint8_t)(color >> 8);
129 uint8_t lo = (uint8_t)color;
130 for (int y = r.y; y < r.y + r.h; ++y) {
131 size_t offset = 2U * (size_t)((y - g->clip.y) * g->clip.w + r.x - g->clip.x);
132 uint8_t *pixel = &g->pixels[offset];
133 for (int x = 0; x < r.w; ++x) {
134 *pixel++ = hi;
135 *pixel++ = lo;
136 }
137 }
138}

140/*
141 * Dessiner les quatre côtés d'un rectangle.
142 *
143 * Quatre rectangles minces sont remplis. Les primitives sous-jacentes effectuent leur
144 * clipping même si un bord est hors de l'écran.
145 */
146void gfx_outline(Gfx *g, Rect r, uint16_t color)
147{
148 if (!rect_valid(r)) {
149 return;
150 }
151 gfx_fill(g, (Rect){r.x, r.y, r.w, 1}, color);
152 gfx_fill(g, (Rect){r.x, (int16_t)(r.y + r.h - 1), r.w, 1}, color);
153 gfx_fill(g, (Rect){r.x, r.y, 1, r.h}, color);
154 gfx_fill(g, (Rect){(int16_t)(r.x + r.w - 1), r.y, 1, r.h}, color);
155}

157/*
158 * Dessiner un disque plein autour d'un centre.
159 *
160 * Intersecte le carré englobant avec la bande, puis conserve les pixels dont dx² + dy² ne
161 * dépasse pas le rayon au carré. Le calcul évite une racine carrée et protège les bords.
162 */
163void gfx_disc(Gfx *g, int cx, int cy, int radius, uint16_t color)
164{
165 Rect box = {(int16_t)(cx - radius), (int16_t)(cy - radius),
166 (int16_t)(2 * radius + 1), (int16_t)(2 * radius + 1)};
167 box = rect_intersection(g->clip, box);
168 for (int y = box.y; y < box.y + box.h; ++y) {
169 for (int x = box.x; x < box.x + box.w; ++x) {
170 int dx = x - cx;
171 int dy = y - cy;
172 if (dx * dx + dy * dy <= radius * radius) {
173 gfx_pixel(g, x, y, color);
174 }
175 }
176 }
177}

219/**
220 * @brief Trouver les sept rangées d'un caractère.
221 *
222 * Les majuscules, minuscules, chiffres et signes utilisent des tables constantes. Un
223 * caractère absent est remplacé par la case espace.
224 * @param[in] ch Caractère à chercher dans la police.
225 * @return Pointeur vers sept octets de glyphes en lecture seule.
226 */
227static const uint8_t *glyph(char ch)
228{
229 if (ch >= 'a' && ch <= 'z') {
230 return lower_letters[ch - 'a'];
231 }
232 if (ch >= 'A' && ch <= 'Z') {
233 return letters[ch - 'A'];
234 }
235 if (ch >= '0' && ch <= '9') {
236 return digits[ch - '0'];
237 }
238 for (size_t i = 0; i < sizeof(symbols) / sizeof(symbols[0]); ++i) {
239 if (symbols[i].ch == ch) {
240 return symbols[i].rows;
241 }
242 }
243 return symbols[0].rows;
244}

246/*
247 * Remplir un triangle par tests de demi-plans.
248 *
249 * Les produits croisés indiquent de quel côté de chaque arête se trouve un pixel. Accepter
250 * les deux orientations évite d'imposer un ordre des sommets.
251 */
252void gfx_triangle(Gfx *g, int ax, int ay, int bx, int by, int cx, int cy, uint16_t color)
253{
254 int x0 = min_int(ax, min_int(bx, cx)), y0 = min_int(ay, min_int(by, cy));
255 int x1 = max_int(ax, max_int(bx, cx)), y1 = max_int(ay, max_int(by, cy));
256 Rect box = rect_intersection(g->clip, (Rect){(int16_t)x0, (int16_t)y0,
257 (int16_t)(x1 - x0 + 1), (int16_t)(y1 - y0 + 1)});
258 for (int y = box.y; y < box.y + box.h; ++y) {
259 for (int x = box.x; x < box.x + box.w; ++x) {
260 /* Un produit croisé par arête classe le pixel. Trois signes cohérents indiquent l’intérieur du triangle. */
261 int ab = (bx - ax) * (y - ay) - (by - ay) * (x - ax);
262 int bc = (cx - bx) * (y - by) - (cy - by) * (x - bx);
263 int ca = (ax - cx) * (y - cy) - (ay - cy) * (x - cx);
264 if ((ab >= 0 && bc >= 0 && ca >= 0) || (ab <= 0 && bc <= 0 && ca <= 0)) {
265 gfx_pixel(g, x, y, color);
266 }
267 }
268 }
269}

271/*
272 * Dessiner une chaîne avec une police de cinq par sept pixels.
273 *
274 * Chaque bit allumé produit un petit rectangle agrandi. L'avance de six pixels réserve un
275 * espace entre deux caractères. Le facteur d'échelle doit être positif.
276 */
277void gfx_text(Gfx *g, int x, int y, const char *text, unsigned scale, uint16_t color)
278{
279 int s = (int)scale;
280 if (s <= 0 || y + 7 * s <= g->clip.y || y >= g->clip.y + g->clip.h) {
281 return;
282 }
283 for (; *text && x < g->clip.x + g->clip.w; ++text, x += 6 * s) {
284 if (x + 5 * s <= g->clip.x) {
285 continue;
286 }
287 const uint8_t *rows = glyph(*text);
288 for (int row = 0; row < 7; ++row) {
289 for (int col = 0; col < 5; ++col) {
290 if ((rows[row] & (16U >> (unsigned)col)) != 0U) {
291 gfx_fill(g, (Rect){(int16_t)(x + col * s), (int16_t)(y + row * s),
292 (int16_t)s, (int16_t)s}, color);
293 }
294 }
295 }
296 }

299/*
300 * Demander une recomposition de tout l'écran.
301 *
302 * Remplace la liste de rectangles par LCD_RECT. Ce repli est aussi utilisé quand une liste
303 * de dommages ne peut plus contenir un changement supplémentaire.
304 */
305void damage_full(Damage *damage)
306{
307 damage->items[0] = LCD_RECT;
308 damage->count = 1;
309}

311/*
312 * Ajouter une zone à recomposer et fusionner les voisines.
313 *
314 * Limite la zone à l'écran, fusionne les enveloppes peu coûteuses puis ajoute la zone
315 * restante. Si la capacité est atteinte, une image complète remplace la liste.
316 */
317void damage_add(Damage *damage, Rect r)
318{
319 r = rect_intersection(r, LCD_RECT);
320 if (!rect_valid(r)) {
321 return;
322 }
323
324 /* Fusionner des régions proches quand le rectangle résultant est peu coûteux. */
325 unsigned i = 0;
326 while (i < damage->count) {
327 Rect other = damage->items[i];
328 Rect merged = rect_union(r, other);
329 /* La fusion est admise si elle ajoute au plus 32 pixels. Cela réduit les commandes SPI sans repeindre une grande zone vide. */
330 if (merged.w * merged.h <= r.w * r.h + other.w * other.h + 32) {
331 r = merged;
332 --damage->count;
333 /* Remplacer la case retirée par la dernière évite de décaler le tableau. Recommencer car la nouvelle emprise peut rejoindre une autre zone. */
334 damage->items[i] = damage->items[damage->count];
335 i = 0;
336 } else {
337 ++i;
338 }
339 }
340 if (damage->count == DAMAGE_CAPACITY) {
341 damage_full(damage);
342 } else {
343 damage->items[damage->count++] = r;
344 }
345}

358/*
359 * Avancer l'envoi non bloquant d'une bande.
360 *
361 * Une bande en vol est d'abord contrôlée. Sinon, un lot de rectangles est capturé et une
362 * bande de 16 lignes au plus est recomposée puis confiée au transport. Le même tampon est
363 * réutilisé seulement après la fin du transfert.
364 */
365void renderer_task(Renderer *r, Damage *pending, DrawScene draw, const void *scene)
366{
367 if (r->in_flight) {
368 TransferState state = r->port.poll(r->port.context);
369 /* Le DMA possède encore le tampon. Tout dessin à cet instant pourrait corrompre la bande envoyée. */
370 if (state == TRANSFER_BUSY) {
371 return;
372 }
373 r->in_flight = false;
374 if (state == TRANSFER_ERROR) {
375 ++r->errors;
376 r->index = r->batch.count;
377 damage_full(pending);
378 return;
379 }
380 ++r->bands;
381 r->pixels += (uint32_t)(r->sent.w * r->sent.h);
382 r->rows += r->sent.h;
383 if (r->rows == r->batch.items[r->index].h) {
384 ++r->index;
385 r->rows = 0;
386 if (r->index == r->batch.count) {
387 ++r->batches;
388 }
389 }
390 /* La prochaine itération laisse une fenêtre SPI au tactile. */
391 return;
392 }
393 if (r->index >= r->batch.count) {
394 if (pending->count == 0) {
395 return;
396 }
397 r->batch = *pending;
398 pending->count = 0;
399 r->index = 0;
400 r->rows = 0;
401 }
402
403 Rect band = r->batch.items[r->index];
404 band.y += r->rows;
405 band.h = (int16_t)min_int(band.h - r->rows, BAND_ROWS);
406 Gfx g = {band, r->buffer};
407 /* Le callback dessine la scène dans la bande seulement. Les coordonnées restent absolues dans l’écran. */
408 draw(scene, &g);
409 r->sent = band;
410 r->in_flight = r->port.send(r->port.context, band, r->buffer,
411 (uint16_t)(2 * band.w * band.h));
412 if (!r->in_flight) {
413 ++r->errors;
414 r->index = r->batch.count;
415 damage_full(pending);
416 }
417}
