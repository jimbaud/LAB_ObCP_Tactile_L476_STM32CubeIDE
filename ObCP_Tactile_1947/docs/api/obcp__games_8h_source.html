1/**
2 * @file obcp_games.h
3 * @brief Modèles, niveaux et commandes des trois jeux à inclinaison.
4 * @ingroup obcp_apps
5 *
6 * Les trois modèles partagent GamePhase et un pas de 20 ms. Le labyrinthe utilise une
7 * carte de cases, le casse-briques des masques de briques et la course un tableau
8 * d'adversaires. Les positions sont flottantes, le rendu emploie des pixels entiers.
9 * GAME_PAUSE fige la simulation. La course possède aussi un bref état de collision.
10 *
11 * @par Repères de lecture
12 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
13 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
14 */
15#ifndef OBCP_GAMES_H
16#define OBCP_GAMES_H

20/**
21 * @brief Dimensions, bornes et constantes du module.
22 */
23enum {
24 GAME_STEP_MS = 20, /**< Pas de simulation de vingt millisecondes, soit 50 pas par seconde. */
25 GAME_LEVELS = 10, /**< Dix niveaux par jeu, indices de zéro à neuf. */
26 RACE_CARS = 6, /**< Six places d'adversaires au maximum dans la course. */
27};
28/**
29 * @brief Phases de jeu partagées par les trois modèles.
30 */
31typedef enum {
32 GAME_READY, /**< Positions préparées, attente du bouton Jouer. */
33 GAME_RUN, /**< Simulation active à pas fixes. */
34 GAME_PAUSE, /**< Simulation et chronomètre suspendus. */
35 GAME_WON, /**< Objectif atteint, proposition du niveau suivant. */
36 GAME_LOST, /**< Temps expiré, balle perdue ou collision. */
37} GamePhase;

39/**
40 * @brief État physique et temporel du labyrinthe.
41 *
42 * La simulation avance par pas fixes de GAME_STEP_MS. Le rendu lit ce modèle sans
43 * l’intégrer.
44 */
45typedef struct {
46 float x; /**< Centre horizontal de la bille ou balle, en pixels flottants. */
47 float y; /**< Centre vertical de la bille ou balle, en pixels flottants. */
48 float vx; /**< Vitesse horizontale, en pixels par seconde. */
49 float vy; /**< Vitesse verticale, en pixels par seconde. */
50 uint32_t elapsed_ms; /**< Temps de jeu cumulé, en millisecondes. La pause ne l’incrémente pas. */
51 uint8_t level; /**< Indice du niveau, de 0 à GAME_LEVELS moins un. */
52 GamePhase phase; /**< Phase prête, en jeu, pause, victoire ou défaite. */
53} Maze;
54/**
55 * @brief Préparer un niveau à son point de départ.
56 *
57 * Remet vitesses et temps à zéro. Un niveau hors plage revient à zéro et la phase initiale
58 * demande l'appui Jouer.
59 * @param[out] m État physique et temporel du labyrinthe, à initialiser par cet appel.
60 * Pointeur valide.
61 * @param[in] level Indice de niveau à base zéro. Une valeur hors plage choisit le premier
62 * niveau.
63 */
64void maze_init(Maze *m, unsigned level);
65/**
66 * @brief Intégrer la physique et le temps sur vingt millisecondes.
67 *
68 * Calcule les vitesses avec inertie, avance séparément X et Y pour permettre le glissement
69 * contre un mur, puis teste arrivée et expiration. Les emprises modifiées sont invalidées.
70 * @param[in,out] m État physique et temporel du labyrinthe. Pointeur valide.
71 * @param[in] tilt_x Commande écran X, adimensionnelle entre −1 et +1.
72 * @param[in] tilt_y Commande écran Y, adimensionnelle entre −1 et +1.
73 * @param[in,out] damage Liste de rectangles à recomposer. Pointeur valide.
74 */
75void maze_step(Maze *m, float tilt_x, float tilt_y, Damage *damage);
76/**
77 * @brief Dessiner les murs, les cases spéciales et la bille.
78 *
79 * Les rangées hors de la bande sont sautées. Le niveau sélectionne la carte, tandis que
80 * les positions physiques ne sont jamais modifiées par ce rendu.
81 * @param[in] m État physique et temporel du labyrinthe, consulté sans modification.
82 * Pointeur valide.
83 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
84 */
85void maze_draw(const Maze *m, Gfx *g);
86/**
87 * @brief Obtenir la cellule de départ commune aux cartes.
88 *
89 * Le départ est placé dans la case intérieure de coordonnées colonne un et ligne un.
90 * @return Rectangle de la case de départ.
91 */
92Rect maze_start(void);
93/**
94 * @brief Obtenir la cellule d'arrivée commune aux cartes.
95 *
96 * L'arrivée occupe la dernière case intérieure en bas à droite. Le test de victoire et le
97 * dessin lisent la même emprise.
98 * @return Rectangle de la case d'arrivée.
99 */
100Rect maze_finish(void);
101/**
102 * @brief Vérifier si le disque de la bille touche une zone interdite.
103 *
104 * Limite les cases examinées à l'emprise du disque. Pour chaque mur, le point de rectangle
105 * le plus proche du centre permet de tester la distance au carré.
106 * @param[in] m État physique et temporel du labyrinthe, consulté sans modification.
107 * Pointeur valide.
108 * @param[in] x Abscisse dans le repère écran, en pixels.
109 * @param[in] y Ordonnée dans le repère écran, en pixels.
110 * @return true si la bille sortirait du plateau ou recouvrirait un mur.
111 */
112bool maze_solid(const Maze *m, float x, float y);
113/**
114 * @brief Convertir le temps restant en secondes affichables.
115 *
116 * Arrondit vers le haut pour que la dernière seconde soit visible. La valeur est nulle dès
117 * que la limite du niveau est atteinte.
118 * @param[in] m État physique et temporel du labyrinthe, consulté sans modification.
119 * Pointeur valide.
120 * @return Nombre de secondes restantes, jamais négatif.
121 */
122uint32_t maze_seconds_left(const Maze *m);

124/**
125 * @brief Balle, raquette et briques du casse-briques.
126 *
127 * La simulation avance par pas fixes de GAME_STEP_MS. Le rendu lit ce modèle sans
128 * l’intégrer.
129 */
130typedef struct {
131 float x; /**< Centre horizontal de la bille ou balle, en pixels flottants. */
132 float y; /**< Centre vertical de la bille ou balle, en pixels flottants. */
133 float vx; /**< Vitesse horizontale, en pixels par seconde. */
134 float vy; /**< Vitesse verticale, en pixels par seconde. */
135 float paddle; /**< Centre horizontal de la raquette, en pixels flottants. */
136 uint32_t bricks; /**< Masque des briques encore présentes. Un bit par brique, de 0 à 31. */
137 uint32_t hard_bricks; /**< Masque des renforts encore présents, inclus dans bricks. */
138 unsigned score; /**< Points du niveau courant, cinq par renfort et dix par destruction. */
139 uint8_t level; /**< Indice du niveau, de 0 à GAME_LEVELS moins un. */
140 GamePhase phase; /**< Phase prête, en jeu, pause, victoire ou défaite. */
141} Breakout;
142/**
143 * @brief Préparer les positions et les briques d'un niveau.
144 *
145 * Un indice hors plage revient au premier niveau. L'état est prêt à jouer, avec les
146 * masques, dimensions et vitesses choisis dans la table constante.
147 * @param[out] b Balle, raquette et briques du casse-briques, à initialiser par cet
148 * appel. Pointeur valide.
149 * @param[in] level Indice de niveau à base zéro. Une valeur hors plage choisit le premier
150 * niveau.
151 */
152void breakout_init(Breakout *b, unsigned level);
153/**
154 * @brief Avancer une simulation de vingt millisecondes.
155 *
156 * Seule GAME_RUN avance. La raquette rejoint 75 % de l'écart à sa cible, puis la balle
157 * fait trois sous-pas. Les anciennes et nouvelles emprises sont invalidées. Une balle
158 * sortie ou un masque vide termine le niveau.
159 * @param[in,out] b Balle, raquette et briques du casse-briques. Pointeur valide.
160 * @param[in] tilt_x Commande écran X, adimensionnelle entre −1 et +1.
161 * @param[in,out] damage Liste de rectangles à recomposer. Pointeur valide.
162 */
163void breakout_step(Breakout *b, float tilt_x, Damage *damage);
164/**
165 * @brief Recomposer les briques, la raquette et la balle.
166 *
167 * Le modèle est lu sans modifier la simulation. Les briques renforcées reçoivent un cadre
168 * clair. Le clipping des primitives limite le travail à la bande demandée.
169 * @param[in] b Balle, raquette et briques du casse-briques, consulté sans modification.
170 * Pointeur valide.
171 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
172 */
173void breakout_draw(const Breakout *b, Gfx *g);
174/**
175 * @brief Calculer l'emprise d'une brique.
176 *
177 * L'indice va de 0 à 31. Le quotient et le reste de la division par huit donnent la rangée
178 * et la colonne.
179 * @param[in] i Indice de l’élément dans la table de géométrie décrite ci-dessus.
180 * @return Rectangle de la brique, en pixels.
181 */
182Rect breakout_brick(unsigned i);

187typedef enum {

189 VEHICLE_BIKE, /**< Moto étroite, emprise 15 × 27 pixels. */
190 VEHICLE_VAN, /**< Fourgonnette, emprise 35 × 44 pixels. */
191 VEHICLE_TRUCK, /**< Camion, emprise 43 × 56 pixels. */
192} Vehicle;
193/**
194 * @brief Adversaire occupant une voie de la course.
195 *
196 * Le type détermine largeur et hauteur. Une place inactive peut être réutilisée par le
197 * générateur de trafic.
198 */
199typedef struct {
200 float y; /**< Ordonnée du bord supérieur du véhicule, en pixels flottants. */
201 uint8_t lane; /**< Voie de 0 à 2, utilisée pour retrouver le centre horizontal. */
202 bool active; /**< true si le véhicule doit avancer et être dessiné. */
203 Vehicle kind; /**< Voiture, moto, fourgonnette ou camion. */
204} Rival;
205/**
206 * @brief État de la route, du joueur et de la progression.
207 *
208 * Les adversaires partagent la vitesse du niveau. Le compte à rebours de collision reste
209 * animé après la défaite.
210 */
211typedef struct {
212 Rival rival[RACE_CARS]; /**< Places des adversaires, actives ou libres. */
213 float player_x; /**< Centre horizontal du joueur, en pixels flottants. */
214 float scroll; /**< Décalage vertical des marquages modulo leur espacement de 38 pixels. */
215 uint16_t spawn_ms; /**< Délai restant avant une tentative d'apparition, en millisecondes. */
216 uint32_t seed; /**< État du générateur pseudo-aléatoire déterministe du trafic. */
217 uint32_t score; /**< Nombre de véhicules entièrement dépassés. */
218 uint8_t last_lane; /**< Voie de la précédente apparition pour éviter sa répétition immédiate. */
219 uint8_t level; /**< Indice de difficulté, de 0 à 9. */
220 uint16_t crash_ms; /**< Durée restante de l'éclat de collision, en millisecondes. */
221 int16_t impact_x; /**< Centre horizontal de l'éclat BOUM, en pixels. */
222 int16_t impact_y; /**< Centre vertical de l'éclat BOUM, en pixels. */
223 GamePhase phase; /**< Phase de la partie commune aux trois jeux. */
224} Race;
225/**

230 * @param[out] r Joueur, adversaires et progression de la course, à initialiser par cet
231 * appel. Pointeur valide.
232 * @param[in] level Indice de niveau à base zéro. Une valeur hors plage choisit le premier
233 * niveau.

235void race_init(Race *r, unsigned level);
236/**
237 * @brief Lire l'objectif de dépassements du niveau.

242 * modification. Pointeur valide.
243 * @return Nombre de véhicules à dépasser pour gagner le niveau.

245unsigned race_goal(const Race *r);
246/**

253 * @param[in] tilt_x Commande écran X, adimensionnelle entre −1 et +1.
254 * @param[in,out] damage Liste de rectangles à recomposer. Pointeur valide.

256void race_step(Race *r, float tilt_x, Damage *damage);

264 * @param[in,out] g Vue de la bande RGB565 et de son tampon accessible en écriture. Pointeur valide.
