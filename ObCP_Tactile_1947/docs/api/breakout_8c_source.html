2 * @file breakout.c
3 * @brief Physique et dessin du casse-briques à dix niveaux.
4 * @ingroup obcp_apps
5 *
6 * Un bit de bricks représente une brique. hard_bricks indique un renfort à retirer au
7 * premier impact. La raquette suit une cible de position. La trajectoire de balle est
8 * divisée en trois sous-pas pour borner le déplacement entre collisions.

10 * @par Repères de lecture

12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.

48/*
49 * Calculer l'emprise d'une brique.
50 *
51 * L'indice va de 0 à 31. Le quotient et le reste de la division par huit donnent la rangée
52 * et la colonne.
53 */
54Rect breakout_brick(unsigned i)
55{
56 return (Rect){(int16_t)(16 + (i % 8U) * 36U), (int16_t)(56 + (i / 8U) * 17U), 32, 13};
57}

59/**
60 * @brief Calculer l'emprise de la raquette.
61 *
62 * La largeur dépend du niveau. La position conservée dans le modèle désigne le centre,
63 * alors que le rectangle utilise son bord gauche.
64 * @param[in] b Balle, raquette et briques du casse-briques, consulté sans modification.
65 * Pointeur valide.
66 * @return Rectangle occupé par la raquette.
67 */
68static Rect paddle_bounds(const Breakout *b)
69{
70 int width = levels[b->level].paddle_width;
71 return (Rect){(int16_t)((int)b->paddle - width / 2), PADDLE_Y, (int16_t)width, 7};
72}

74/**
75 * @brief Calculer l'ancienne ou la nouvelle emprise de balle.
76 *
77 * Le rayon est ajouté de part et d'autre du centre pour que le rectangle englobe aussi les
78 * pixels du bord.
79 * @param[in] b Balle, raquette et briques du casse-briques, consulté sans modification.
80 * Pointeur valide.
81 * @return Rectangle englobant la balle.
82 */
83static Rect ball_bounds(const Breakout *b)
84{
85 return (Rect){(int16_t)((int)b->x - BALL_R), (int16_t)((int)b->y - BALL_R),
86 2 * BALL_R + 1, 2 * BALL_R + 1};
87}

91 *
92 * Un indice hors plage revient au premier niveau. L'état est prêt à jouer, avec les

104/**
105 * @brief Avancer la balle d'un sous-pas et traiter ses impacts.
106 *
107 * Teste murs, raquette puis briques. L'ancienne position aide à choisir la composante de
108 * vitesse à inverser. Un impact sur renfort enlève d'abord ce renfort.
109 * @param[in,out] b Balle, raquette et briques du casse-briques. Pointeur valide.
110 * @param[in] dt Durée du sous-pas, en secondes.
111 * @param[in,out] damage Liste de rectangles à recomposer. Pointeur valide.
112 */
113static void move_ball(Breakout *b, float dt, Damage *damage)
114{
115 float old_x = b->x;
116 float old_y = b->y;
117 b->x += b->vx * dt;
118 b->y += b->vy * dt;
119 if (b->x < LEFT + BALL_R) {
120 b->x = LEFT + BALL_R;
121 b->vx = -b->vx;
122 } else if (b->x > RIGHT - BALL_R) {
123 b->x = RIGHT - BALL_R;
124 b->vx = -b->vx;
125 }
126 if (b->y < TOP + BALL_R) {
127 b->y = TOP + BALL_R;
128 b->vy = -b->vy;
129 }
130 int half = levels[b->level].paddle_width / 2;
131 if (b->vy > 0.0f && old_y + BALL_R <= PADDLE_Y && b->y + BALL_R >= PADDLE_Y &&
132 b->x + BALL_R >= b->paddle - half && b->x - BALL_R <= b->paddle + half) {
133 b->y = PADDLE_Y - BALL_R;
134 b->vx = 5.4f * (b->x - b->paddle);
135 b->vy = levels[b->level].bounce;
136 }
137 for (unsigned i = 0; i < 32; ++i) {
138 uint32_t bit = UINT32_C(1) << i;
139 Rect brick = breakout_brick(i);
140 if ((b->bricks & bit) == 0U || !rect_valid(rect_intersection(brick, ball_bounds(b)))) {
141 continue;
142 }
143 if ((b->hard_bricks & bit) != 0U) {
144 b->hard_bricks &= ~bit;
145 b->score += 5U;
146 } else {
147 b->bricks &= ~bit;
148 b->score += 10U;
149 }
150 if (old_y + BALL_R <= brick.y || old_y - BALL_R >= brick.y + brick.h) {
151 b->y = old_y;
152 b->vy = -b->vy;
153 } else {
154 b->x = old_x;
155 b->vx = -b->vx;
156 }
157 damage_add(damage, brick);
158 break;
159 }
160}

162/*
163 * Avancer une simulation de vingt millisecondes.
164 *
165 * Seule GAME_RUN avance. La raquette rejoint 75 % de l'écart à sa cible, puis la balle
166 * fait trois sous-pas. Les anciennes et nouvelles emprises sont invalidées. Une balle
167 * sortie ou un masque vide termine le niveau.
168 */
169void breakout_step(Breakout *b, float tilt_x, Damage *damage)
170{
171 if (b->phase != GAME_RUN) {
172 return;
173 }
174 Rect old_paddle = paddle_bounds(b);
175 Rect old_ball = ball_bounds(b);
176 int half = levels[b->level].paddle_width / 2;
177 /* Une inclinaison désigne maintenant une position. Le joueur peut viser
178 * rapidement un bord sans attendre un déplacement à vitesse constante.
179 * Le filtre d'entrée et son gain X sont communs aux trois jeux. */
180 float target = (LEFT + RIGHT) / 2.0f + ((RIGHT - LEFT) / 2.0f - half) * tilt_x;
181 b->paddle += 0.75f * (target - b->paddle);
182 if (b->paddle < LEFT + half) {
183 b->paddle = LEFT + half;
184 } else if (b->paddle > RIGHT - half) {
185 b->paddle = RIGHT - half;
186 }
187
188 /* Trois sous-pas bornent le déplacement entre deux tests de collision. */
189 for (unsigned i = 0; i < 3; ++i) {
190 move_ball(b, GAME_STEP_MS / 3000.0f, damage);
191 }
192 damage_add(damage, old_paddle);
193 damage_add(damage, paddle_bounds(b));
194 damage_add(damage, old_ball);
195 damage_add(damage, ball_bounds(b));
196 if (b->bricks == 0U || b->y - BALL_R >= LCD_HEIGHT) {
197 b->phase = b->bricks == 0U ? GAME_WON : GAME_LOST;
198 damage_full(damage);
199 }
200}

202/*
203 * Recomposer les briques, la raquette et la balle.
204 *
205 * Le modèle est lu sans modifier la simulation. Les briques renforcées reçoivent un cadre
206 * clair. Le clipping des primitives limite le travail à la bande demandée.
207 */
208void breakout_draw(const Breakout *b, Gfx *g)
209{
210 static const uint16_t colors[4] = {
211 RGB565(244,100,120), RGB565(248,177,66), RGB565(45,202,178), RGB565(80,143,246)
212 };
213 gfx_outline(g, (Rect){LEFT - 1, TOP - 1, RIGHT - LEFT + 2, LCD_HEIGHT - TOP + 1},
214 RGB565(56,85,116));
215 for (unsigned i = 0; i < 32; ++i) {
216 if ((b->bricks & (UINT32_C(1) << i)) != 0) {
217 gfx_fill(g, breakout_brick(i), colors[i / 8U]);
218 if ((b->hard_bricks & (UINT32_C(1) << i)) != 0U) {
219 Rect brick = breakout_brick(i);
220 gfx_outline(g, (Rect){brick.x + 3, brick.y + 3, brick.w - 6, brick.h - 6},
221 RGB565(250,248,240));
222 }
223 }
224 }
225 gfx_fill(g, paddle_bounds(b), RGB565(226,241,252));
226 gfx_disc(g, (int)b->x, (int)b->y, BALL_R, RGB565(252,209,82));
227}
