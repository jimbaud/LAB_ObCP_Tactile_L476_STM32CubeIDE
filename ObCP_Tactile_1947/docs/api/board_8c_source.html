141/**
142 * @brief Démarrer une impulsion de LED d'état.
143 *
144 * Sélectionne vert pendant 300 ms pour un succès ou rouge pendant 500 ms pour un échec. Le
145 * début et la durée sont enregistrés pour une extinction ultérieure sans attente active.
146 * @param[in] ok true pour un état réussi, false pour signaler un échec.
147 * @param[in] now Tick courant en millisecondes non signées.
148 */
149static void led_signal(bool ok, uint32_t now)
150{
151 HAL_GPIO_WritePin(LED_GREEN_GPIO_Port, LED_GREEN_Pin, ok ? GPIO_PIN_SET : GPIO_PIN_RESET);
152 HAL_GPIO_WritePin(LED_RED_GPIO_Port, LED_RED_Pin, ok ? GPIO_PIN_RESET : GPIO_PIN_SET);
153 led_since = now;
154 led_duration = ok ? 300U : 500U;
155}

157/**
158 * @brief Éteindre les sorties de LED à l'échéance.
159 *
160 * La soustraction de ticks non signés conserve le délai lors du débordement du compteur.
161 * Une durée nulle indique qu'aucune impulsion n'est active.
162 * @param[in] now Tick courant en millisecondes non signées.
163 */
164static void led_update(uint32_t now)
165{
166 if (led_duration != 0U && now - led_since >= led_duration) {
167 HAL_GPIO_WritePin(LED_GREEN_GPIO_Port, LED_GREEN_Pin, GPIO_PIN_RESET);
168 HAL_GPIO_WritePin(LED_RED_GPIO_Port, LED_RED_Pin, GPIO_PIN_RESET);
169 led_duration = 0;
170 }
171}

173/**
174 * @brief Actualiser la reconnaissance du tactile.
175 *
176 * Le sondage peut être différé par SPI1 occupé. Un nouveau contrôleur réinitialise le
177 * filtre, met à jour le diagnostic et peut ouvrir un calibrage si son profil manque.
178 * @param[in] auto_calibrate true pour proposer le calibrage d’un nouveau contrôleur résistif sans
179 * profil.
180 * @return true si le sondage a été effectué, même si aucun contrôleur n'a été reconnu.
181 */
182static bool probe_touch(bool auto_calibrate)
183{
184 TouchDevice detected;
185 if (!touch_probe(&detected)) {
186 return false;
187 }
188 bool ready = detected.controller != TOUCH_CTRL_NONE;
189 bool changed = ready && detected.controller != touch_device.controller;
190 if (changed || app.diag.touch_ok != ready) {
191 damage_full(&app.dirty);
192 }
193 app.diag.touch_ok = ready;
194 if (ready) {
195 touch_device = detected;
196 app.diag.touch_controller = detected.controller;
197 app.diag.touch_id = detected.id;
198 if (changed) {
199 app.diag.hits = 0;
200 touch_filter_state = (ResistiveFilter){0};
201 if (app.screen == SCREEN_TOUCH_CAL) {
202 app_open(&app, touch_resistive(detected.controller) ? SCREEN_TOUCH_CAL : SCREEN_DIAG);
203 } else if (auto_calibrate && (app.screen == SCREEN_MENU || app.screen == SCREEN_SPLASH) &&
204 touch_resistive(detected.controller) &&
205 touch_cal_get(&app.touch_cal, detected.controller) == NULL) {
206 if (app.screen == SCREEN_SPLASH) {
207 app.splash_target = SCREEN_TOUCH_CAL;
208 } else {
209 app_open(&app, SCREEN_TOUCH_CAL);
210 }
211 }
212 }
213 }
214 touch_failures = 0;
215 return true;
216}

218/*
219 * Initialiser l'application après les périphériques CubeMX.
220 *
221 * Charge les objets persistants, initialise TFT et capteur, sonde le tactile, prépare
222 * l'accueil et les bases de temps. À appeler une fois depuis le bloc USER CODE 2 de
223 * main().
224 */
225void obcp_start(void)
226{
227 bool boot_pressed = HAL_GPIO_ReadPin(USER_BUTTON_GPIO_Port, USER_BUTTON_Pin) == GPIO_PIN_RESET;
228 app_init(&app);
229 pwm_audio_init();
230 flash = obcp_flash_storage();
231 app.diag.flash_ok = flash.read != NULL;
232 app_read_book(&app, &flash);
233 app_read_calibration(&app, &flash);
234 app_read_axes(&app, &flash);
235 app_read_preferences(&app, &flash);
236
237 lcd_bus_ok = lcd_init(&app.diag.tft_id);
238 renderer.port = lcd_port();
239 (void)probe_touch(!boot_pressed);
240 app.diag.accel_ok = accel_init(&hspi3) != ACCEL_NONE;
241 app.diag.accel_name = accel_chip_name();
242
243 if (boot_pressed) {
244 app_open(&app, SCREEN_DIAG);
245 }
246 uint32_t now = HAL_GetTick();
247 button_init(&button, boot_pressed, now);
248 boot_touch.armed = app.screen == SCREEN_MENU && touch_resistive(touch_device.controller);
249 app_start_splash(&app);
250 led_signal(lcd_bus_ok && app.diag.touch_ok && app.diag.accel_ok, now);
251 last_sample = now;
252 last_game = now;
253 last_rates = now;
254 last_touch = now;
255 last_touch_probe = now;
256}

258/**
259 * @brief Faire avancer une acquisition tactile et distribuer son résultat.
260 *
261 * WAIT poursuit une conversion ou attend le bus. EMPTY garde le dernier contact. Une
262 * erreur répétée relance la reconnaissance. Les entrées de secours, le calibrage brut et
263 * les coordonnées écran sont traités dans cet ordre.
264 * @param[in] now Tick courant en millisecondes non signées.
265 */
266static void sample_touch(uint32_t now)
267{
268 if (!touch_sampling) {
269 if (now - last_touch < 10U) {
270 return;
271 }
272 last_touch = now;
273 }
274 RawTouch raw;
275 TouchRead result = touch_read(&touch_device, &raw);
276 /* WAIT annonce une conversion en cours ou un bus occupé. Il faut revenir sans inventer un relâchement du doigt. */
277 touch_sampling = result == TOUCH_READ_WAIT;
278 if (touch_sampling) {
279 return;
280 }
281 if (result == TOUCH_READ_ERROR) {
282 ++obcp_live.touch_errors;
283 if (++touch_failures >= 3U) {
284 app.diag.touch_ok = false;
285 boot_touch.armed = false;
286 last_touch_probe = now;
287 touch_filter_state = (ResistiveFilter){0};
288 app_touch(&app, touch_update(&touch, false, 0, 0));
289 if (app.screen == SCREEN_TOUCH_CAL) {
290 touch_cal_interrupt(&app.touch_cal_session);
291 damage_full(&app.dirty);
292 }
293 }
294 return;
295 }
296 touch_failures = 0;
297 /* EMPTY signifie absence de nouvel échantillon dans la FIFO. Le dernier état tactile reste utilisable. */
298 if (result == TOUCH_READ_EMPTY) {
299 return;
300 }
301 app.diag.z1 = raw.z1;
302 app.diag.z2 = raw.z2;
303 if (raw.down) {
304 app.diag.raw_x = raw.point.x;
305 app.diag.raw_y = raw.point.y;
306 }
307 if (app.screen != SCREEN_MENU && app.screen != SCREEN_SPLASH) {
308 boot_touch.armed = false;
309 }
310 BootTouchEvent boot_event = touch_boot_sample(&boot_touch, raw.down, now);
311 if (boot_event == BOOT_TOUCH_CALIBRATE) {
312 app_force_touch_calibration(&app);
313 }
314 if (boot_event != BOOT_TOUCH_NONE) {
315 return;
316 }
317 if (app.screen == SCREEN_SPLASH) {
318 touch_wait_release = raw.down;
319 return;
320 }
321 if (app.screen == SCREEN_TOUCH_CAL) {
322 /* Le calibrage travaille dans le repère brut et consomme ce geste. */
323 touch = (TouchState){0};
324 touch_filter_state = (ResistiveFilter){0};
325 touch_wait_release = raw.down;
326 app_calibration_sample(&app, raw);
327 return;
328 }
329 /* Consommer le relâchement du geste ayant traversé l’accueil ou le calibrage évite un clic involontaire sur la nouvelle page. */
330 if (touch_wait_release) {
331 touch_wait_release = raw.down;
332 return;
333 }
334
335 bool resistive = touch_resistive(touch_device.controller);
336 if (resistive) {
337 raw = touch_filter(&touch_filter_state, raw);
338 }
339 bool pressed = raw.down;
340 int16_t x = 0, y = 0;
341 if (pressed) {
342 if (resistive) {
343 const TouchCalibration *cal = touch_cal_get(&app.touch_cal, touch_device.controller);
344 pressed = touch_cal_map(cal, raw.point, &x, &y);
345 } else {
346 pressed = touch_landscape(raw.point.x, raw.point.y, OBCP_TOUCH_FLIPPED, &x, &y);
347 }
348 }
349 if (pressed) {
350 app.diag.x = x;
351 app.diag.y = y;
352 }
353 app_touch(&app, touch_update(&touch, pressed, x, y));
354}

356/**
357 * @brief Servir le tactile et échantillonner l'accéléromètre.
358 *
359 * Vise un départ toutes les 10 ms pour le capteur. Met à jour le filtre et fait avancer
360 * l'assistant d'axes quand sa page est ouverte. Les transitions de disponibilité
361 * invalident la scène.
362 * @param[in] now Tick courant en millisecondes non signées.
363 */
364static void sample_inputs(uint32_t now)
365{
366 if (app.diag.touch_ok) {
367 sample_touch(now);
368 } else if (now - last_touch_probe >= 500U && probe_touch(true)) {
369 last_touch_probe = now;
370 }
371 if (now - last_sample < 10U) {
372 return;
373 }
374 last_sample = now;
375 if (app.diag.accel_ok) {
376 bool ready = app.tilt.calibrated && app.tilt.valid;
377 accel_g_t g = {0};
378 bool ok = accel_read(&g);
379 if (ok) {
380 ++obcp_live.accel_reads;
381 }
382 tilt_sample(&app.tilt, ok, g.x, g.y, g.z);
383 bool guide_changed = app.screen == SCREEN_TILT && tilt_guide_sample(&app.tilt_guide, &app.tilt);
384 if (guide_changed || ready != (app.tilt.calibrated && app.tilt.valid)) {
385 damage_full(&app.dirty);
386 }
387 }
388}

403/**
404 * @brief Publier les mesures et les compteurs de diagnostic.
405 *
406 * Les débits sont calculés sur un intervalle d'au moins une seconde. Les autres champs
407 * copient l'état du dernier contact, des pilotes et du stockage vers obcp_live.
408 * @param[in] now Tick courant en millisecondes non signées.
409 */
410static void update_rates(uint32_t now)
411{
412 /* La soustraction de ticks uint32_t reste correcte au rebouclage, tant que les intervalles observés restent courts devant 2^32 ms. */
413 uint32_t elapsed = now - last_rates;
414 if (elapsed >= 1000U) {
415 app.diag.frames_per_second = (renderer.batches - last_frames) * 1000U / elapsed;
416 app.diag.pixels_per_second = (renderer.pixels - last_pixels) * 1000U / elapsed;
417 last_frames = renderer.batches;
418 last_pixels = renderer.pixels;
419 last_rates = now;
420 obcp_live.frames_per_second = app.diag.frames_per_second;
421 obcp_live.pixels_per_second = app.diag.pixels_per_second;
422 }
423 app.diag.audio_missed = pwm_audio_missed();
424 app.diag.lcd_errors = renderer.errors;
425 obcp_live.audio_missed = app.diag.audio_missed;
426 obcp_live.lcd_errors = app.diag.lcd_errors;
427 obcp_live.last_flash_ms = app.diag.last_save_ms;
428 obcp_live.lcd_id = app.diag.tft_id;
429 obcp_live.touch_id = app.diag.touch_id;
430 obcp_live.touch_hits = app.diag.hits;
431 obcp_live.flash_ok = app.diag.flash_ok;
432 obcp_live.lcd_io_ok = lcd_bus_ok;
433 obcp_live.touch_ok = app.diag.touch_ok;
434 obcp_live.touch_down = touch.down;
435 obcp_live.touch_raw_x = app.diag.raw_x;
436 obcp_live.touch_raw_y = app.diag.raw_y;
437 obcp_live.touch_z1 = app.diag.z1;
438 obcp_live.touch_z2 = app.diag.z2;
439 obcp_live.touch_x = app.diag.x;
440 obcp_live.touch_y = app.diag.y;
441 obcp_live.touch_controller = (uint8_t)app.diag.touch_controller;
442 obcp_live.touch_calibrated = touch_cal_get(&app.touch_cal, app.diag.touch_controller) != NULL;
443 obcp_live.tilt_axes = app.tilt.axes;
444 obcp_live.diag_pattern = app.test_color;
445}

447/*
448 * Servir une itération de toutes les tâches de l'application.
449 *
450 * Priorise l'audio, les entrées et au plus quatre pas de simulation de rattrapage. Sert le
451 * rendu puis un travail flash lorsque son message a pu être affiché. À appeler
452 * continuellement depuis USER CODE 3.
453 */
454void obcp_task(void)
455{
456 uint32_t now = HAL_GetTick();
457 pwm_audio_task(&app.audio, app_wants_audio(&app));
458 sample_button(now);
459 sample_inputs(now);
460 unsigned steps = 0;
461 /* Le pas physique reste fixé à 20 ms. Au plus quatre rattrapages évitent qu’un retard bloque indéfiniment le service audio et tactile. */
462 while (now - last_game >= GAME_STEP_MS && steps < 4U) {
463 last_game += GAME_STEP_MS;
464 app_step(&app);
465 ++steps;
466 }
467 if (now - last_game >= GAME_STEP_MS) {
468 last_game = now;
469 }
470 update_rates(now);
471 if (now - last_diag >= 160U) {
472 last_diag = now;
473 app_diag_refresh(&app);
474 }
475 if (lcd_bus_ok) {
476 renderer_task(&renderer, &app.dirty, app_render, &app);
477 }
478 if (app.job == JOB_NONE) {
479 last_flash_job = now;
480 } else if (renderer_idle(&renderer, &app.dirty) || now - last_flash_job >= 450U) {
481 /* Laisser passer le bandeau avant l'effacement. Arrêter l'audio
482 * explicitement avant de bloquer la boucle par l'écriture flash. */
483 pwm_audio_task(&app.audio, false);
484 uint32_t start = HAL_GetTick();
485 StoreResult result = app_store_task(&app, &flash);
486 uint32_t end = HAL_GetTick();
487 app.diag.last_save_ms = end - start;
488 last_game = end;
489 led_signal(result == STORE_OK, end);
490 }
491 led_update(HAL_GetTick());
492}
