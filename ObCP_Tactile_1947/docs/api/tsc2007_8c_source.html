21/** @brief Constantes privées de ce module, avec leurs unités. */
22enum {
23 TSC_ADDRESS = 0x48U << 1, /**< Adresse I²C 7 bits 0x48, décalée d’un bit pour la HAL. */
24 TSC_SLEEP = 0x00, /**< Commande d’arrêt de l’ADC entre deux acquisitions. */
25 TSC_X = 0xC4, /**< Commande de mesure de X sur 12 bits, ADC maintenu actif. */
26 TSC_Y = 0xD4, /**< Commande de mesure de Y sur 12 bits, ADC maintenu actif. */
27 TSC_Z1 = 0xE4, /**< Commande de première mesure liée à la pression. */
28 TSC_Z2 = 0xF4, /**< Commande de seconde mesure liée à la pression. */
29 TSC_TIMEOUT_MS = 2, /**< Délai maximal d’un échange I²C bloquant court, en ms. */
30 TSC_SETTLE_MS = 2, /**< Attente coopérative de stabilisation entre commande et lecture, en ms. */
31 TSC_ADC_MAX = 4095, /**< Pleine échelle d’une conversion 12 bits. */
32};

43/**
44 * @brief Envoyer une commande de conversion TSC2007.
45 *
46 * Vérifie le bit BUSY du bus et borne la transaction HAL à deux millisecondes. L'attente
47 * de conversion est gérée séparément.
48 * @param[in] command Octet de commande du protocole.
49 * @return true si l'octet a été transmis.
50 */
51static bool send_command(uint8_t command)
52{
53 return !__HAL_I2C_GET_FLAG(&hi2c1, I2C_FLAG_BUSY) &&
54 HAL_I2C_Master_Transmit(&hi2c1, TSC_ADDRESS, &command, 1, TSC_TIMEOUT_MS) == HAL_OK;
55}

57/**
58 * @brief Décoder une conversion 12 bits reçue sur I²C.
59 *
60 * La valeur occupe les douze bits de poids fort. Les quatre bits bas doivent être nuls
61 * pour accepter le format de réponse.
62 * @param[out] value Pointeur valide recevant la valeur du registre ou de la conversion.
63 * @return true si deux octets de format attendu sont reçus.
64 */
65static bool read_conversion(uint16_t *value)
66{
67 uint8_t bytes[2];
68 if (__HAL_I2C_GET_FLAG(&hi2c1, I2C_FLAG_BUSY) ||
69 HAL_I2C_Master_Receive(&hi2c1, TSC_ADDRESS, bytes, 2, TSC_TIMEOUT_MS) != HAL_OK ||
70 (bytes[1] & 0x0FU) != 0U) {
71 return false;
72 }
73 /* Le résultat de 12 bits occupe la partie haute des deux octets. */
74 *value = (uint16_t)((uint16_t)bytes[0] << 4 | bytes[1] >> 4);
75 return true;
76}

78/*
79 * Tester le protocole de conversion du TSC2007.
80 *
81 * Remet l'état d'acquisition à zéro, demande TEMP0 avec mise en sommeil et attend une
82 * milliseconde au seul démarrage. Le contrôle suppose un périphérique dédié à 0x48.
83 */
84bool tsc2007_probe(void)
85{
86 adc.index = 0;
87 adc.converting = false;
88 adc.contact = false;
89 adc.sleeping = false;
90 if (!send_command(TSC_SLEEP)) {
91 return false;
92 }
93 /* Ce délai unique concerne la recherche. L'acquisition normale cède
94 * la main pendant chaque conversion pour servir l'audio et le TFT. */
95 HAL_Delay(1);
96 uint16_t temperature;
97 return read_conversion(&temperature) && temperature > 0U && temperature < TSC_ADC_MAX;
98}

100/**
101 * @brief Finaliser une observation et demander le sommeil ADC.
102 *
103 * Prépare l'état de sommeil et la dernière observation, avec X/Y et les deux mesures de
104 * pression. Une erreur de commande empêche de déclarer le résultat valide.
105 * @param[out] sample Structure RawTouch valide recevant l’observation physique.
106 * @param[in] contact État de présence retenu après les mesures de pression.
107 * @return TOUCH_READ_OK ou TOUCH_READ_ERROR.
108 */
109static TouchRead finish(RawTouch *sample, bool contact)
110{
111 adc.index = 0;
112 if (!send_command(TSC_SLEEP)) {
113 return TOUCH_READ_ERROR;
114 }
115 adc.started = HAL_GetTick();
116 adc.sleeping = true;
117 adc.contact = contact;
118 *sample = (RawTouch){{adc.value[2], adc.value[3]}, adc.value[0], adc.value[1], contact};
119 return TOUCH_READ_OK;
120}

122/*
123 * Avancer d'une étape dans l'observation résistive.
124 *
125 * Alterner commande, attente puis lecture permet de servir l'écran et l'audio. Z1 et Z2
126 * décident du contact avec hystérésis. X et Y sont lus seulement quand le contact est
127 * retenu.
128 */
129TouchRead tsc2007_sample(RawTouch *sample)
130{
131 static const uint8_t commands[] = {TSC_Z1, TSC_Z2, TSC_X, TSC_Y};
132 if (adc.sleeping) {
133 /* TEMP0 termine sa conversion avant l'arrêt de l'ADC. */
134 if (HAL_GetTick() - adc.started < TSC_SETTLE_MS) {
135 return TOUCH_READ_WAIT;
136 }
137 adc.sleeping = false;
138 }
139 if (!adc.converting) {
140 if (!send_command(commands[adc.index])) {
141 adc.index = 0;
142 return TOUCH_READ_ERROR;
143 }
144 adc.started = HAL_GetTick();
145 adc.converting = true;
146 return TOUCH_READ_WAIT;
147 }
148 if (HAL_GetTick() - adc.started < TSC_SETTLE_MS) {
149 return TOUCH_READ_WAIT;
150 }
151 adc.converting = false;
152 if (!read_conversion(&adc.value[adc.index])) {
153 adc.index = 0;
154 return TOUCH_READ_ERROR;
155 }
156 ++adc.index;
157 if (adc.index == 2U) {
158 uint16_t z1 = adc.value[0], z2 = adc.value[1];
159 /* Dalle ouverte : Z1 proche de zéro, Z2 à pleine échelle.
160 * L'hystérésis évite les bascules à la limite d'un appui léger. */
161 if (z1 < (adc.contact ? 45U : 80U) || z2 >= 4090U || z2 < z1) {
162 return finish(sample, false);
163 }
164 }
165 if (adc.index < sizeof(commands)) {
166 return TOUCH_READ_WAIT;
167 }
168 return finish(sample, adc.value[2] < TSC_ADC_MAX && adc.value[3] < TSC_ADC_MAX);
169}
