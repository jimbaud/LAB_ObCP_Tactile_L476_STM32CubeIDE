1/**
2 * @file accelerometer.h
3 * @brief Mesures XYZ cohérentes sur le SPI3 de la carte.
4 * @ingroup obcp_drivers
5 *
6 * Le capteur est reconnu par WHO_AM_I, puis configuré à 100 Hz et ±2 g. Le pilote conserve
7 * un pointeur vers la HAL et convertit les six octets XYZ en accélérations exprimées en g.
8 * Appels synchrones depuis la boucle principale.
9 *
10 * @par Repères de lecture
11 * Les coordonnées écran sont en pixels. Les durées suffixées ms sont en
12 * millisecondes. Les adresses du service de stockage sont des offsets en octets.
13 */

21/**
22 * @brief Identités de capteur reconnues par WHO_AM_I.
23 */
24typedef enum {
25 ACCEL_NONE = 0, /**< Aucun capteur compatible configuré. */
26 ACCEL_LIS3DH, /**< WHO_AM_I vaut 0x33, sensibilité haute résolution du LIS3DH. */
27 ACCEL_LIS3DSH, /**< WHO_AM_I vaut 0x3F, sensibilité du LIS3DSH. */
28} accel_chip_t;

37} accel_axis_t;

39/**
40 * @brief Accélération XYZ dans le repère du capteur.

42 * Une carte immobile mesure la pesanteur. Le module de jeu enlève ensuite son neutre aux

45typedef struct {

51/**
52 * @brief Reconnaître et configurer le composant connecté.
53 *
54 * Mémorise le bus, lit WHO_AM_I puis choisit les registres et la sensibilité LIS3DH ou
55 * LIS3DSH. Une erreur de configuration ramène le pilote à ACCEL_NONE.
56 * @param[in,out] hspi Handle HAL du bus SPI déjà initialisé. Pointeur valide.
57 * @return Type reconnu et configuré, ou ACCEL_NONE en cas d'échec.
58 */
59accel_chip_t accel_init(SPI_HandleTypeDef *hspi);

61/**

66 * @return Dernier type enregistré par accel_init().

68accel_chip_t accel_chip(void);

70/**
71 * @brief Obtenir un nom lisible du capteur.
72 *
73 * Le texte appartient au programme en flash. L'appelant peut l'afficher et ne doit pas

77const char *accel_chip_name(void);

79/**
80 * @brief Lire les trois axes dans une même rafale.
81 *
82 * BDU protège la cohérence des registres pendant leur lecture. Chaque paire little-endian
83 * est recomposée en entier signé sur 16 bits.
84 * @param[out] raw Tableau valide de trois entiers signés pour X, Y et Z bruts.
85 * @return true si les six octets sont lus, false si le capteur est absent ou si le SPI échoue.
86 */
87bool accel_read_raw(int16_t raw[3]);

89/**
90 * @brief Convertir la mesure XYZ en g.
91 *
92 * Applique successivement la sensibilité en mg par unité brute, le décalage et l'échelle
93 * d'étalonnage, puis la division par 1000. Cette correction de capteur précède le neutre
94 * des jeux.
95 * @param[out] g Structure de résultat XYZ valide, à remplir en g.
96 * @return true si une mesure valide a été acquise.
97 */
98bool accel_read(accel_g_t *g);

100/**
101 * @brief Lire une indication de température du capteur.
102 *
103 * La formule utilise l'octet signé et le décalage de 25 °C du modèle. Cette indication
104 * sert à observer une variation du capteur. L'application tactile ne l'utilise pas pour
105 * piloter les jeux.
106 * @param[out] celsius Pointeur valide recevant l’indication entière en degrés Celsius.
107 * @return true si le registre du capteur reconnu a été lu.
108 */
109bool accel_read_temperature(int *celsius);

111/**
112 * @brief Étalonner un axe à partir de deux mesures opposées.
113 *
114 * Les mesures de référence correspondent à −1 g et +1 g en mg. Le milieu donne le décalage
115 * et leur différence donne l'échelle. Un axe invalide ou un intervalle inversé est ignoré.
116 * @param[in] axis Axe brut X, Y ou Z à étalonner.
117 * @param[in] min_mg Mesure d’étalonnage quand l’axe voit −1 g, exprimée en mg.
118 * @param[in] max_mg Mesure d’étalonnage quand l’axe voit +1 g, en mg, supérieure à min_mg.
119 */
120void accel_set_calibration(accel_axis_t axis, float min_mg, float max_mg);
