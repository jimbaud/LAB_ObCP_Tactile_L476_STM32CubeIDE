3 * @brief Canevas indexé, diamètres de trait et palette persistante.
4 * @ingroup obcp_apps
5 *
6 * Deux pixels tiennent dans un octet. Les seize indices désignent la palette du dessin. La
7 * case zéro sert au fond et à la gomme. Un dessin enrichi ajoute 36 octets aux pixels, et
8 * le lecteur accepte aussi le format historique de 32 000 octets.

39/*
40 * Initialiser un canevas, une palette et le crayon.
41 *
42 * Met les pixels à l'indice zéro, copie la palette de départ, choisit l'encre huit et le
43 * diamètre cinq pixels. Le dessin n'est pas marqué modifié à cette étape.

45void paint_init(Paint *p)

47 memset(p, 0, sizeof(*p));

49 p->ink = 8;

66/*
67 * Lire l'indice de couleur d'un pixel.
68 *
69 * Calcule l'octet commun à deux abscisses, choisit le demi-octet bas ou haut puis applique
70 * le masque 15. Une coordonnée hors canevas donne l'indice de fond.
71 */
72uint8_t paint_get(const Paint *p, int x, int y)
73{
74 if (!rect_contains(canvas, x, y)) {
75 return 0;
76 }
77 /* Deux abscisses voisines partagent un octet. X pair utilise les bits 0 à 3, X impair les bits 4 à 7. */
78 size_t offset = (size_t)(y * LCD_WIDTH + x) / 2U;
79 unsigned shift = (x & 1) == 0 ? 0U : 4U;
80 return (uint8_t)((p->image.pixels[offset] >> shift) & 15U);
81}

83/**
84 * @brief Modifier le demi-octet d'un seul pixel.
85 *
86 * Le masque efface uniquement les quatre bits du pixel choisi et conserve son voisin. Le
87 * test du canevas protège aussi les disques tracés près d'un bord.
88 * @param[in,out] p Canevas et palette du dessin. Pointeur valide.
89 * @param[in] x Abscisse dans le repère écran, en pixels.
90 * @param[in] y Ordonnée dans le repère écran, en pixels.
91 * @param[in] index Indice de palette compris entre zéro et quinze.
92 */
93static void plot(Paint *p, int x, int y, uint8_t index)
94{
95 if (!rect_contains(canvas, x, y)) {
96 return;
97 }
98 /* Deux abscisses voisines partagent un octet. X pair utilise les bits 0 à 3, X impair les bits 4 à 7. */
99 size_t offset = (size_t)(y * LCD_WIDTH + x) / 2U;
100 unsigned shift = (x & 1) == 0 ? 0U : 4U;
101 /* Le masque 0x0F ou 0xF0 protège le demi-octet du voisin pendant la modification. */
102 uint8_t mask = (uint8_t)(15U << shift);
103 p->image.pixels[offset] = (uint8_t)((p->image.pixels[offset] & (uint8_t)~mask) | (index << shift));
104}

106/**
107 * @brief Déposer un disque de crayon ou de gomme.
108 *
109 * Parcourt une petite emprise carrée et écrit les pixels dont la distance au centre ne
110 * dépasse pas le rayon. plot() garde le disque à l'intérieur du canevas.
111 * @param[in,out] p Canevas et palette du dessin. Pointeur valide.
112 * @param[in] x Abscisse dans le repère écran, en pixels.
113 * @param[in] y Ordonnée dans le repère écran, en pixels.
114 * @param[in] radius Rayon entier positif ou nul, en pixels.
115 * @param[in] index Indice de palette compris entre zéro et quinze.
116 */
117static void dab(Paint *p, int x, int y, int radius, uint8_t index)
118{
119 for (int dy = -radius; dy <= radius; ++dy) {
120 for (int dx = -radius; dx <= radius; ++dx) {
121 if (dx * dx + dy * dy <= radius * radius) {
122 plot(p, x + dx, y + dy, index);
123 }
124 }
125 }
126}

128/*
129 * Relier deux relevés tactiles par un trait continu.
130 *
131 * Bresenham choisit des centres entiers successifs et dab() y applique le diamètre
132 * courant. Le rectangle renvoyé couvre les deux extrémités et l'épaisseur du trait.
133 */
134Rect paint_segment(Paint *p, int x0, int y0, int x1, int y1)
135{
136 if (!rect_contains(canvas, x0, y0) || !rect_contains(canvas, x1, y1)) {
137 return (Rect){0};
138 }
139 int radius = PAINT_DIAMETERS[p->brush] / 2;
140 uint8_t index = p->erase ? 0U : (uint8_t)(p->ink & 15U);
141 Rect from = {(int16_t)(x0 - radius), (int16_t)(y0 - radius),
142 (int16_t)(2 * radius + 1), (int16_t)(2 * radius + 1)};
143 Rect to = {(int16_t)(x1 - radius), (int16_t)(y1 - radius), from.w, from.h};
144
145 /* Bresenham relie deux relevés tactiles par des pixels entiers. */
146 int dx = abs(x1 - x0);
147 int dy = -abs(y1 - y0);
148 int sx = x0 < x1 ? 1 : -1;
149 int sy = y0 < y1 ? 1 : -1;
150 int error = dx + dy;
151 for (;;) {
152 dab(p, x0, y0, radius, index);
153 if (x0 == x1 && y0 == y1) {
154 break;
155 }
156 /* L’erreur cumulée décide des avances sur X et Y. Les deux tests peuvent être vrais pour un pas diagonal. */
157 int twice = 2 * error;
158 if (twice >= dy) {
159 error += dy;
160 x0 += sx;
161 }
162 if (twice <= dx) {
163 error += dx;
164 y0 += sy;
165 }
166 }
167 p->modified = true;
168 return rect_intersection(rect_union(from, to), canvas);
169}

171/*
172 * Convertir les indices du canevas vers RGB565.
173 *
174 * Lit chaque indice dans la seule intersection entre canevas et bande, puis consulte la
175 * palette attachée au dessin. Modifier une case recolore donc tous les pixels de cet
176 * indice.
177 */
178void paint_draw(const Paint *p, Gfx *g)
179{
180 Rect area = rect_intersection(canvas, g->clip);
181 for (int y = area.y; y < area.y + area.h; ++y) {
182 for (int x = area.x; x < area.x + area.w; ++x) {
183 gfx_pixel(g, x, y, p->palette[paint_get(p, x, y)]);
184 }
185 }
186}

203/*
204 * Encoder la palette et sauvegarder le dessin enrichi.
205 *
206 * Ajoute signature PT, version, choix de diamètre et couleurs little-endian, puis confie
207 * l'image complète au service A/B.
208 */
209StoreResult paint_save(Paint *p, Storage *storage, SaveObject object)
210{
211 /* Le format sur flash est écrit octet par octet. Il ne dépend pas de l’ordre mémoire des uint16_t du processeur. */
212 uint8_t *data = p->image.metadata;
213 data[0] = 'P';
214 data[1] = 'T';
215 data[2] = 1;
216 data[3] = p->brush;
217 for (unsigned i = 0; i < 16U; ++i) {
218 data[4U + i * 2U] = (uint8_t)p->palette[i];
219 data[5U + i * 2U] = (uint8_t)(p->palette[i] >> 8);
220 }
221 return storage_save(storage, object, &p->image, sizeof(p->image));
222}

224/*
225 * Lire un dessin historique ou enrichi et sa palette.
226 *
227 * La longueur permet de distinguer les pixels historiques de PaintImage. Le format enrichi
228 * est validé avant de décoder les couleurs. Un succès termine le geste et retire
229 * l'indicateur modifié.
230 */
231StoreResult paint_load(Paint *p, Storage *storage, SaveObject object)
232{
233 size_t size = 0;
234 StoreResult result = storage_read_sized(storage, object, &p->image, sizeof(p->image), &size);
235 if (result != STORE_OK) {
236 return result;
237 }
238 if (size == PAINT_BYTES) {
239 /* Les dessins 0.1 à 0.3 utilisaient la palette fixe d'origine. */
240 memcpy(p->palette, PAINT_PALETTE, sizeof(p->palette));
241 p->brush = 2;
242 } else {
243 const uint8_t *data = p->image.metadata;
244 if (size != sizeof(p->image) || data[0] != 'P' || data[1] != 'T' ||
245 data[2] != 1U || data[3] >= PAINT_BRUSHES) {
246 return STORE_INVALID;
247 }
248 p->brush = data[3];
249 for (unsigned i = 0; i < 16U; ++i) {
250 p->palette[i] = (uint16_t)(data[4U + i * 2U] | (uint16_t)data[5U + i * 2U] << 8);
251 }
252 }
253 p->stroke = p->modified = false;
254 return STORE_OK;
255}
