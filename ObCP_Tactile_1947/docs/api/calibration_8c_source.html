38const RawPoint TOUCH_CAL_TARGETS[CAL_TARGET_COUNT] = {
39 {26, 60}, {LCD_WIDTH - 27, 60},
40 {26, LCD_HEIGHT - 27}, {LCD_WIDTH - 27, LCD_HEIGHT - 27}

43/*
44 * Calculer une transformation affine de trois points bruts.
45 *
46 * Le déterminant de deux vecteurs permet d'inverser la base. Un déterminant trop petit
47 * rejette des observations alignées ou trop proches. Le candidat remplace cal seulement si
48 * le calcul est admis.
49 */
50bool touch_cal_fit(TouchCalibration *cal, const RawPoint p[CAL_FIT_COUNT])
51{
52 for (unsigned i = 0; i < CAL_FIT_COUNT; ++i) {
53 if (p[i].x > CAL_RAW_MAX || p[i].y > CAL_RAW_MAX) {
54 return false;
55 }
56 }
57 int32_t dx1 = (int32_t)p[1].x - p[0].x, dy1 = (int32_t)p[1].y - p[0].y;
58 int32_t dx2 = (int32_t)p[2].x - p[0].x, dy2 = (int32_t)p[2].y - p[0].y;
59 /* Le déterminant mesure l’aire orientée des deux vecteurs. Près de zéro, l’inversion amplifierait les erreurs. */
60 int32_t det = dx1 * dy2 - dx2 * dy1;
61 if (det > -CAL_MIN_DET && det < CAL_MIN_DET) {
62 return false;
63 }
64
65 /* Inverser la base (P1 - P0, P2 - P0) et passer de cette base
66 * aux pixels. Un déterminant négatif représente un axe inversé. */
67 const RawPoint *target = TOUCH_CAL_TARGETS;
68 float sx = (float)(target[1].x - target[0].x) / (float)det;
69 float sy = (float)(target[2].y - target[0].y) / (float)det;
70 TouchCalibration fit = {
71 .x_from_x = sx * (float)dy2, .x_from_y = -sx * (float)dx2,
72 .y_from_x = -sy * (float)dy1, .y_from_y = sy * (float)dx1,
73 .valid = true
74 };
75 fit.x_offset = (float)target[0].x - fit.x_from_x * p[0].x - fit.x_from_y * p[0].y;
76 fit.y_offset = (float)target[0].y - fit.y_from_x * p[0].x - fit.y_from_y * p[0].y;
77 memcpy(fit.points, p, sizeof(fit.points));
78 /* Publier le résultat seulement après validation. Une tentative refusée conserve le précédent profil. */
79 *cal = fit;
80 return true;
81}

83/**
84 * @brief Arrondir et borner une coordonnée calculée.
85 *
86 * Les valeurs sous zéro et au-delà du dernier pixel sont saturées. L'ajout de 0,5 réalise
87 * l'arrondi avant la conversion en entier.
88 * @param[in] value Valeur à utiliser dans ce traitement.
89 * @param[in] maximum Dernière coordonnée admise dans le résultat.
90 * @return Coordonnée comprise entre zéro et maximum.
91 */
92static int16_t pixel(float value, int maximum)
93{
94 if (value <= 0.0f) {
95 return 0;
96 }
97 if (value >= (float)maximum) {
98 return (int16_t)maximum;
99 }
100 return (int16_t)(value + 0.5f);
101}

103/*
104 * Transformer un point brut en pixels paysage.
105 *
106 * Vérifie le profil et les bornes des conversions 12 bits, applique les deux expressions
107 * affines puis borne les résultats à l'écran.
108 */
109bool touch_cal_map(const TouchCalibration *cal, RawPoint raw, int16_t *x, int16_t *y)
110{
111 if (cal == NULL || !cal->valid || raw.x > CAL_RAW_MAX || raw.y > CAL_RAW_MAX) {
112 return false;
113 }
114 *x = pixel(cal->x_from_x * raw.x + cal->x_from_y * raw.y + cal->x_offset, LCD_WIDTH - 1);
115 *y = pixel(cal->y_from_x * raw.x + cal->y_from_y * raw.y + cal->y_offset, LCD_HEIGHT - 1);
116 return true;
117}

142/*
143 * Faire avancer l'acquisition d'une cible résistive.
144 *
145 * Six mesures proches sont moyennées, puis validées au relâchement. Les trois premières
146 * cibles construisent la transformation. La quatrième vérifie l'erreur et peut relancer
147 * toute la session.
148 */
149TouchCalEvent touch_cal_sample(TouchCalSession *s, RawTouch raw)
150{
151 if (s->wait_release) {
152 s->wait_release = raw.down;
153 return raw.down ? CAL_UNCHANGED : CAL_READY;
154 }
155 if (raw.down) {
156 if (s->samples >= CAL_STABLE_SAMPLES) {
157 return CAL_UNCHANGED;
158 }
159 if (s->samples == 0U || abs((int)raw.point.x - s->anchor.x) > CAL_MAX_DRIFT ||
160 abs((int)raw.point.y - s->anchor.y) > CAL_MAX_DRIFT) {
161 /* Un déplacement important recommence la moyenne sur la même cible, sans valider un point instable. */
162 s->anchor = raw.point;
163 s->sum_x = s->sum_y = 0;
164 s->samples = 0;
165 }
166 s->sum_x += raw.point.x;
167 s->sum_y += raw.point.y;
168 return ++s->samples == CAL_STABLE_SAMPLES ? CAL_HELD : CAL_UNCHANGED;
169 }
170
171 /* Passer à la cible suivante au relâchement. Un contact bref ou
172 * instable laisse la cible en attente. */
173 bool held = s->samples == CAL_STABLE_SAMPLES;
174 s->samples = 0;
175 if (!held) {
176 return CAL_UNCHANGED;
177 }
178 RawPoint p = {(uint16_t)(s->sum_x / CAL_STABLE_SAMPLES),
179 (uint16_t)(s->sum_y / CAL_STABLE_SAMPLES)};
180 if (s->target < CAL_FIT_COUNT) {
181 s->observations[s->target++] = p;
182 s->retry = false;
183 if (s->target < CAL_FIT_COUNT || touch_cal_fit(&s->candidate, s->observations)) {
184 return CAL_ADVANCED;
185 }
186 } else {
187 /* Cette quatrième cible n’a participé à aucun coefficient. Son erreur constitue un contrôle indépendant. */
188 const RawPoint expected = TOUCH_CAL_TARGETS[CAL_FIT_COUNT];
189 int16_t x, y;
190 if (touch_cal_map(&s->candidate, p, &x, &y) &&
191 abs((int)x - expected.x) <= CAL_MAX_ERROR_PX &&
192 abs((int)y - expected.y) <= CAL_MAX_ERROR_PX) {
193 return CAL_COMPLETE;
194 }
195 }
196 *s = (TouchCalSession){.retry = true};
197 return CAL_RETRY;
198}

200/**
201 * @brief Associer un contrôleur résistif à sa place dans le livre.
202 *
203 * Le TSC2007 et le STMPE610 ont chacun un profil. Le capacitif utilise déjà des
204 * coordonnées natives et n'occupe pas de profil résistif.
205 * @param[in] controller Famille de contrôleur tactile.
206 * @return Zéro pour TSC2007, un pour STMPE610, moins un sinon.
207 */
208static int profile_index(TouchController controller)
209{
210 if (controller == TOUCH_CTRL_TSC2007) {
211 return 0;
212 }
213 if (controller == TOUCH_CTRL_STMPE610) {
214 return 1;
215 }
216 return -1;
217}

245/*
246 * Charger et reconstruire les deux profils résistifs.
247 *
248 * Commence par un livre vide. Vérifie la version et l'orientation du TFT, décode les
249 * points little-endian puis recalcule les coefficients.
250 */
251StoreResult touch_cal_load(TouchCalBook *book, Storage *storage, uint8_t orientation)
252{
253 *book = (TouchCalBook){0};
254 uint8_t data[CAL_WIRE_BYTES];
255 StoreResult result = storage_read(storage, SAVE_TOUCH_CAL, data, sizeof(data));
256 if (result != STORE_OK) {
257 return result;
258 }
259 if (data[0] != CAL_VERSION || data[1] != orientation) {
260 return STORE_INVALID;
261 }
262 const uint8_t *read = data + 2;
263 for (unsigned i = 0; i < 2U; ++i) {
264 RawPoint p[CAL_FIT_COUNT];
265 for (unsigned j = 0; j < CAL_FIT_COUNT; ++j) {
266 p[j].x = (uint16_t)(read[0] | (uint16_t)read[1] << 8);
267 p[j].y = (uint16_t)(read[2] | (uint16_t)read[3] << 8);
268 read += 4;
269 }
270 (void)touch_cal_fit(&book->profile[i], p);
271 }
272 return STORE_OK;
273}

275/*
276 * Encoder puis enregistrer les deux profils.
277 *
278 * Le format contient version, orientation et observations brutes. Un profil absent est
279 * encodé par des points nuls. Le service A/B fournit la validation et la résistance aux
280 * coupures.
281 */
282StoreResult touch_cal_save(const TouchCalBook *book, Storage *storage, uint8_t orientation)
283{
284 /* Encoder les observations en little-endian, indépendamment du
285 * remplissage des structures C et du format des flottants. */
286 uint8_t data[CAL_WIRE_BYTES] = {CAL_VERSION, orientation};
287 uint8_t *write = data + 2;
288 for (unsigned i = 0; i < 2U; ++i) {
289 const TouchCalibration *cal = &book->profile[i];
290 for (unsigned j = 0; j < CAL_FIT_COUNT; ++j) {
291 if (cal->valid) {
292 RawPoint p = cal->points[j];
293 write[0] = (uint8_t)p.x;
294 write[1] = (uint8_t)(p.x >> 8);
295 write[2] = (uint8_t)p.y;
296 write[3] = (uint8_t)(p.y >> 8);
297 }
298 write += 4;
299 }
300 }
301 return storage_save(storage, SAVE_TOUCH_CAL, data, sizeof(data));
302}
